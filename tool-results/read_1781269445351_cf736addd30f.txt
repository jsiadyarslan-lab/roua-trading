   141→  /**
   142→   * Place a new order — Unified V1/V2 pipeline
   143→   * POST /api/trading/orders
   144→   *
   145→   * #18 UNIFIED: Tries V2 pipeline first (idempotency + state manager + BullMQ)
   146→   * when V2 services are available. Falls back to V1 pipeline if V2 is down.
   147→   *
   148→   * V2 pipeline flow:
   149→   * 1. Validate request body
   150→   * 2. Check idempotencyKey → 409 if duplicate
   151→   * 3. Create OrderCommand & register in OrderStateManager (PENDING)
   152→   * 4. Run RiskGatekeeperService checks
   153→   * 5. Update to ACCEPTED & send to execution queue
   154→   * 6. Return { orderId, status }
   155→   *
   156→   * V1 pipeline (fallback):
   157→   * 1. Validate + RiskGatekeeper
   158→   * 2. TradingService.placeOrder (direct execution)
   159→   */
   160→  @Post('orders')
   161→  @Throttle({ medium: { limit: 10, ttl: 60000 } })
   162→  async placeOrder(@Req() req: any, @Body() body: PlaceOrderDto, @Res({ passthrough: true }) res: Response) {
   163→    const userId = req.user.id;
   164→
   165→    // ── Shared validation (used by both V1 and V2 pipelines) ──
   166→    const request: PlaceOrderRequest = {
   167→      credentialId: body.credentialId,
   168→      symbol: body.symbol,
   169→      side: body.side as OrderSide,
   170→      type: body.type as OrderType,
   171→      quantity: Number(body.quantity),
   172→      price: body.price != null ? Number(body.price) : undefined,
   173→      stopLoss: body.stopLoss != null ? Number(body.stopLoss) : undefined,
   174→      takeProfit: body.takeProfit != null ? Number(body.takeProfit) : undefined,
   175→      signalId: body.signalId,
   176→    };
   177→
   178→    // Validate required fields
   179→    if (
   180→      !request.credentialId ||
   181→      !request.symbol ||
   182→      !request.side ||
   183→      !request.type ||
   184→      !request.quantity
   185→    ) {
   186→      throw new BadRequestException(
   187→        'بيانات الطلب غير مكتملة — يرجى تعبئة جميع الحقول المطلوبة',
   188→      );
   189→    }
   190→
   191→    if (!['BUY', 'SELL'].includes(request.side)) {
   192→      throw new BadRequestException(
   193→        'جانب الطلب يجب أن يكون BUY أو SELL',
   194→      );
   195→    }
   196→
   197→    if (!['MARKET', 'LIMIT'].includes(request.type)) {
   198→      throw new BadRequestException('نوع الطلب غير صالح');
   199→    }
   200→
   201→    if (request.quantity <= 0) {
   202→      throw new BadRequestException(
   203→        'الكمية يجب أن تكون أكبر من صفر',
   204→      );
   205→    }
   206→
   207→    // ── Stop-loss is MANDATORY (enforced by RiskGatekeeper check #1) ──
   208→    if (!body.stopLoss || Number(body.stopLoss) <= 0) {
   209→      throw new BadRequestException(
   210→        'وقف الخسارة إجباري. لا يمكن تقديم أمر بدون وقف خسارة — هذا القانون الأول في منصة رؤى.',
   211→      );
   212→    }
   213→
   214→    // ── LIMIT orders require a price (#18: added from V2 validation) ──
   215→    if (request.type === 'LIMIT' && !request.price) {
   216→      throw new BadRequestException('سعر الحد مطلوب للطلبات المحددة (LIMIT)');
   217→    }
   218→
   219→    // ═══════════════════════════════════════════════════════════════
   220→    // #18 UNIFIED: Try V2 pipeline first when V2 services are available
   221→    // ═══════════════════════════════════════════════════════════════
   222→    if (this._isV2Available()) {
   223→      return this._placeOrderV2(userId, body, request, req, res);
   224→    }
   225→
   226→    // ═══════════════════════════════════════════════════════════════
   227→    // V1 FALLBACK (DEPRECATED) — used when V2 services are unavailable
   228→    // ═══════════════════════════════════════════════════════════════
   229→    this.logger.warn('⚠️ V2 pipeline unavailable — falling back to V1 (deprecated)');
   230→    return this._placeOrderV1(userId, request, req, res);
   231→  }

   233→  /**
   234→   * #18: V2 order pipeline — Idempotency → State Manager → Risk → Queue
   235→   * Extracted from OrderController for unified access.
   236→   */
   237→  private async _placeOrderV2(
   238→    userId: string,
   239→    body: PlaceOrderDto,
   240→    request: PlaceOrderRequest,
   241→    req: any,
   242→    res: Response,
   243→  ) {
   244→    // ── Step 1: Check idempotency (if key provided) ──
   245→    const idempotencyKey = (body as any).idempotencyKey ||
   246→      `v1-${userId}-${request.symbol}-${request.side}-${request.type}-${request.quantity}-${request.price || 'market'}-${Math.floor(Date.now() / 1000)}`;
   247→
   248→    const isUnique = await this.idempotencyService!.checkAndLock(idempotencyKey);
   249→    if (!isUnique) {
   250→      throw new ConflictException(
   251→        'تم استلام هذا الطلب مسبقاً. لا يمكن تكرار نفس idempotencyKey خلال 24 ساعة.',
   252→      );
   253→    }
   254→
   255→    // ── Step 2: Build OrderCommand ──
   256→    const command: OrderCommand = {
   257→      userId,
   258→      exchangeCredentialId: request.credentialId,
   259→      symbol: request.symbol,
   260→      side: request.side === 'BUY' ? OrderSideEnum.BUY : OrderSideEnum.SELL,
   261→      type: request.type === 'MARKET' ? OrderTypeEnum.MARKET : OrderTypeEnum.LIMIT,
   262→      quantity: request.quantity,
   263→      price: request.price,
   264→      stopLoss: request.stopLoss ?? 0,
   265→      takeProfit: request.takeProfit,
   266→      idempotencyKey,
   267→      clientOrderId: (body as any).clientOrderId,
   268→      ipAddress: req.ip,
   269→      userAgent: req.headers['user-agent'],
   270→    };
   271→
   272→    // ── Step 3: Create order in PENDING state ──
   273→    let order: any;
   274→    try {
   275→      order = await this.stateManager!.createOrder(command);
   276→    } catch (error: any) {
   277→      await this.idempotencyService!.releaseLock(idempotencyKey);
   278→      throw error;
   279→    }
   280→
   281→    // ── Step 4: Run risk checks ──
   282→    const riskResult = await this.riskGatekeeper.validateOrder(command);
   283→
   284→    if (!riskResult.allowed) {
   285→      await this.stateManager!.rejectOrder(
   286→        order.id,
   287→        riskResult.reason || 'فشل في فحص المخاطر',
   288→        riskResult.failedCheck,
   289→      );
   290→      await this.idempotencyService!.releaseLock(idempotencyKey);
   291→      throw new ForbiddenException(
   292→        `🛡️ تم رفض الطلب: ${riskResult.reason}`,
   293→      );
   294→    }
   295→
   296→    // ── Step 5: Update status to ACCEPTED ──
   297→    await this.stateManager!.updateOrderStatus(order.id, 'ACCEPTED', {
   298→      riskScore: riskResult.riskScore,
   299→      validatedAt: new Date().toISOString(),
   300→    });
   301→
   302→    // ── Step 6: Send to execution queue ──
   303→    const queueMessage = {
   304→      orderId: order.id,
   305→      userId: command.userId,
   306→      exchangeCredentialId: command.exchangeCredentialId,
   307→      symbol: command.symbol,
   308→      side: command.side,
   309→      type: command.type,
   310→      quantity: command.quantity,
   311→      price: command.price,
   312→      stopLoss: command.stopLoss,
   313→      takeProfit: command.takeProfit,
   314→      clientOrderId: command.clientOrderId,
   315→      idempotencyKey: command.idempotencyKey,
   316→      submittedAt: new Date(),
   317→    };
   318→
   319→    try {
   320→      await this.orderProducer!.sendOrder(queueMessage);
   321→      this.logger.log(`📤 Order ${order.id} submitted to RabbitMQ order_queue`);
   322→    } catch (rabbitError: any) {
   323→      this.logger.warn(
   324→        `RabbitMQ failed for ${order.id}: ${rabbitError.message} — trying BullMQ fallback`,
   325→      );
   326→      try {
   327→        if (!this.executionQueue) {
   328→          throw new Error('BullMQ execution_queue not available');
   329→        }
   330→        await this.executionQueue.add('execute', queueMessage, {
   331→          jobId: command.idempotencyKey,
   332→          attempts: 3,
   333→          backoff: { type: 'exponential', delay: 5000 },
   334→        });
   335→        this.logger.log(`📤 Order ${order.id} added to BullMQ execution_queue (fallback)`);
   336→      } catch (bullError: any) {
   337→        this.logger.error(
   338→          `Both queues failed for order ${order.id}. Order is ACCEPTED but not submitted. ` +
   339→          `RabbitMQ: ${rabbitError.message}, BullMQ: ${bullError.message}`,
   340→        );
   341→      }
   342→    }
   343→
   344→    // ── Step 7: Return V2 response ──
   345→    return {
   346→      success: true,
   347→      data: {
   348→        orderId: order.id,
   349→        status: 'ACCEPTED',
   350→        idempotencyKey,
   351→        riskScore: riskResult.riskScore,
   352→        pipeline: 'v2',
   353→      },
   354→    };
   355→  }

   357→  /**
   358→   * #18: V1 order pipeline (DEPRECATED) — direct TradingService execution
   359→   * Only used when V2 services (idempotency, state manager) are unavailable.
   360→   */
   361→  private async _placeOrderV1(
   362→    userId: string,
   363→    request: PlaceOrderRequest,
   364→    req: any,
   365→    res: Response,
   366→  ) {
   367→    // ── Risk Gatekeeper Check ──
   368→    const riskResult = await this.riskGatekeeper.validateOrder({
   369→      userId,
   370→      exchangeCredentialId: request.credentialId,
   371→      symbol: request.symbol,
   372→      side: request.side as PrismaOrderSide,
   373→      type: request.type as PrismaOrderType,
   374→      quantity: request.quantity,
   375→      price: request.price,
   376→      stopLoss: request.stopLoss!,
   377→      idempotencyKey: `v1-${userId}-${request.symbol}-${request.side}-${request.type}-${request.quantity}-${request.price || 'market'}-${Math.floor(Date.now() / 1000)}`,
   378→    });
   379→
   380→    if (!riskResult.allowed) {
   381→      throw new ForbiddenException(
   382→        `🛡️ تم رفض الطلب: ${riskResult.reason || 'فشل في فحص المخاطر'}`,
   383→      );
   384→    }
   385→
   386→    // Set skipRiskCheck=true because RiskGatekeeper already validated above.
   387→    request.skipRiskCheck = true;
   388→
   389→    // Mark V1 as deprecated in response headers
   390→    this._addDeprecationHeaders(res);
   391→
   392→    const result = await this.tradingService.placeOrder(
   393→      userId,
   394→      request,
   395→      req.ip,
   396→      req.headers['user-agent'],
   397→    );
   398→
   399→    // Add pipeline indicator to the response
   400→    if (result && typeof result === 'object') {
   401→      (result as any).pipeline = 'v1-deprecated';
   402→    }
   403→    return result;
   404→  }

   406→  /**
   407→   * Cancel an order — Unified V1/V2 pipeline

   412→  @Delete('orders/:id')

   414→  async cancelOrder(@Req() req: any, @Param('id') orderId: string) {
   415→    // ── V2 path: State manager with event sourcing ──
   416→    if (this.stateManager) {
   417→      const order = await this.stateManager.findOrderById(orderId);
   418→
   419→      if (!order) {
   420→        throw new NotFoundException('الطلب غير موجود');
   421→      }
   422→
   423→      if (order.userId !== req.user.id) {
   424→        throw new ForbiddenException('ليس لديك صلاحية إلغاء هذا الطلب');
   425→      }
   426→
   427→      if (!['PENDING', 'ACCEPTED'].includes(order.status)) {
   428→        throw new BadRequestException(
   429→          `لا يمكن إلغاء طلب بحالة "${order.status}"`,
   430→        );
   431→      }
   432→
   433→      await this.stateManager.updateOrderStatus(orderId, 'CANCELLED', {
   434→        cancelledBy: req.user.id,
   435→        cancelledAt: new Date().toISOString(),
   436→      });
   437→
   438→      return {
   439→        success: true,
   440→        data: { orderId, status: 'CANCELLED' },
   441→      };
   442→    }
   443→
   444→    // ── V1 fallback (deprecated) ──
   445→    return this.tradingService.cancelOrder(
   446→      req.user.id,
   447→      orderId,
   448→      req.ip,
   449→      req.headers['user-agent'],

   562→      );
   563→      // Map Prisma Position model to frontend-friendly trade format
   564→      const trades = (Array.isArray(positions) ? positions : []).map((p: any) => {
   565→        const lastTrade = p.trades?.length > 0 ? p.trades[p.trades.length - 1] : null;
   566→        return {
   567→          id: p.id,
   568→          symbol: p.symbol,
   569→          side: p.side?.toLowerCase() === 'buy' ? 'long' : 'short',
   570→          entryPrice: Number(p.entryPrice) || 0,
   571→          exitPrice: lastTrade ? Number(lastTrade.price) : (Number(p.exitPrice) || 0),
   572→          qty: Number(p.quantity) || 0,
   573→          realizedPnl: Number(p.realizedPnl) || 0,
   574→          realizedPct: p.entryPrice > 0 ? ((Number(p.realizedPnl) || 0) / (Number(p.entryPrice) * Number(p.quantity))) * 100 : 0,
   575→          closeTime: p.closedAt ? new Date(p.closedAt).getTime() : Date.now(),
   576→          status: p.status,
   577→        };
   578→      });
   579→      return { success: true, trades };
   580→    } catch (error: any) {
   581→      this.logger.error(
   582→        `❌ Failed to fetch trading history: ${error.message}`,
   583→        error.stack,
   584→      );
   585→      throw error;
   586→    }
   587→  }
