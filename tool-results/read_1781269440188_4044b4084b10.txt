    18→/**
    19→ * OrderExecutorService — Executes trades with safety and precision
    20→ *
    21→ * Handles the final step of the trading pipeline:
    22→ * 1. Validates the signal one last time before execution
    23→ * 2. Places the order through the TradingService
    24→ * 3. Handles slippage and partial fills
    25→ * 4. Records the execution result
    26→ * 5. Creates a full audit trail
    27→ *
    28→ * Execution Pipeline:
    29→ * ┌─────────────────────────────────────────────────────────────┐
    30→ * │ 1. Pre-flight check (signal + risk still valid)             │
    31→ * │ 2. Construct order request with idempotency key             │
    32→ * │ 3. Execute via TradingService.placeOrder()                  │
    33→ * │ 4. Record execution result + audit trail                    │
    34→ * │ 5. Update Redis agent state                                 │
    35→ * │ 6. Return TradeExecution result                             │
    36→ * └─────────────────────────────────────────────────────────────┘
    37→ *
    38→ * Safety Features:
    39→ * - Idempotency key prevents duplicate orders
    40→ * - Mandatory stop-loss on every order
    41→ * - Slippage tolerance check
    42→ * - Execution timeout protection
    43→ * - Full audit trail for every decision
    44→ */
    45→@Injectable()
    46→export class OrderExecutorService implements OnModuleDestroy {
    47→  private readonly logger = new Logger(OrderExecutorService.name);
    48→
    49→  /** Slippage tolerance: reject if actual price deviates more than this % */
    50→  private readonly MAX_SLIPPAGE_PERCENT = 1.0;
    51→
    52→  /** Track executed orders for deduplication */
    53→  private readonly recentOrders = new Map<string, Date>();
    54→
    55→  /** Reference to the cleanup interval for proper disposal */
    56→  private cleanupInterval: ReturnType<typeof setInterval> | null = null;
    57→
    58→  // FIX: Dependencies from forwardRef modules are marked @Optional() to prevent
    59→  // NestJS crash if forwardRef resolution fails. When null, the execute method
    60→  // returns an error instead of crashing the entire AutonomousTraderAgentModule.
    61→  constructor(
    62→    private readonly prisma: PrismaService,
    63→    private readonly redis: RedisService,

    68→    @Optional() private readonly exchangeService: ExchangeService,
    69→    @Optional() private readonly credentialsService: CredentialsService,
    70→    @Optional() private readonly tradeCoordination: TradeCoordinationService,
    71→  ) {
    72→    this.logger.log('⚡ Order Executor initialized — safe execution ready');
    73→
    74→    // Clean up old order entries every 5 minutes
    75→    this.cleanupInterval = setInterval(() => this._cleanupOldOrders(), 5 * 60 * 1000);
    76→  }

    83→    }
    84→  }

    86→  /**
    87→   * Execute a validated signal as a live trade
    88→   */
    89→  async execute(
    90→    userId: string,
    91→    signal: EvaluatedSignal,
    92→    risk: RiskAssessment,
    93→    credentialId: string,
    94→  ): Promise<TradeExecution> {
    95→    const startTime = Date.now();
    96→    const idempotencyKey = `${userId}-${signal.id}`;
    97→    let coordinationLockAcquired = false; // #18: Track if coordination lock was acquired
    98→
    99→    // FIX: Guard against missing forwardRef dependencies
   100→    if (!this.orderDispatcher) {
   101→      return {
   102→        success: false,
   103→        error: 'نظام التنفيذ غير متاح حالياً — يرجى المحاولة لاحقاً',
   104→        executionTimeMs: Date.now() - startTime,
   105→      };
   106→    }
   107→
   108→    this.logger.log(
   109→      `⚡ Executing: ${signal.action} ${risk.positionSize.toFixed(6)} ${signal.symbol} ` +
   110→      `@ ${signal.entryPrice} (SL: ${signal.stopLoss}, TP: ${signal.takeProfit})`,
   111→    );
   112→
   113→    try {
   114→      // ═══════════════════════════════════════════════════════════════
   115→      // #18: Trade Coordination — prevent Agent and SmartExecutor conflicts.
   116→      // Checks if ANY source has an open position on this symbol, and
   117→      // acquires a distributed lock to prevent race conditions.
   118→      // FAIL-OPEN: If coordination service is unavailable, trade proceeds.
   119→      // ═══════════════════════════════════════════════════════════════
   120→      if (this.tradeCoordination) {
   121→        try {
   122→          const coordination = await this.tradeCoordination.canOpenPosition(userId, signal.symbol, 'agent');
   123→          if (!coordination.allowed) {
   124→            this.logger.warn(
   125→              `⚡ ORDER REJECTED (coordination): ${coordination.reason}`,
   126→            );
   127→            return {
   128→              success: false,
   129→              error: coordination.reason,
   130→              executionTimeMs: Date.now() - startTime,
   131→            };
   132→          }
   133→
   134→          const lockAcquired = await this.tradeCoordination.acquireTradeLock(userId, signal.symbol, 'agent');
   135→          if (!lockAcquired) {
   136→            this.logger.warn(
   137→              `⚡ ORDER REJECTED: Trade lock not acquired for ${signal.symbol}`,
   138→            );
   139→            return {
   140→              success: false,
   141→              error: 'Trade lock not acquired — another execution in progress',
   142→              executionTimeMs: Date.now() - startTime,
   143→            };
   144→          }
   145→          coordinationLockAcquired = true;
   146→        } catch (coordErr: any) {
   147→          // FAIL-OPEN: Coordination check failed — log but proceed
   148→          this.logger.warn(`⚡ Trade coordination check failed: ${coordErr.message} — proceeding anyway`);
   149→        }
   150→      } else {
   151→        // Fallback: No TradeCoordinationService — use the old V146b check
   152→        const existingPosition = await this.prisma.position.findFirst({
   153→          where: { userId, symbol: signal.symbol, status: 'OPEN', source: 'agent' },
   154→        });
   155→
   156→        if (existingPosition) {
   157→          this.logger.warn(
   158→            `⚡ ORDER REJECTED: Agent already has position for ${signal.symbol} ` +
   159→            `(existing: ${existingPosition.side})`,
   160→          );
   161→          return {
   162→            success: false,
   163→            error: `يوجد مركز خاص بالوكيل لـ ${signal.symbol} (${existingPosition.side}) — لا يمكن فتح مركز آخر`,
   164→            executionTimeMs: Date.now() - startTime,
   165→          };
   166→        }
   167→      }
   168→
   169→      // Pre-flight: Check for duplicate order (in-memory 30s window)
   170→      if (this._isDuplicateOrder(userId, signal.symbol, signal.action)) {
   171→        return {
   172→          success: false,
   173→          error: 'أمر مكرر — تم تقديم أمر مشابه مؤخراً',
   174→          executionTimeMs: Date.now() - startTime,
   175→        };
   176→      }
   177→
   178→      // Pre-flight: Verify stop-loss is set (NON-NEGOTIABLE)
   179→      if (!signal.stopLoss || signal.stopLoss <= 0) {
   180→        this.logger.error('⚡ ORDER REJECTED: No stop-loss');
   181→        return {
   182→          success: false,
   183→          error: 'وقف الخسارة إجباري — لا يمكن تنفيذ أمر بدون وقف خسارة',
   184→          executionTimeMs: Date.now() - startTime,
   185→        };
   186→      }
   187→
   188→      // Pre-flight: Verify credential has trade permission
   189→      // DATA ISOLATION: Use findFirst with userId to prevent accessing other users' credentials
   190→      const credential = await this.prisma.exchangeCredential.findFirst({
   191→        where: { id: credentialId, userId },
   192→      });
   193→
   194→      // Paper trading mode — route through TradingService for proper record creation
   195→      // FIX: Previously, the Agent bypassed TradingService and created Position + AutonomousTrade
   196→      // directly. This caused THREE critical issues:
   197→      //   1. Position.create fails with P2002 unique constraint (userId+symbol+side+status)
   198→      //      when a previous position is stuck OPEN — every subsequent trade for that pair fails
   199→      //   2. Returns success:true even when Position creation fails — inconsistent state
   200→      //   3. No Order or Trade records created — breaks daily loss tracking and trade history
   201→      // Now: Route through TradingService.placeOrder() which handles paper-trading via
   202→      // _executePaperTrade(), creates all 4 records (Order+Position+Trade+Signal) in a
   203→      // single transaction, and handles P2002 with upsert logic.
   204→      if (credential && credential.exchange === 'paper-trading') {
   205→        this.logger.log(`⚡ Paper trading mode — routing through TradingService for ${signal.action} ${signal.symbol}`);
   206→
   207→        // Get live price for realistic execution
   208→        let executionPrice = signal.entryPrice;
   209→        try {
   210→          const liveQuote = await this.exchangeService.getQuote(signal.symbol);
   211→          if (liveQuote && liveQuote.price && liveQuote.price > 0) {
   212→            const slippagePercent = 0.01 + Math.random() * 0.04;
   213→            const slippageDirection = signal.action === 'BUY' ? 1 : -1;
   214→            executionPrice = liveQuote.price * (1 + slippageDirection * slippagePercent / 100);
   215→            this.logger.log(
   216→              `⚡ Paper trade using live price: ${liveQuote.price.toFixed(2)} → execution: ${executionPrice.toFixed(2)}`,
   217→            );
   218→          }
   219→        } catch (quoteErr: any) {
   220→          this.logger.warn(`⚡ Could not get live quote for ${signal.symbol}: ${quoteErr.message} — using signal price`);
   221→        }
   222→
   223→        // Validate execution price
   224→        if (!executionPrice || executionPrice <= 0) {
   225→          return {
   226→            success: false,
   227→            error: `سعر التنفيذ غير صالح (${executionPrice}) لـ ${signal.symbol} — تم إلغاء الأمر`,
   228→            executionTimeMs: Date.now() - startTime,
   229→          };
   230→        }
   231→
   232→        // Validate minimum trade value
   233→        const tradeValue = risk.positionSize * executionPrice;
   234→        if (tradeValue < 1) {
   235→          return {
   236→            success: false,
   237→            error: `قيمة الصفقة صغيرة جداً ($${tradeValue.toFixed(4)}) — تم الإلغاء`,
   238→            executionTimeMs: Date.now() - startTime,
   239→          };
   240→        }
   241→
   242→        try {
   243→          // Route through TradingService — creates Order + Position + Trade + Signal in one transaction
   244→          const orderRequest = {
   245→            credentialId,
   246→            symbol: signal.symbol,
   247→            side: signal.action as OrderSide,
   248→            type: signal.type as OrderType,
   249→            quantity: risk.positionSize,
   250→            price: executionPrice,
   251→            stopLoss: signal.stopLoss,
   252→            takeProfit: signal.takeProfit,
   253→            source: 'agent' as const,
   254→            idempotencyKey,
   255→          };
   256→
   257→          // ✅ FIX: Route through OrderDispatcher — prevents duplicate orders with SmartExecutor
   258→          const dispatchResult = await this.orderDispatcher.submitOrder({
   259→            source: 'agent' as const,
   260→            userId,
   261→            credentialId: orderRequest.credentialId,
   262→            symbol: orderRequest.symbol,
   263→            side: (orderRequest.side as string).toUpperCase() === 'BUY' ? 'BUY' : 'SELL',
   264→            quantity: orderRequest.quantity,
   265→            price: orderRequest.price || 0,
   266→            stopLoss: orderRequest.stopLoss,
   267→            takeProfit: orderRequest.takeProfit,
   268→            signalId:  signal?.id,
   269→            briefId:   (signal?.metadata as any)?.briefId ?? null, // V175: ربط الصفقة بالـ brief
   270→            isPaperTrading: true,
   271→            timeframe: signal?.timeframe, // V132: Pass timeframe for smart idempotency TTL
   272→          });
   273→          if (!dispatchResult.success) {
   274→            throw new Error(dispatchResult.error || dispatchResult.message || 'فشل الموزع');
   275→          }
   276→          const order = { id: dispatchResult.orderId || 'unknown' };
   277→          const executionTimeMs = Date.now() - startTime;
   278→          const calculatedSlippage = this._calculateSlippage(signal.entryPrice, executionPrice, signal.action);
   279→
   280→          // Track this order to prevent immediate duplicates
   281→          this.recentOrders.set(`${userId}:${signal.symbol}:${signal.action}`, new Date());
   282→
   283→          // Also record in AutonomousTrade table for agent-specific analytics
   284→          try {
   285→            await this.prisma.autonomousTrade.create({
   286→              data: {
   287→                userId,
   288→                agentRunId: `run-${userId}-${signal.strategy}`,
   289→                symbol: signal.symbol,
   290→                side: signal.action as any,
   291→                orderType: signal.type as any,
   292→                strategy: signal.strategy as any,
   293→                status: 'FILLED',
   294→                entryPrice: executionPrice,
   295→                stopLoss: signal.stopLoss,
   296→                takeProfit: signal.takeProfit,
   297→                quantity: risk.positionSize,
   298→                filledQuantity: risk.positionSize,
   299→                pnl: null,
   300→                fee: executionPrice * risk.positionSize * 0.001,
   301→                feeCurrency: 'USD',
   302→                riskScore: risk.riskScore,
   303→                confidence: signal.confidence,
   304→                riskRewardRatio: risk.riskRewardRatio,
   305→                reasoning: signal.reasoning,
   306→                signalData: JSON.stringify(signal.metadata || {}),
   307→                metadata: JSON.stringify({
   308→                  paperTrading: true,
   309→                  executionTimeMs,
   310→                  orderId: order.id,
   311→                }),
   312→                execution: JSON.stringify({
   313→                  success: true,
   314→                  paperTrading: true,
   315→                  orderId: order.id,
   316→                  filledQuantity: risk.positionSize,
   317→                  averagePrice: executionPrice,
   318→                  slippage: calculatedSlippage,
   319→                  executionTimeMs,
   320→                }),
   321→                credentialId,
   322→                exchangeOrderId: null,
   323→                openedAt: new Date(),
   324→              },
   325→            });
   326→          } catch (tradeErr: any) {
   327→            this.logger.error(`Failed to record AutonomousTrade: ${tradeErr.message}`);
   328→            // Non-fatal — the order was still executed via TradingService
   329→          }
   330→
   331→          this.recentOrders.set(`${userId}:${signal.symbol}:${signal.action}`, new Date());
   332→
   333→          // Audit log
   334→          await this.audit?.log({
   335→            userId,
   336→            action: 'AGENT_PAPER_TRADE_EXECUTED',
   337→            resource: 'autonomous-trader',
   338→            details: JSON.stringify({
   339→              orderId: order.id,
   340→              symbol: signal.symbol,
   341→              side: signal.action,
   342→              quantity: risk.positionSize,
   343→              executionPrice,
   344→              stopLoss: signal.stopLoss,
   345→              takeProfit: signal.takeProfit,
   346→              strategy: signal.strategy,
   347→              paperTrading: true,
   348→            }),
   349→          });
   350→
   351→          this.logger.log(
   352→            `✅ Paper order executed: ${signal.action} ${risk.positionSize} ${signal.symbol} ` +
   353→            `@ ${executionPrice.toFixed(2)} via TradingService (order: ${order.id})`,
   354→          );
   355→
   356→          return {
   357→            success: true,
   358→            orderId: order.id,
   359→            exchangeOrderId: undefined,
   360→            filledQuantity: risk.positionSize,
   361→            averagePrice: executionPrice,
   362→            fee: executionPrice * risk.positionSize * 0.001,
   363→            feeCurrency: 'USD',
   364→            slippage: calculatedSlippage,
   365→            executionTimeMs,
   366→          };
   367→        } catch (orderErr: any) {
   368→          const executionTimeMs = Date.now() - startTime;
   369→          this.logger.error(`⚡ TradingService.placeOrder failed for paper trade: ${orderErr.message}`);
   370→
   371→          // Check if it's a P2002 unique constraint (duplicate position)
   372→          const isDuplicate = orderErr.message?.includes('Unique constraint') ||
   373→            orderErr.message?.includes('P2002') ||
   374→            orderErr.message?.includes('already has an open position');
   375→
   376→          return {
   377→            success: false,
   378→            error: isDuplicate
   379→              ? `يوجد مركز مفتوح بالفعل لـ ${signal.symbol} — لا يمكن فتح مركز آخر`
   380→              : `فشل تنفيذ الصفقة الورقية: ${orderErr.message}`,
   381→            executionTimeMs,
   382→          };
   383→        }
   384→      }
   385→
   386→      // DATA ISOLATION: credential was already filtered by userId in findFirst above
   387→      if (!credential || !credential.isValid) {
   388→        return {
   389→          success: false,
   390→          error: 'بيانات الاعتماد غير صالحة',
   391→          executionTimeMs: Date.now() - startTime,
   392→        };
   393→      }
   394→
   395→      const permissions = JSON.parse(credential.permissions || '["read"]');
   396→      if (!permissions.includes('trade')) {
   397→        return {
   398→          success: false,
   399→          error: 'مفتاح API لا يملك صلاحية التداول — لا يمكن سحب الأموال',
   400→          executionTimeMs: Date.now() - startTime,
   401→        };
   402→      }
   403→
   404→      // V147 FIX: Block SELL on spot exchanges — you can't short-sell on spot.
   405→      // Spot exchanges only allow SELL of assets you already own.
   406→      // The Agent opens NEW positions, so SELL = "go short" which is impossible on spot.
   407→      // This requires a margin/futures account to execute.
   408→      const isSpotExchange = credential.exchange !== 'paper-trading' &&
   409→        !credential.testnet &&
   410→        credential.exchange !== 'alpaca'; // Alpaca supports short selling
   411→      if (isSpotExchange && signal.action === 'SELL') {
   412→        this.logger.warn(
   413→          `⚡ V147 ORDER REJECTED: SELL ${signal.symbol} on spot exchange ${credential.exchange} — short selling requires margin/futures`,
   414→        );
   415→        return {
   416→          success: false,
   417→          error: `بيع ${signal.symbol} غير ممكن على حساب سبوت (${credential.exchange}) — يحتاج حساب مارجن/فيوتشر للبيع على المكشوف`,
   418→          executionTimeMs: Date.now() - startTime,
   419→        };
   420→      }
   421→
   422→      // ═══════════════════════════════════════════════════════════════
   423→      // V150 FIX: Pre-trade balance check for REAL exchanges.
   424→      // Before submitting an order to Binance/KuCoin/etc., verify the
   425→      // account has sufficient balance. This prevents:
   426→      //   1. Wasting API calls on orders that will be rejected
   427→      //   2. "binance Account has insufficient balance" errors every minute
   428→      //   3. Unnecessary error logs and retry attempts
   429→      //
   430→      // FIX: Paper trading now checks AgentSettings.paperBalance — not unlimited.
   431→      // Real exchanges check live balance via CredentialsService.
   432→      // ═══════════════════════════════════════════════════════════════
   433→      try {
   434→        const balanceCheck = await this._checkSufficientBalance(
   435→          credential,
   436→          signal.symbol,
   437→          signal.action as OrderSide,
   438→          risk.positionSize,
   439→          signal.entryPrice,
   440→        );
   441→        if (!balanceCheck.sufficient) {
   442→          this.logger.warn(
   443→            `⚡ V150 ORDER REJECTED: Insufficient balance for ${signal.action} ${risk.positionSize} ${signal.symbol} ` +
   444→            `on ${credential.exchange} — need $${balanceCheck.required.toFixed(2)}, have $${balanceCheck.available.toFixed(2)}`
   445→          );
   446→          return {
   447→            success: false,
   448→            error: `رصيد غير كافي في ${credential.exchange} — يحتاج $${balanceCheck.required.toFixed(2)}، المتاح $${balanceCheck.available.toFixed(2)}`,
   449→            executionTimeMs: Date.now() - startTime,
   450→          };
   451→        }
   452→      } catch (balanceErr: any) {
   453→        this.logger.warn(
   454→          `⚡ V150: Pre-trade balance check failed for ${credential.exchange}: ${balanceErr.message} — proceeding with order submission`
   455→        );
   456→      }
   457→
   458→      // Construct the order request
   459→      const orderRequest: PlaceOrderRequest = {
   460→        credentialId,
   461→        symbol: signal.symbol,
   462→        side: signal.action as OrderSide,
   463→        type: signal.type as OrderType,
   464→        quantity: risk.positionSize,
   465→        price: signal.type === OrderType.LIMIT ? signal.entryPrice : undefined,
   466→        stopLoss: signal.stopLoss, // MANDATORY
   467→        takeProfit: signal.takeProfit,
   468→        idempotencyKey,
   469→      };
   470→
   471→      // FIX: Route ALL orders (real + paper) through OrderDispatcher → BullMQ pipeline
   472→      const dispatchResult = await this.orderDispatcher.submitOrder({
   473→        source: 'agent',
   474→        userId,
   475→        credentialId: orderRequest.credentialId,
   476→        symbol: orderRequest.symbol,
   477→        side: (orderRequest.side as string).toUpperCase() === 'BUY' ? 'BUY' : 'SELL',
   478→        quantity: orderRequest.quantity,
   479→        price: typeof orderRequest.price === 'number' ? orderRequest.price : undefined,
   480→        stopLoss: orderRequest.stopLoss,
   481→        takeProfit: orderRequest.takeProfit,
   482→        signalId: signal?.id,
   483→        isPaperTrading: false,
   484→        timeframe: signal?.timeframe, // V132: Pass timeframe for smart idempotency TTL
   485→      });
   486→      if (!dispatchResult.success) {
   487→        throw new Error(dispatchResult.error || dispatchResult.message || 'فشل الموزع');
   488→      }
   489→      const order = { id: dispatchResult.orderId || 'unknown', filledQuantity: orderRequest.quantity, averagePrice: orderRequest.price || 0, fee: 0, feeCurrency: 'USD', exchangeOrderId: null as any };
   490→
   491→      const executionTimeMs = Date.now() - startTime;
   492→
   493→      // Record successful execution
   494→      this.recentOrders.set(`${userId}:${signal.symbol}:${signal.action}`, new Date());
   495→
   496→      // Audit log — detailed record for compliance
   497→      await this.audit?.log({
   498→        userId,
   499→        action: 'AGENT_TRADE_EXECUTED',
   500→        resource: 'autonomous-trader',
   501→        details: JSON.stringify({
   502→          orderId: order.id,
   503→          symbol: signal.symbol,
   504→          side: signal.action,
   505→          type: signal.type,
   506→          quantity: risk.positionSize,
   507→          entryPrice: signal.entryPrice,
   508→          stopLoss: signal.stopLoss,
   509→          takeProfit: signal.takeProfit,
   510→          confidence: signal.confidence,
   511→          strategy: signal.strategy,
   512→          riskScore: risk.riskScore,
   513→          riskRewardRatio: risk.riskRewardRatio,
   514→          executionTimeMs,
   515→          reasoning: signal.reasoning,
   516→        }),
   517→      });
   518→
   519→      // Record in AutonomousTrade table for agent-specific analytics
   520→      try {
   521→        await this.prisma.autonomousTrade.create({
   522→          data: {
   523→            userId,
   524→            agentRunId: `run-${userId}-${signal.strategy}`, // matches agent session
   525→            symbol: signal.symbol,
   526→            side: signal.action as any,
   527→            orderType: signal.type as any,
   528→            strategy: signal.strategy as any,
   529→            status: 'FILLED',
   530→            entryPrice: signal.entryPrice,
   531→            stopLoss: signal.stopLoss,
   532→            takeProfit: signal.takeProfit,
   533→            quantity: risk.positionSize,
   534→            filledQuantity: Number(order.filledQuantity) || risk.positionSize,
   535→            pnl: null, // Will be updated when position closes
   536→            fee: Number(order.fee) || 0,
   537→            feeCurrency: order.feeCurrency || 'USD',
   538→            riskScore: risk.riskScore,
   539→            confidence: signal.confidence,
   540→            riskRewardRatio: risk.riskRewardRatio,
   541→            reasoning: signal.reasoning,
   542→            signalData: JSON.stringify(signal.metadata || {}),
   543→            metadata: JSON.stringify({ orderId: order.id, executionTimeMs }),
   544→            execution: JSON.stringify({
   545→              success: true,
   546→              orderId: order.id,
   547→              exchangeOrderId: order.exchangeOrderId,
   548→              filledQuantity: Number(order.filledQuantity) || risk.positionSize,
   549→              averagePrice: Number(order.averagePrice) || signal.entryPrice,
   550→              fee: Number(order.fee) || 0,
   551→              slippage: this._calculateSlippage(signal.entryPrice, Number(order.averagePrice) || signal.entryPrice, signal.action),
   552→              executionTimeMs,
   553→            }),
   554→            credentialId,
   555→            exchangeOrderId: null,
   556→            openedAt: new Date(),
   557→          },
   558→        });
   559→      } catch (tradeErr: any) {
   560→        this.logger.error(`Failed to record AutonomousTrade: ${tradeErr.message}`);
   561→        // Non-fatal — the order was still executed successfully
   562→      }
   563→
   564→      this.logger.log(
   565→        `✅ Order executed: ${order.id} — ${signal.action} ${risk.positionSize} ${signal.symbol} ` +
   566→        `(${executionTimeMs}ms)`,
   567→      );
   568→
   569→      return {
   570→        success: true,
   571→        orderId: order.id,
   572→        exchangeOrderId: undefined,
   573→        filledQuantity: Number(order.filledQuantity) || risk.positionSize,
   574→        averagePrice: Number(order.averagePrice) || signal.entryPrice,
   575→        fee: Number(order.fee) || 0,
   576→        feeCurrency: order.feeCurrency || undefined,
   577→        slippage: this._calculateSlippage(signal.entryPrice, Number(order.averagePrice) || signal.entryPrice, signal.action),
   578→        executionTimeMs,
   579→      };
   580→    } catch (error: any) {
   581→      const executionTimeMs = Date.now() - startTime;
   582→
   583→      this.logger.error(
   584→        `❌ Order execution failed for ${signal.symbol}: ${error.message}`,
   585→      );
   586→
   587→      // Audit the failure
   588→      await this.audit?.log({
   589→        userId,
   590→        action: 'AGENT_TRADE_FAILED',
   591→        resource: 'autonomous-trader',
   592→        details: JSON.stringify({
   593→          symbol: signal.symbol,
   594→          side: signal.action,
   595→          error: error.message,
   596→          executionTimeMs,
   597→        }),
   598→      });
   599→
   600→      return {
   601→        success: false,
   602→        error: `فشل في التنفيذ: ${error.message}`,
   603→        executionTimeMs,
   604→      };
   605→    } finally {
   606→      // #18: Always release the coordination lock if it was acquired
   607→      if (coordinationLockAcquired && this.tradeCoordination) {
   608→        await this.tradeCoordination.releaseTradeLock(userId, signal.symbol);
   609→      }
   610→    }
   611→  }

   613→  /**
   614→   * Emergency close all open positions for a user
   615→   * Used when daily loss limit is reached or agent is emergency-stopped
   616→   */
   617→  async emergencyCloseAll(userId: string): Promise<{
   618→    closedCount: number;
   619→    errors: number;
   620→    totalPnL: number;
   621→  }> {
   622→    this.logger.warn(`🚨 Emergency close all positions for user ${userId}`);
   623→
   624→    let closedCount = 0;
   625→    let errors = 0;
   626→    let totalPnL = 0;
   627→
   628→    // FIX: Guard against missing forwardRef dependencies
   629→    if (!this.tradingService) {
   630→      this.logger.error('🚨 TradingService not available — cannot close positions');
   631→      return { closedCount: 0, errors: 0, totalPnL: 0 };
   632→    }
   633→
   634→    try {
   635→      const positions = await this.prisma.position.findMany({
   636→        where: { userId, status: 'OPEN' },
   637→      });
   638→
   639→      for (const position of positions) {
   640→        try {
   641→          const result = await this.tradingService.closePosition(userId, {
   642→            positionId: position.id,
   643→          });
   644→
   645→          if (result.pnl) {
   646→            totalPnL += result.pnl;
   647→          }
   648→
   649→          closedCount++;
   650→
   651→          await this.audit?.log({
   652→            userId,
   653→            action: 'AGENT_EMERGENCY_CLOSE',
   654→            resource: 'autonomous-trader',
   655→            details: JSON.stringify({
   656→              positionId: position.id,
   657→              symbol: position.symbol,
   658→              pnl: result.pnl,
   659→            }),
   660→          });
   661→        } catch (error: any) {
   662→          this.logger.error(
   663→            `Failed to close position ${position.id}: ${error.message}`,
   664→          );
   665→          errors++;
   666→        }
   667→      }
   668→    } catch (error: any) {
   669→      this.logger.error(`Emergency close failed: ${error.message}`);
   670→    }
   671→
   672→    this.logger.log(
   673→      `🚨 Emergency close complete: ${closedCount} closed, ${errors} errors, PnL: ${totalPnL.toFixed(2)}`,
   674→    );
   675→
   676→    return { closedCount, errors, totalPnL };
   677→  }

   681→  private _isDuplicateOrder(userId: string, symbol: string, side: string): boolean {
   682→    const key = `${userId}:${symbol}:${side}`;
   683→    const lastOrder = this.recentOrders.get(key);
   684→
   685→    if (!lastOrder) return false;
   686→
   687→    // Consider duplicate if within last 30 seconds
   688→    const timeSinceLastOrder = Date.now() - lastOrder.getTime();
   689→    return timeSinceLastOrder < 30000;
   690→  }
   691→
   692→  private _calculateSlippage(expectedPrice: number, actualPrice: number, side: string): number {
   693→    if (!expectedPrice || !actualPrice) return 0;
   694→    return Math.abs((actualPrice - expectedPrice) / expectedPrice) * 100;
   695→  }

   697→  private _cleanupOldOrders(): void {
   698→    const cutoff = Date.now() - 5 * 60 * 1000; // 5 minutes
   699→    for (const [key, date] of this.recentOrders.entries()) {
   700→      if (date.getTime() < cutoff) {
   701→        this.recentOrders.delete(key);
   702→      }
   703→    }
   704→  }

   706→  /**
   707→   * V150: Check if the exchange account has sufficient balance for a trade.
   708→   * Uses CredentialsService.fetchAllExchangeBalances() which caches results
   709→   * for 5 seconds, so this is efficient and won't hammer the exchange API.
   710→   *
   711→   * Prevents "insufficient balance" errors from Binance that would occur
   712→   * anyway, saving API calls and reducing log noise.
   713→   *
   714→   * @returns { sufficient: boolean, required: number, available: number }
   715→   */
   716→  private async _checkSufficientBalance(
   717→    credential: any,
   718→    symbol: string,
   719→    side: string,
   720→    quantity: number,
   721→    price: number,
   722→  ): Promise<{ sufficient: boolean; required: number; available: number }> {
   723→    // Calculate order value (notional)
   724→    const orderValue = Math.abs(quantity * price);
   725→    const required = orderValue * 1.005; // Add 0.5% buffer for fees/slippage
   726→
   727→    // Use CredentialsService to fetch balance (with 5-second cache)
   728→    if (!this.credentialsService) {
   729→      return { sufficient: true, required, available: Infinity };
   730→    }
   731→
   732→    try {
   733→      const balances = await this.credentialsService.fetchAllExchangeBalances(credential.userId);
   734→
   735→      // Paper trading: use paper balance from AgentSettings
   736→      if (credential.exchange === 'paper-trading') {
   737→        const paperExchange = balances.exchanges.find((e: any) => e.exchange === 'paper-trading');
   738→        const available = paperExchange?.available ?? 0;
   739→        return { sufficient: available >= required, required, available };
   740→      }
   741→      // Find the specific exchange's available balance
   742→      const exchangeBalance = balances.exchanges.find(
   743→        (e: any) => e.credentialId === credential.id ||
   744→          e.exchange === credential.exchange ||
   745→          (credential.exchange.includes('test') && e.isTestnet && e.exchange.includes(credential.exchange.replace('_test', '').replace('_future_test', '')))
   746→      );
   747→
   748→      if (!exchangeBalance) {
   749→        // Exchange not found in balance response — skip check
   750→        return { sufficient: true, required, available: Infinity };
   751→      }
   752→
   753→      const available = exchangeBalance.available || 0;
   754→      const sufficient = available >= required;
   755→
   756→      if (!sufficient) {
   757→        this.logger.warn(
   758→          `⚡ V150: Balance check FAILED for ${credential.exchange}: ` +
   759→          `need $${required.toFixed(2)}, have $${available.toFixed(2)} available`
   760→        );
   761→      }
   762→
   763→      return { sufficient, required, available };
   764→    } catch (err: any) {
   765→      // Balance check failed — log warning but don't block the trade
   766→      // (the exchange itself will reject if balance is insufficient)
   767→      this.logger.warn(`⚡ V150: Balance check error for ${credential.exchange}: ${err.message}`);
   768→      return { sufficient: true, required, available: Infinity };
   769→    }
   770→  }
