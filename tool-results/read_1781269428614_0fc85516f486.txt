   261→    app.getHttpAdapter().getInstance().get('/api/health', async (req: any, res: any) => {
   262→      const start = Date.now();
   263→      const checks: Record<string, { status: string; latencyMs?: number }> = {};
   264→
   265→      // FIX v6: Only use prisma.isAvailable() for health check — do NOT run $queryRaw.
   266→      // Running `SELECT 1` on every health check (every 30s from Railway) creates
   267→      // unnecessary DB queries. With connection_limit=1, each query occupies the
   268→      // single connection slot, potentially blocking other queries.
   269→      // isAvailable() checks the internal connection flag — zero DB overhead.
   270→      try {
   271→        if (prisma.isAvailable()) {
   272→          checks.database = { status: 'ok' };
   273→        } else {
   274→          // DIAGNOSTIC: Include last error and URL prefix when DB is unavailable
   275→          const diag = prisma.getDiagnosticInfo();
   276→          checks.database = {
   277→            status: 'error',
   278→            lastError: diag.lastError || 'No error recorded',
   279→            urlPrefix: diag.urlPrefix || 'Not set',
   280→            failures: diag.failures,
   281→          } as any;
   282→        }
   283→      } catch (err: any) {
   284→        checks.database = { status: 'error', lastError: err?.message } as any;
   285→      }
   286→
   287→      try {
   288→        const redisStart = Date.now();
   289→        if (redisService && typeof redisService.ping === 'function') {
   290→          await redisService.ping();
   291→          checks.redis = { status: 'ok', latencyMs: Date.now() - redisStart };
   292→        } else {
   293→          checks.redis = { status: 'degraded' };
   294→        }
   295→      } catch {
   296→        checks.redis = { status: 'degraded' };
   297→      }
   298→
   299→      const mem = process.memoryUsage();
   300→      const memMB = Math.round(mem.heapUsed / 1024 / 1024);
   301→      checks.memory = {
   302→        status: memMB > 512 ? 'warning' : 'ok',
   303→      };
   304→
   305→      const hasError = Object.values(checks).some(c => c.status === 'error');
   306→      const allOk = Object.values(checks).every(c => c.status === 'ok');
   307→      // FIX: Always return 200 — even when DB/Redis are degraded.
   308→      // Railway healthcheck requires 200 to mark the replica as healthy.
   309→      // Returning 503 on DB/Redis failure causes "1/1 replicas never became healthy"
   310→      // and prevents deployment. The 'status' field in the response body
   311→      // still reflects the real health state for monitoring dashboards.
   312→      const statusCode = 200;
   313→
   314→      res.status(statusCode).json({
   315→        status: hasError ? 'degraded' : (allOk ? 'ok' : 'degraded'),
   316→        uptime: Math.round(process.uptime()),
   317→        checks,
   318→        responseTimeMs: Date.now() - start,
   319→      });
   320→    });
