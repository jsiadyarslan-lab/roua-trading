   174→  /**
   175→   * FIX: Trigger an initial council session after startup — but ONLY after
   176→   * confirming that at least 2 AI models are available.
   177→   * Previously, the session fired at 30 seconds regardless of AI readiness,
   178→   * producing HOLD results that got cached and blocked all subsequent sessions.
   179→   * Now we poll AI health every 10 seconds and only start the session once
   180→   * ≥2 models are confirmed working, with a max wait of 3 minutes.
   181→   */
   182→  private _triggerStartupSession(): void {
   183→    const MAX_WAIT_MS = 3 * 60 * 1000; // 3 minutes max wait
   184→    const POLL_INTERVAL_MS = 10 * 1000; // Check every 10 seconds
   185→    const startTime = Date.now();
   186→
   187→    const checkAndTrigger = async (): Promise<void> => {
   188→      const elapsed = Date.now() - startTime;
   189→      if (elapsed > MAX_WAIT_MS) {
   190→        this.logger.warn('🏛️ Startup session: max wait reached (3 min) — triggering session even with limited AI models');
   191→        try {
   192→          const result = await this.runHourlySession();
   193→          this.logger.log(
   194→            `🏛️ Startup session (forced) complete: ${result.pairsAnalyzed} pairs, ` +
   195→            `${result.briefsIssued} new briefs, ${result.briefsModified} modified`,
   196→          );
   197→        } catch (error: any) {
   198→          this.logger.error(`🏛️ Startup session (forced) failed: ${error.message}`);
   199→        }
   200→        return;
   201→      }
   202→
   203→      try {
   204→        const models = await this.orchestrator.getModelsStatus();
   205→        const working = models.filter((m: any) => m.available || m.keyAvailable).length;
   206→
   207→        if (working >= 2) {
   208→          this.logger.log(`🏛️ ${working} AI models ready — triggering startup council session`);
   209→          const result = await this.runHourlySession();
   210→          this.logger.log(
   211→            `🏛️ Startup session complete: ${result.pairsAnalyzed} pairs, ` +
   212→            `${result.briefsIssued} new briefs, ${result.briefsModified} modified`,
   213→          );
   214→        } else {
   215→          this.logger.log(`🏛️ Only ${working}/2 AI models ready — waiting ${POLL_INTERVAL_MS / 1000}s before retry...`);
   216→          setTimeout(checkAndTrigger, POLL_INTERVAL_MS);
   217→        }
   218→      } catch (error: any) {
   219→        this.logger.warn(`🏛️ AI health check failed: ${error.message} — retrying in ${POLL_INTERVAL_MS / 1000}s`);
   220→        setTimeout(checkAndTrigger, POLL_INTERVAL_MS);
   221→      }
   222→    };
   223→
   224→    // Start checking after 30 seconds (give NestJS time to initialize)
   225→    setTimeout(checkAndTrigger, 30000);
   226→  }

   268→  @Cron('7,37 * * * *')
   269→  async runAgentSession(): Promise<void> {
   270→    // FIX: Skip cycle when DB is unavailable to prevent connection pool exhaustion
   271→    if (!this.prisma.isAvailable?.()) {
   272→      return;
   273→    }
   274→
   275→    // Check AUTO_TRADING_ENABLED
   276→    try {
   277→      const s = await this.prisma.$queryRaw<any[]>`SELECT value FROM "Setting" WHERE key = 'AUTO_TRADING_ENABLED' LIMIT 1`.catch(() => []);
   278→      if (s?.[0] && String(s[0].value) !== 'true') return;
   279→    } catch {}
   280→
   281→    if (this.isAgentInSession) return; // V130: Use own guard
   282→
   283→    this.isAgentInSession = true;
   284→    try {
   285→      this.logger.log('🏛️ Agent Council: generating M30/H1/H4/D1/W1 briefs...');
   286→
   287→      // ═══════════════════════════════════════════════════════════════════
   288→      // V132 FIX: Only generate briefs for BINANCE-supported pairs.
   289→      //
   290→      // PROBLEM: ALL_COUNCIL_PAIRS includes forex (EUR/USD, GBP/USD),
   291→      // stocks (AAPL, MSFT), and commodities (XAU/USD). These pairs CANNOT
   292→      // be executed on Binance (crypto-only exchange). Briefs for non-crypto
   293→      // pairs were being generated, wasting AI API calls, and then failing
   294→      // at execution with "binance does not have market symbol AAPL".
   295→      //
   296→      // FIX: Only use BINANCE_SUPPORTED_PAIRS for brief generation.
   297→      // All current users trade on Binance (binance_test). Future multi-
   298→      // exchange support can filter per-user based on their exchange.
   299→      // ═══════════════════════════════════════════════════════════════════
   300→      // V190: Read maxPairsPerSession from DB admin settings
   301→      const councilCfg = await this._getCouncilConfig();
   302→      const agentPairs = BINANCE_SUPPORTED_PAIRS.slice(0, councilCfg.maxPairsPerSession);
   303→      this.logger.log(`🏛️ Agent Council: analyzing ${agentPairs.length} crypto pairs (maxPairs=${councilCfg.maxPairsPerSession}): ${agentPairs.join(', ')}`);
   304→
   305→      // V132: Parallel processing — process all pairs concurrently instead of sequentially.
   306→      // Previously, pairs were processed one-by-one, taking 20-30 minutes for 15 pairs.
   307→      // Now: All pairs process in parallel, limited to 3 concurrent to respect AI rate limits.
   308→      const agentResults = await this._parallelProcess(
   309→        agentPairs,
   310→        async (pair) => {
   311→          const pairResult = { pairs: 0, briefs: 0, errors: 0 };
   312→          try {
   313→            const marketData = await this.orchestrator.fetchQuickMarketData(pair);
   314→            if (!marketData?.price) return pairResult;
   315→
   316→            for (const tf of AGENT_TIMEFRAMES as any[]) {
   317→              // Slow timeframes (H4, D1, W1): only top 3 pairs to reduce AI costs
   318→              if (AGENT_SLOW_TIMEFRAMES.includes(tf as any) && agentPairs.indexOf(pair) >= 3) continue;
   319→              await this._analyzePairTimeframe(pair, tf, marketData.price, { pairs: 0, briefs: 0, errors: 0, sessionId: 'agent-session', durationMs: 0 } as any);
   320→              pairResult.briefs++;
   321→            }
   322→            pairResult.pairs = 1;
   323→          } catch (e: any) {
   324→            this.logger.warn(`Agent session: ${pair} failed — ${e.message}`);
   325→            pairResult.errors++;
   326→          }
   327→          return pairResult;
   328→        },
   329→        3, // max 3 concurrent AI calls to respect rate limits
   330→      );
   331→
   332→      const agentTotal = agentResults.reduce((acc, r) => ({
   333→        pairs: acc.pairs + r.pairs,
   334→        briefs: acc.briefs + r.briefs,
   335→        errors: acc.errors + r.errors,
   336→      }), { pairs: 0, briefs: 0, errors: 0 });
   337→      this.logger.log(`🏛️ Agent Council complete: ${agentTotal.pairs} pairs, ${agentTotal.briefs} briefs, ${agentTotal.errors} errors`);
   338→    } catch (e: any) {
   339→      this.logger.warn(`Agent session error: ${e.message}`);
   340→    } finally {
   341→      this.isAgentInSession = false;
   342→    }
   343→  }

   345→  @Cron('*/15 * * * *')
   346→  async runHourlySession(): Promise<CouncilSessionResult> {
   347→    // FIX: Skip cycle when DB is unavailable to prevent connection pool exhaustion
   348→    if (!this.prisma.isAvailable?.()) {
   349→      return {
   350→        timestamp: new Date().toISOString(),
   351→        pairsAnalyzed: 0,
   352→        briefsIssued: 0,
   353→        briefsModified: 0,
   354→        briefsCancelled: 0,
   355→        briefsExecuted: 0,
   356→        durationMs: 0,
   357→      };
   358→    }
   359→
   360→    // V175: Sanctuary halt check — يمنع الجلسات عند الخسارة المتتالية
   361→    try {
   362→      const haltUntil = await this.redis?.get('council:sanctuary:halt');
   363→      if (haltUntil && new Date(haltUntil) > new Date()) {
   364→        this.logger.warn(`🏛️ Council HALTED by Sanctuary until ${haltUntil}`);
   365→        return { timestamp: new Date().toISOString(), pairsAnalyzed: 0, briefsIssued: 0, briefsModified: 0, briefsCancelled: 0, briefsExecuted: 0, durationMs: 0 };
   366→      }
   367→    } catch { /* non-critical — don't block trading */ }
   368→
   369→    // ═══════════════════════════════════════════════════════════════════
   370→    // FIX: Single AUTO_TRADING_ENABLED check (removed double-guard).
   371→    // Previously there were TWO separate checks that both defaulted to
   372→    // "skip if false" — this meant that if AUTO_TRADING_ENABLED was
   373→    // false in DB (the common default), BOTH checks would skip the
   374→    // session, generating ZERO briefs. The Smart Executor then has
   375→    // nothing to execute → no trades ever.
   376→    //
   377→    // Now: Single check that defaults to TRUE (generate briefs) when
   378→    // the DB setting doesn't exist. The Smart Executor already has
   379→    // its own user-level enable check, so we don't need to block
   380→    // brief generation here. Briefs without enabled users are harmless.
   381→    // ═══════════════════════════════════════════════════════════════════
   382→    try {
   383→      // SAFETY: Default to FALSE — auto trading must be explicitly enabled by admin.
   384→      // A new deployment should NOT start trading until the operator confirms readiness.
   385→      let autoTradingEnabled = false;
   386→      try {
   387→        const dbSetting = await this.prisma.setting.findUnique({
   388→          where: { key: 'AUTO_TRADING_ENABLED' },
   389→        });
   390→        if (dbSetting) {
   391→          autoTradingEnabled = JSON.parse(dbSetting.value);
   392→        }
   393→        // If no DB setting exists, keep default FALSE (safe — operator must enable)
   394→      } catch {
   395→        // DB lookup failed — keep default FALSE (fail safe)
   396→      }
   397→
   398→      if (!autoTradingEnabled) {
   399→        this.logger.debug('🏛️ AUTO_TRADING_ENABLED=false in DB — skipping council session');
   400→        return {
   401→          timestamp: new Date().toISOString(),
   402→          pairsAnalyzed: 0,
   403→          briefsIssued: 0,
   404→          briefsModified: 0,
   405→          briefsCancelled: 0,
   406→          briefsExecuted: 0,
   407→          durationMs: 0,
   408→        };
   409→      }
   410→    } catch {
   411→      // If we can't check the setting, PROCEED (generate briefs)
   412→      // This is the opposite of the old behavior which would skip on error
   413→    }
   414→
   415→    if (this.isExecutorInSession) {
   416→      this.logger.warn('🏛️ Previous executor council session still running — skipping');
   417→      return {
   418→        timestamp: new Date().toISOString(),
   419→        pairsAnalyzed: 0,
   420→        briefsIssued: 0,
   421→        briefsModified: 0,
   422→        briefsCancelled: 0,
   423→        briefsExecuted: 0,
   424→        durationMs: 0,
   425→      };
   426→    }
   427→
   428→    this.isExecutorInSession = true;
   429→    const startTime = Date.now();
   430→
   431→    const result: CouncilSessionResult = {
   432→      timestamp: new Date().toISOString(),
   433→      pairsAnalyzed: 0,
   434→      briefsIssued: 0,
   435→      briefsModified: 0,
   436→      briefsCancelled: 0,
   437→      briefsExecuted: 0,
   438→      durationMs: 0,
   439→      diagnostics: [],
   440→    };
   441→
   442→    try {
   443→      this.logger.log('🏛️ Strategic Council convening hourly session...');
   444→
   445→      // Check daily cost cap
   446→      // V190: Read dailyCostCapUsd from DB admin settings
   447→      const councilCfg = await this._getCouncilConfig();
   448→      const dailyCostCap = councilCfg.dailyCostCapUsd;
   449→      const todayCost = await this._getTodayCost();
   450→      if (todayCost >= dailyCostCap) {
   451→        this.logger.warn(`💰 Daily cost cap reached ($${todayCost.toFixed(2)}/$${dailyCostCap}) — skipping session`);
   452→        return result;
   453→      }
   454→
   455→      // ═══════════════════════════════════════════════════════════════════
   456→      // V132 FIX: Only generate briefs for BINANCE-supported pairs.
   457→      //
   458→      // PROBLEM: ALL_COUNCIL_PAIRS includes 15 pairs (7 crypto + 3 forex +
   459→      // 4 stocks + 1 commodity). Non-crypto briefs waste AI API calls and
   460→      // always fail at execution on Binance. This was the #1 cause of
   461→      // "binance does not have market symbol AAPL/GBP/USD" errors.
   462→      //
   463→      // FIX: Only use BINANCE_SUPPORTED_PAIRS (7 crypto pairs) for
   464→      // executor brief generation. Non-crypto pairs can be analyzed for
   465→      // market context in the future when multi-exchange support is added.
   466→      // ═══════════════════════════════════════════════════════════════════
   467→      // V190: Read maxPairsPerSession from DB admin settings
   468→      const executorPairs = BINANCE_SUPPORTED_PAIRS.slice(0, councilCfg.maxPairsPerSession);
   469→      this.logger.log(`🏛️ Executor Council: analyzing ${executorPairs.length} crypto pairs (maxPairs=${councilCfg.maxPairsPerSession}): ${executorPairs.join(', ')}`);
   470→
   471→      // V132: Parallel processing — process all pairs concurrently instead of sequentially.
   472→      // Previously: 15 pairs × 3 timeframes × 5-10s AI call = 225-450s (4-8 minutes)
   473→      // Now: 7 pairs in parallel with concurrency=3 = ~2-3 minutes total
   474→      const pairResults = await this._parallelProcess(
   475→        executorPairs,
   476→        async (pair) => {
   477→          // Check cost before each pair
   478→          const cost = await this._getTodayCost();
   479→          if (cost >= dailyCostCap) {
   480→            this.logger.warn('💰 Daily cost cap reached — stopping session early');
   481→            return { analyzed: false, error: 'cost_cap' };
   482→          }
   483→
   484→          try {
   485→            await this._analyzePair(pair, result);
   486→            return { analyzed: true };
   487→          } catch (error: any) {
   488→            if (error.message?.includes('Too many database connections') || error.message?.includes('connection pool')) {
   489→              this.logger.error(`🏛️ DB connection exhaustion detected during ${pair} analysis — breaking`);
   490→              return { analyzed: false, error: 'db_exhaustion' };
   491→            }
   492→            this.logger.error(`🏛️ Council failed for ${pair}: ${error.message}`);
   493→            return { analyzed: false, error: error.message };
   494→          }
   495→        },
   496→        3, // max 3 concurrent AI calls to respect rate limits
   497→      );
   498→
   499→      // Count successfully analyzed pairs
   500→      result.pairsAnalyzed = pairResults.filter(r => r.analyzed).length;
   501→
   502→      // Check if DB exhaustion was detected — break if so
   503→      if (pairResults.some(r => r.error === 'db_exhaustion')) {
   504→        this.logger.error('🏛️ Stopping session early due to DB connection exhaustion');
   505→      }
   506→
   507→      // Expire outdated briefs
   508→      await this._expireOutdatedBriefs();
   509→
   510→      // Mark executed briefs (those that SmartExecutor has already executed)
   511→      await this._markExecutedBriefs();
   512→
   513→      result.durationMs = Date.now() - startTime;
   514→
   515→      this.logger.log(
   516→        `🏛️ Strategic Council session complete: ${result.pairsAnalyzed} pairs, ` +
   517→        `${result.briefsIssued} new briefs, ${result.briefsModified} modified, ` +
   518→        `${result.briefsCancelled} cancelled, ${result.briefsExecuted} executed (${result.durationMs}ms)`,
   519→      );
   520→
   521→      // Store session result in Redis
   522→      await this.redis.set(
   523→        'strategic-council:last_session',
   524→        JSON.stringify(result),
   525→        3600000, // 1 hour TTL
   526→      );
   527→
   528→      // Publish council completion event for Smart Executor
   529→      try {
   530→        await this.redis.publish(
   531→          'council:session_complete',
   532→          JSON.stringify({
   533→            timestamp: result.timestamp,
   534→            briefsIssued: result.briefsIssued,
   535→            briefsModified: result.briefsModified,
   536→            activeBriefs: await this.getActiveBriefsCount(),
   537→          }),
   538→        );
   539→      } catch (pubError: any) {
   540→        this.logger.debug(`Failed to publish council event: ${pubError.message}`);
   541→      }
   542→
   543→      await this.audit.log({
   544→        userId: 'system',
   545→        action: 'STRATEGIC_COUNCIL_SESSION',
   546→        resource: 'strategic-council',
   547→        details: JSON.stringify(result),
   548→      });
   549→    } catch (error: any) {
   550→      this.logger.error(`🏛️ Strategic Council session failed: ${error.message}`);
   551→    } finally {
   552→      this.isExecutorInSession = false;
   553→    }
   554→
   555→    return result;
   556→  }

   618→
   619→      this.logger.log(
   620→        `🏛️ Manual session [${sessionId}] complete: ${result.pairsAnalyzed} pairs, ` +
   621→        `${result.briefsIssued} new briefs, ${result.briefsModified} modified (${result.durationMs}ms)`,
   622→      );

   624→      // Store session result in Redis so the frontend can poll /session/last

   626→        'strategic-council:last_session',
   627→        JSON.stringify(result),
   628→        3600000, // 1 hour TTL
   629→      );
   630→
   631→      // Publish council completion event for Smart Executor
   632→      try {

   648→        userId,

   696→  /**
   697→   * V134: Get CONSOLIDATED active briefs — ONE direction per pair.
   698→   *
   699→   * ROOT CAUSE FIX for "opened and closed after 1 second":
   700→   *   The Council generates briefs per pair+timeframe. Different timeframes
   701→   *   can have OPPOSITE directions (e.g., BTC/USDT M1=BUY, M5=SELL).
   702→   *   When the SmartExecutor processes all briefs sequentially, it:
   703→   *     1. Opens BUY from M1 brief
   704→   *     2. Sees SELL from M5 brief → closes BUY → opens SELL
   705→   *     3. Next tick: M1 BUY again → closes SELL → opens BUY
   706→   *     → INFINITE LOOP of open→close every tick
   707→   *
   708→   * FIX: Consolidate all briefs for the same pair into ONE direction.
   709→   * The direction with the HIGHEST confidence wins. If confidence is tied,
   710→   * the shorter timeframe wins (more recent data).
   711→   *
   712→   * This ensures the executor sees AT MOST ONE brief per pair,
   713→   * eliminating the open→close→open loop entirely.
   714→   */
   715→  async getConsolidatedBriefs(userId?: string): Promise<TradingBriefDTO[]> {
   716→    const allBriefs = await this.getActiveBriefs(userId);
   717→
   718→    if (allBriefs.length === 0) return [];
   719→
   720→    // ═══════════════════════════════════════════════════════════════════
   721→    // V143 FIX: Separate consolidation for EXECUTOR and AGENT timeframes.
   722→    //
   723→    // ROOT CAUSE: Previously, getConsolidatedBriefs() grouped ALL briefs
   724→    // by pair regardless of timeframe type. This meant executor briefs
   725→    // (M1/M5/M15) could CANCEL agent briefs (M30/H1/H4/D1/W1) and
   726→    // vice versa. Since M1 briefs get 2x weight in the consolidation
   727→    // vote, they almost always won, causing agent-timeframe briefs to
   728→    // be cancelled (isActive=false). The Agent then couldn't see any
   729→    // M30+ briefs → ZERO agent trades.
   730→    //
   731→    // Example: BTC/USDT has M1=BUY (conf 55) and M30=SELL (conf 60).
   732→    // Old: M1 BUY score = 55 * 2.0 = 110, M30 SELL score = 60 * 1.0 = 60
   733→    // → BUY wins → M30 SELL cancelled → Agent has no briefs
   734→    //
   735→    // FIX: Consolidate WITHIN each timeframe group separately.
   736→    //   - Executor briefs only conflict with other executor briefs
   737→    //   - Agent briefs only conflict with other agent briefs
   738→    // Each system gets its own independent signal per pair.
   739→    // ═══════════════════════════════════════════════════════════════════
   740→    const executorBriefs = allBriefs.filter(b => EXECUTOR_TIMEFRAMES.includes(b.timeframe));
   741→    const agentBriefs = allBriefs.filter(b => AGENT_TIMEFRAMES.includes(b.timeframe));
   742→
   743→    const consolidated: TradingBriefDTO[] = [];
   744→
   745→    // Consolidate executor briefs independently
   746→    const executorConsolidated = await this._consolidateBriefsByPair(executorBriefs, 'EXECUTOR');
   747→    consolidated.push(...executorConsolidated);
   748→
   749→    // Consolidate agent briefs independently
   750→    const agentConsolidated = await this._consolidateBriefsByPair(agentBriefs, 'AGENT');
   751→    consolidated.push(...agentConsolidated);
   752→
   753→    this.logger.log(
   754→      `🏛️ V143 Consolidation: ${allBriefs.length} raw briefs → ${consolidated.length} consolidated ` +
   755→      `(executor: ${executorConsolidated.length}, agent: ${agentConsolidated.length}) ` +
   756→      `(pairs: ${consolidated.map(b => b.pair + ':' + b.direction + ':' + b.timeframe).join(', ')})`
   757→    );
   758→
   759→    return consolidated;
   760→  }
   761→
   762→  /**
   763→   * V143: Consolidate briefs by pair within a specific timeframe group.
   764→   * Each group (EXECUTOR or AGENT) resolves conflicts independently.
   765→   * Losing briefs are only cancelled WITHIN the same group — executor
   766→   * briefs never cancel agent briefs and vice versa.
   767→   */
   768→  private async _consolidateBriefsByPair(
   769→    briefs: TradingBriefDTO[],
   770→    group: 'EXECUTOR' | 'AGENT',
   771→  ): Promise<TradingBriefDTO[]> {
   772→    if (briefs.length === 0) return [];
   773→
   774→    // Group by pair
   775→    const byPair = new Map<string, TradingBriefDTO[]>();
   776→    for (const brief of briefs) {
   777→      const existing = byPair.get(brief.pair) || [];
   778→      existing.push(brief);
   779→      byPair.set(brief.pair, existing);
   780→    }
   781→
   782→    const consolidated: TradingBriefDTO[] = [];
   783→
   784→    for (const [pair, pairBriefs] of byPair) {
   785→      if (pairBriefs.length === 0) continue;
   786→
   787→      if (pairBriefs.length === 1) {
   788→        consolidated.push(pairBriefs[0]);
   789→        continue;
   790→      }
   791→
   792→      // Multiple briefs for same pair — check if they agree on direction
   793→      const buyBriefs = pairBriefs.filter(b => b.direction === 'BUY');
   794→      const sellBriefs = pairBriefs.filter(b => b.direction === 'SELL');
   795→
   796→      if (sellBriefs.length === 0) {
   797→        const best = buyBriefs.sort((a, b) => b.confidence - a.confidence)[0];
   798→        consolidated.push(best);
   799→      } else if (buyBriefs.length === 0) {
   800→        const best = sellBriefs.sort((a, b) => b.confidence - a.confidence)[0];
   801→        consolidated.push(best);
   802→      } else {
   803→        // CONFLICTING DIRECTIONS within the same group
   804→        const TF_WEIGHT: Record<string, number> = {
   805→          M1: 2.0, M5: 1.5, M15: 1.2,  // Executor: shorter = stronger
   806→          M30: 2.0, H1: 1.5, H4: 1.2,  // Agent: shorter = stronger (within agent group)
   807→          D1: 0.8, W1: 0.5,
   808→        };
   809→
   810→        let buyScore = buyBriefs.reduce((sum, b) =>
   811→          sum + b.confidence * (TF_WEIGHT[b.timeframe] || 1.0), 0);
   812→        let sellScore = sellBriefs.reduce((sum, b) =>
   813→          sum + b.confidence * (TF_WEIGHT[b.timeframe] || 1.0), 0);
   814→
   815→        // V177 FIX #14: Direction balance enforcement
   816→        // If one direction dominates (>70% of today's executed trades),
   817→        // boost the other direction's scores to encourage diversity
   818→        try {
   819→          const todayStart = new Date();
   820→          todayStart.setHours(0, 0, 0, 0);
   821→          const todayBuyCount = await this.prisma.position.count({
   822→            where: { side: 'BUY', status: { in: ['OPEN', 'CLOSED'] }, openedAt: { gte: todayStart } },
   823→          });
   824→          const todaySellCount = await this.prisma.position.count({
   825→            where: { side: 'SELL', status: { in: ['OPEN', 'CLOSED'] }, openedAt: { gte: todayStart } },
   826→          });
   827→          const totalToday = todayBuyCount + todaySellCount;
   828→          if (totalToday >= 4) { // Only enforce balance after 4+ trades
   829→            const buyRatio = todayBuyCount / totalToday;
   830→            const sellRatio = todaySellCount / totalToday;
   831→            if (sellRatio > 0.70) {
   832→              // SELL dominates — boost BUY score
   833→              buyScore *= 1.3;
   834→              this.logger.debug(`🏛️ V177 Direction balance: SELL dominates (${(sellRatio*100).toFixed(0)}%) — boosting BUY score by 1.3x`);
   835→            } else if (buyRatio > 0.70) {
   836→              // BUY dominates — boost SELL score
   837→              sellScore *= 1.3;
   838→              this.logger.debug(`🏛️ V177 Direction balance: BUY dominates (${(buyRatio*100).toFixed(0)}%) — boosting SELL score by 1.3x`);
   839→            }
   840→          }
   841→        } catch { /* non-critical — balance enforcement is best-effort */ }
   842→
   843→        const winningBriefs = buyScore >= sellScore ? buyBriefs : sellBriefs;
   844→        const losingBriefs = buyScore >= sellScore ? sellBriefs : buyBriefs;
   845→        const winningDirection = buyScore >= sellScore ? 'BUY' : 'SELL';
   846→
   847→        const best = winningBriefs.sort((a, b) => b.confidence - a.confidence)[0];
   848→
   849→        this.logger.log(
   850→          `🏛️ V143 CONSOLIDATION [${group}]: ${pair} has conflicting directions ` +
   851→          `(BUY=${buyBriefs.length} score=${buyScore.toFixed(0)}, ` +
   852→          `SELL=${sellBriefs.length} score=${sellScore.toFixed(0)}) ` +
   853→          `→ Winner: ${winningDirection} (best confidence=${best.confidence}%, ` +
   854→          `timeframe=${best.timeframe})`
   855→        );
   856→
   857→        consolidated.push(best);
   858→
   859→        // V143: Only cancel losing briefs WITHIN the same timeframe group.
   860→        // Previously, ALL losing briefs were cancelled, which destroyed
   861→        // agent signals when executor briefs won the vote.
   862→        for (const losing of losingBriefs) {
   863→          try {
   864→            await this.prisma.tradingBrief.update({
   865→              where: { id: losing.id },
   866→              data: {
   867→                isActive: false,
   868→                reviewStatus: 'CANCELLED',
   869→                analysisSummary: `V143: Cancelled by ${group} consolidation — ${pair} has ${winningDirection} consensus within ${group} timeframes`,
   870→              },
   871→            });
   872→          } catch { /* non-critical */ }
   873→        }
   874→      }
   875→    }
   876→
   877→    return consolidated;
   878→  }

   987→   * or empty string if no relevant news found (non-blocking).
   988→   */
   989→  private async _fetchNewsContextForPair(pair: string): Promise<string> {
   990→    try {
   991→      // Extract base symbol from pair (e.g., 'BTC' from 'BTC/USDT')
   992→      const baseSymbol = pair.split('/')[0];
   993→
   994→      // 1. Try RAG-based retrieval (semantic search — most relevant results)
   995→      const ragContext = await this.ragService.retrieveRelevantContext(
   996→        `${baseSymbol} cryptocurrency market news sentiment`,
   997→        5,
   998→      );
   999→      if (ragContext) {
  1000→        this.logger.debug(`🏛️ V143: RAG context found for ${pair}: ${ragContext.length} chars`);
  1001→      }
  1002→
  1003→      // 2. Also fetch latest news from NewsService (structured data with sentiment scores)
  1004→      const latestNews = await this.newsService.getLatestNews({
  1005→        symbol: baseSymbol,
  1006→        limit: 5,
  1007→      });
  1008→
  1009→      // Format structured news into context
  1010→      let newsContext = '';
  1011→
  1012→      if (latestNews && latestNews.length > 0) {
  1013→        const newsItems = latestNews.slice(0, 5).map((article: any, i: number) => {
  1014→          const sentiment = article.sentimentLabel || 'neutral';
  1015→          const impact = article.impactLevel || 'medium';
  1016→          const score = typeof article.sentiment === 'number' ? article.sentiment.toFixed(2) : '0';
  1017→          const title = article.translatedTitle || article.title || '';
  1018→          const summary = article.summary || '';
  1019→          const assets = article.affectedAssets || '';
  1020→          const hoursAgo = article.publishedAt
  1021→            ? Math.round((Date.now() - new Date(article.publishedAt).getTime()) / (60 * 60 * 1000))
  1022→            : '?';
  1023→
  1024→          return `[${i + 1}] (${sentiment}, تأثير=${impact}, نقاط=${score}, منذ ${hoursAgo}ساعة) ${title}${summary ? ' — ' + summary : ''}${assets ? ' | أصول متأثرة: ' + assets : ''}`;
  1025→        }).join('\n');
  1026→
  1027→        // Calculate aggregate news sentiment
  1028→        const sentimentScores = latestNews
  1029→          .map((a: any) => typeof a.sentiment === 'number' ? a.sentiment : 0)
  1030→          .filter((s: number) => s !== 0);
  1031→        const avgSentiment = sentimentScores.length > 0
  1032→          ? sentimentScores.reduce((a: number, b: number) => a + b, 0) / sentimentScores.length
  1033→          : 0;
  1034→        const highImpactCount = latestNews.filter(
  1035→          (a: any) => a.impactLevel === 'high' || a.impactLevel === 'HIGH'
  1036→        ).length;
  1037→
  1038→        const sentimentDirection = avgSentiment > 0.2 ? 'إيجابي 🟢' : avgSentiment < -0.2 ? 'سلبي 🔴' : 'محايد ⚪';
  1039→        const riskLevel = highImpactCount > 0 ? 'عالي ⚠️' : 'مقبول ✅';
  1040→
  1041→        newsContext = `\n\n📰 سياق الأخبار المحللة لـ ${pair} (${latestNews.length} خبر حديث):\n` +
  1042→          `الملخص: اتجاه المشاعر ${sentimentDirection} (المعدل=${avgSentiment.toFixed(2)})، مستوى المخاطرة: ${riskLevel} (${highImpactCount} خبر عالي التأثير)\n` +
  1043→          `الأخبار:\n${newsItems}`;
  1044→      }
  1045→
  1046→      // Combine RAG context (semantic relevance) + structured news (sentiment scores)
  1047→      // + market sentiment from rouatradingnews (Fear & Greed, Arab sentiment, geopolitical risk)
  1048→      const parts: string[] = [];
  1049→
  1050→      if (ragContext) parts.push(`📚 سياق RAG ذي الصلة:\n${ragContext}`);
  1051→      if (newsContext) parts.push(newsContext);
  1052→
  1053→      // V145: Also inject market sentiment from rouatradingnews
  1054→      try {
  1055→        const marketSentiment = await this.newsIntegration.getSentimentForAI();
  1056→        if (marketSentiment) parts.push(marketSentiment);
  1057→      } catch { /* non-blocking */ }
  1058→
  1059→      const combined = parts.join('\n\n---\n\n');
  1060→
  1061→      if (combined) {
  1062→        this.logger.log(`🏛️ V143: News context injected for ${pair} (${combined.length} chars, ${latestNews?.length || 0} articles, RAG=${ragContext ? 'yes' : 'no'})`);
  1063→      } else {
  1064→        this.logger.debug(`🏛️ V143: No news context for ${pair} — proceeding without news`);
  1065→      }
  1066→
  1067→      return combined;
  1068→    } catch (error: any) {
  1069→      // Non-blocking: news context failure should NEVER prevent brief generation
  1070→      this.logger.warn(`🏛️ V143: News context fetch failed for ${pair}: ${error.message} — proceeding without news`);
  1071→      return '';
  1072→    }
  1073→  }

  1075→  /**
  1076→   * V143: Calculate news risk score for a pair.
  1077→   * Returns a value from -1 (strong sell signal from news) to +1 (strong buy signal),
  1078→   * along with risk flags that the executor can use to gate trades.
  1079→   *
  1080→   * This is used by both the Strategic Council (to adjust confidence)
  1081→   * and the Smart Executor (as a pre-execution risk gate).
  1082→   */
  1083→  async getNewsRiskScore(pair: string): Promise<{
  1084→    score: number;          // -1 to +1 (negative = bearish news, positive = bullish news)
  1085→    riskLevel: 'low' | 'medium' | 'high' | 'critical';
  1086→    opposingNews: boolean;  // true if high-impact news opposes the brief direction
  1087→    sentimentLabel: string; // human-readable label
  1088→    highImpactCount: number;
  1089→    recentArticleCount: number;
  1090→  }> {
  1091→    const defaultResult = {
  1092→      score: 0,
  1093→      riskLevel: 'low' as const,
  1094→      opposingNews: false,
  1095→      sentimentLabel: 'لا أخبار متاحة',
  1096→      highImpactCount: 0,
  1097→      recentArticleCount: 0,
  1098→    };
  1099→
  1100→    try {
  1101→      const baseSymbol = pair.split('/')[0];
  1102→      const latestNews = await this.newsService.getLatestNews({
  1103→        symbol: baseSymbol,
  1104→        limit: 10,
  1105→      });
  1106→
  1107→      if (!latestNews || latestNews.length === 0) return defaultResult;
  1108→
  1109→      // Filter to last 6 hours only (stale news is irrelevant)
  1110→      const sixHoursAgo = new Date(Date.now() - 6 * 60 * 60 * 1000);
  1111→      const recentNews = latestNews.filter((a: any) =>
  1112→        a.publishedAt && new Date(a.publishedAt) >= sixHoursAgo
  1113→      );
  1114→
  1115→      if (recentNews.length === 0) return defaultResult;
  1116→
  1117→      // Calculate weighted sentiment score
  1118→      let weightedScore = 0;
  1119→      let totalWeight = 0;
  1120→      let highImpactCount = 0;
  1121→
  1122→      for (const article of recentNews) {
  1123→        const sentiment = typeof article.sentiment === 'number' ? article.sentiment : 0;
  1124→        const impact = article.impactLevel?.toLowerCase();
  1125→        const hoursAgo = article.publishedAt
  1126→          ? (Date.now() - new Date(article.publishedAt).getTime()) / (60 * 60 * 1000)
  1127→          : 24;
  1128→
  1129→        // Weight: high impact = 3x, medium = 2x, low = 1x
  1130→        // Decay: recent news gets more weight (1.0 at 0h → 0.2 at 6h)
  1131→        const impactWeight = impact === 'high' ? 3 : impact === 'medium' ? 2 : 1;
  1132→        const timeDecay = Math.max(0.2, 1 - (hoursAgo / 6));
  1133→        const weight = impactWeight * timeDecay;
  1134→
  1135→        weightedScore += sentiment * weight;
  1136→        totalWeight += weight;
  1137→
  1138→        if (impact === 'high') highImpactCount++;
  1139→      }
  1140→
  1141→      const score = totalWeight > 0 ? Math.max(-1, Math.min(1, weightedScore / totalWeight)) : 0;
  1142→
  1143→      // Determine risk level
  1144→      let riskLevel: 'low' | 'medium' | 'high' | 'critical';
  1145→      if (highImpactCount >= 3 && Math.abs(score) > 0.5) riskLevel = 'critical';
  1146→      else if (highImpactCount >= 2 && Math.abs(score) > 0.3) riskLevel = 'high';
  1147→      else if (highImpactCount >= 1 || Math.abs(score) > 0.3) riskLevel = 'medium';
  1148→      else riskLevel = 'low';
  1149→
  1150→      const sentimentLabel = score > 0.3 ? 'إيجابي قوي' : score > 0.1 ? 'إيجابي خفيف'
  1151→        : score < -0.3 ? 'سلبي قوي' : score < -0.1 ? 'سلبي خفيف' : 'محايد';
  1152→
  1153→      return {
  1154→        score,
  1155→        riskLevel,
  1156→        opposingNews: false, // Determined by caller based on brief direction
  1157→        sentimentLabel,
  1158→        highImpactCount,
  1159→        recentArticleCount: recentNews.length,
  1160→      };
  1161→    } catch (error: any) {
  1162→      this.logger.warn(`🏛️ V143: News risk score failed for ${pair}: ${error.message}`);
  1163→      return defaultResult;
  1164→    }
  1165→  }

  1229→        const quote = await this.exchangeService.getQuote(pair);

  1292→  /**
  1293→   * Analyze a pair on a specific timeframe
  1294→   * 1. Check if there's an existing active brief
  1295→   * 2. Get AI consensus analysis
  1296→   * 3. Decide: issue new, modify existing, or cancel
  1297→   */
  1298→  private async _analyzePairTimeframe(
  1299→    pair: string,
  1300→    timeframe: BriefTimeframe,
  1301→    currentPrice: number,
  1302→    result: CouncilSessionResult,
  1303→  ): Promise<void> {
  1304→    // Find existing active brief for this pair+timeframe
  1305→    const existingBrief = await this.prisma.tradingBrief.findFirst({
  1306→      where: {
  1307→        pair,
  1308→        timeframe: timeframe as any,
  1309→        isActive: true,
  1310→        reviewStatus: { in: ['ACTIVE', 'MODIFIED'] },
  1311→      },
  1312→    });
  1313→

  1315→    // This is the KEY integration point — previously the council was news-blind.
  1316→    // Now it sees recent news sentiment, impact levels, and affected assets
  1317→    // before making BUY/SELL/HOLD decisions.
  1318→    const newsContext = await this._fetchNewsContextForPair(pair);
  1319→    const newsRisk = await this.getNewsRiskScore(pair);
  1320→

  1322→    // FIX: Pass forceFresh=true to bypass stale Redis cache from startup session.
  1323→    // The startup session (30s after boot) produces fallback/HOLD results that
  1324→    // get cached for 10 minutes, blocking all subsequent sessions from issuing briefs.
  1325→    // By forcing fresh AI calls, each Council session gets the CURRENT model state.
  1326→    // V143: News context is injected via Redis cache key — the orchestrator reads
  1327→    // it as part of the prompt context when generating analysis.
  1328→    const consensus = await this.orchestrator.getConsensusAnalysis(pair, {
  1329→      forceFresh: true,
  1330→      newsContext: newsContext || undefined,
  1331→    } as any);
  1332→

  1334→    // If news sentiment strongly opposes the AI recommendation, reduce confidence.
  1335→    // If news sentiment supports the AI recommendation, boost confidence slightly.
  1336→    let newsAdjustedConfidence = consensus.consensusScore;
  1337→    const newsDirection: BriefDirection = consensus.recommendation === 'BUY' ? 'BUY' : 'SELL';
  1338→    const newsSupportsDirection = (newsDirection === 'BUY' && newsRisk.score > 0.1) ||
  1339→                                   (newsDirection === 'SELL' && newsRisk.score < -0.1);
  1340→    const newsOpposesDirection = (newsDirection === 'BUY' && newsRisk.score < -0.3) ||
  1341→                                  (newsDirection === 'SELL' && newsRisk.score > 0.3);
  1342→

  1344→      newsAdjustedConfidence = Math.max(0, consensus.consensusScore - 15);
  1345→      this.logger.warn(`🏛️ V143: News OPPOSES ${newsDirection} for ${pair} (news score=${newsRisk.score.toFixed(2)}, risk=${newsRisk.riskLevel}) — confidence reduced ${consensus.consensusScore}% → ${newsAdjustedConfidence}%`);
  1346→    } else if (newsOpposesDirection) {
  1347→      newsAdjustedConfidence = Math.max(0, consensus.consensusScore - 8);
  1348→      this.logger.log(`🏛️ V143: News slightly opposes ${newsDirection} for ${pair} (news score=${newsRisk.score.toFixed(2)}) — confidence adjusted ${consensus.consensusScore}% → ${newsAdjustedConfidence}%`);
  1349→    } else if (newsSupportsDirection) {
  1350→      newsAdjustedConfidence = Math.min(95, consensus.consensusScore + 5);
  1351→      this.logger.log(`🏛️ V143: News supports ${newsDirection} for ${pair} (news score=${newsRisk.score.toFixed(2)}) — confidence boosted ${consensus.consensusScore}% → ${newsAdjustedConfidence}%`);
  1352→    }
  1353→
  1354→    // V143: Apply news-adjusted confidence directly to the consensus object.
  1355→    // This way, all downstream code (including the effectiveConsensus = consensus line)
  1356→    // uses the news-adjusted values.
  1357→    consensus.consensusScore = newsAdjustedConfidence;
  1358→    if (newsContext) {
  1359→      consensus.masterStrategy = (consensus.masterStrategy || '') +
  1360→        `\n\n📰 سياق الأخبار: مشاعر=${newsRisk.sentimentLabel}, مخاطر=${newsRisk.riskLevel}, نقاط=${newsRisk.score.toFixed(2)} (${newsRisk.recentArticleCount} خبر حديث)`;
  1361→    }
  1362→
  1363→    // FIX: When AI models fail (isFallback=true, confidence=0), generate a technical-analysis
  1364→    // based brief instead of cancelling everything. This prevents the entire pipeline from
  1365→    // stalling when AI providers are down.
  1366→    const isAIFallback = consensus.isFallback === true || consensus.consensusScore === 0;
  1367→
  1368→    // FIX: Detailed decision logging — helps diagnose why briefs aren't being created.
  1369→    // Previously, the Council silently produced 0 briefs with no explanation.
  1370→    // V143: Include news risk info in diagnostics.
  1371→    this.logger.log(
  1372→      `🏛️ Decision point for ${pair} ${timeframe}: ` +
  1373→      `recommendation=${consensus.recommendation}, score=${consensus.consensusScore}%, ` +
  1374→      `isFallback=${isAIFallback}, analyses=${consensus.analyses?.length || 0}, ` +
  1375→      `newsRisk=${newsRisk.riskLevel}(${newsRisk.score.toFixed(2)}), ` +
  1376→      `existingBrief=${existingBrief ? existingBrief.id : 'none'}`,
  1377→    );
  1378→    result.diagnostics?.push(`${pair} ${timeframe}: rec=${consensus.recommendation} score=${consensus.consensusScore}% fallback=${isAIFallback} models=${consensus.analyses?.length || 0} newsRisk=${newsRisk.riskLevel}(${newsRisk.score.toFixed(2)})`);
  1379→
  1380→    // FIX: Even when AI gives HOLD, try technical analysis as fallback.
  1381→    // In active Forex markets, there's ALWAYS a direction. The AI saying HOLD
  1382→    // doesn't mean the market is flat — it means the AI is being cautious.
  1383→    // We use technical momentum to override AI caution and generate actionable signals.
  1384→    if (!isAIFallback && consensus.recommendation === 'HOLD') {
  1385→      // AI said HOLD — try technical analysis override before giving up
  1386→      const technicalOverride = await this._generateTechnicalFallbackBrief(pair, timeframe, currentPrice);
  1387→      if (technicalOverride && technicalOverride.recommendation !== 'HOLD') {
  1388→        // Technical analysis found a direction — use it instead of HOLD
  1389→        this.logger.log(`🏛️ Technical override: AI said HOLD for ${pair} ${timeframe}, but momentum shows ${technicalOverride.recommendation}`);
  1390→        // Create brief using technical analysis
  1391→        const direction: BriefDirection = technicalOverride.recommendation === 'BUY' ? 'BUY' : 'SELL';
  1392→        let { entryPrice, stopLoss, takeProfit, strictRules } = this._calculateLevels(currentPrice, direction, timeframe);
  1393→        ({ stopLoss, takeProfit } = this._validateAndFixLevels(direction, entryPrice, stopLoss, takeProfit, timeframe));
  1394→
  1395→        if (existingBrief) {
  1396→          const sameDirection = existingBrief.direction === direction;
  1397→          const priceDiff = Math.abs(Number(existingBrief.entryPrice) - entryPrice) / entryPrice;
  1398→          if (sameDirection && priceDiff < 0.005) {

  1800→      // A low-confidence signal with proper SL/TP is ALWAYS better than no signal,
  1801→      // because the Smart Executor can't execute anything without a brief.
  1802→      // Use deterministic hash-based direction so the same pair always gets the
  1803→      // same direction across sessions (prevents flip-flopping).
  1804→      this.logger.warn(`🏛️ Technical fallback error for ${pair}: ${err.message} — using deterministic fallback`);
  1805→
  1806→      // Deterministic direction based on pair name hash + current hour
  1807→      // This ensures: (1) same pair gets consistent direction, (2) direction changes hourly
  1808→      const hash = this._deterministicHash(pair + new Date().getUTCHours().toString());
  1809→      const fallbackDir: 'BUY' | 'SELL' = hash % 2 === 0 ? 'BUY' : 'SELL';
  1810→      const fallbackConfidence = 58; // V175: رُفع ليتجاوز default MIN_CONSENSUS_SCORE=55

  1816→        analyses: [
  1817→          { role: 'محلل تقني', model: 'Technical/Deterministic-Fallback', vote: fallbackDir, confidence: fallbackConfidence, reason: `إشارة احتياطية حتمية لـ ${pair} — بيانات السوق غير متاحة` },
  1818→        ],
  1819→      };
  1820→    }
  1821→  }

  1823→  /**
  1824→   * FIX: Simple deterministic hash for consistent direction assignment.
  1825→   * Same input always produces the same output, preventing direction flip-flopping.
  1826→   */
  1827→  private _deterministicHash(input: string): number {
  1828→    let hash = 0;
  1829→    for (let i = 0; i < input.length; i++) {
  1830→      const char = input.charCodeAt(i);
  1831→      hash = ((hash << 5) - hash) + char;
  1832→      hash = hash & hash; // Convert to 32-bit integer
  1833→    }
  1834→    return Math.abs(hash);
  1835→  }

  1837→  /**
  1838→   * Calculate entry price, stop loss, take profit, and strict rules
  1839→   * based on current price, direction, and timeframe
  1840→   */
  1841→  private _calculateLevels(
  1842→    currentPrice: number,
  1843→    direction: BriefDirection,
  1844→    timeframe: BriefTimeframe,
  1845→  ): {
  1846→    entryPrice: number;
  1847→    stopLoss: number;
  1848→    takeProfit: number;
  1849→    strictRules: StrictRules;
  1850→  } {
  1851→    const { sl, tp, maxSlippage } = TIMEFRAME_RR[timeframe];
  1852→
  1853→    let entryPrice: number;
  1854→    let stopLoss: number;
  1855→    let takeProfit: number;
  1856→
  1857→    if (direction === 'BUY') {
  1858→      entryPrice = currentPrice;
  1859→      stopLoss = currentPrice * (1 - sl);
  1860→      takeProfit = currentPrice * (1 + tp);
  1861→    } else {
  1862→      entryPrice = currentPrice;
  1863→      stopLoss = currentPrice * (1 + sl);
  1864→      takeProfit = currentPrice * (1 - tp);
  1865→    }
  1866→
  1867→    const strictRules: StrictRules = {
  1868→      maxEntryPrice: direction === 'BUY' ? currentPrice * (1 + maxSlippage) : undefined,
  1869→      minEntryPrice: direction === 'SELL' ? currentPrice * (1 - maxSlippage) : undefined,
  1870→      maxSlippage,
  1871→    };
  1872→
  1873→    return { entryPrice, stopLoss, takeProfit, strictRules };
  1874→  }

  1876→  private _validateAndFixLevels(
  1877→    direction: BriefDirection,
  1878→    entryPrice: number,
  1879→    stopLoss: number,
  1880→    takeProfit: number,
  1881→    timeframe: BriefTimeframe,
  1882→  ): { stopLoss: number; takeProfit: number } {
  1883→    const { sl, tp } = TIMEFRAME_RR[timeframe];
  1884→    const valid = direction === 'BUY'
  1885→      ? stopLoss < entryPrice && takeProfit > entryPrice
  1886→      : stopLoss > entryPrice && takeProfit < entryPrice;
  1887→    if (!valid) {
  1888→      this.logger.warn(`🏛️ SL/TP invalid for ${direction} @ ${entryPrice}: SL=${stopLoss} TP=${takeProfit} — recalculating`);
  1889→      return direction === 'BUY'
  1890→        ? { stopLoss: entryPrice * (1 - sl), takeProfit: entryPrice * (1 + tp) }
  1891→        : { stopLoss: entryPrice * (1 + sl), takeProfit: entryPrice * (1 - tp) };
  1892→    }
  1893→    return { stopLoss, takeProfit };
  1894→  }
