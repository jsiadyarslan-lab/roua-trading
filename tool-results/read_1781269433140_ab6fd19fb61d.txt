    42→  private readonly exchangeCacheTimestamps = new Map<string, number>(); // TTL tracking
    43→  private readonly EXCHANGE_CACHE_TTL_MS = 5 * 60 * 1000; // 5 minutes
    44→  private readonly MAX_CACHE_SIZE = 50; // prevent unbounded growth

    67→  private _cleanExchangeCache(): void {
    68→    const now = Date.now();
    69→    let cleaned = 0;
    70→    for (const [key] of this.exchangeCache.entries()) {
    71→      const ts = this.exchangeCacheTimestamps.get(key) || 0;
    72→      if (now - ts > this.EXCHANGE_CACHE_TTL_MS) {
    73→        this.exchangeCache.delete(key);
    74→        this.exchangeCacheTimestamps.delete(key);
    75→        cleaned++;
    76→      }
    77→    }
    78→    if (cleaned > 0) {
    79→      this.logger.debug(`🗑️ ExchangeCache: cleaned ${cleaned} expired instances (${this.exchangeCache.size} remaining)`);
    80→    }
    81→    // Hard limit: evict oldest if over MAX_CACHE_SIZE
    82→    if (this.exchangeCache.size > this.MAX_CACHE_SIZE) {
    83→      const oldest = [...this.exchangeCache.entries()].sort((a, b) => (this.exchangeCacheTimestamps.get(a[0]) || 0) - (this.exchangeCacheTimestamps.get(b[0]) || 0));
    84→      oldest.slice(0, this.exchangeCache.size - this.MAX_CACHE_SIZE).forEach(([k]) => { this.exchangeCache.delete(k); this.exchangeCacheTimestamps.delete(k); });
    85→    }
    86→  }

    93→  /**
    94→   * Place a new order
    95→   * 1. Validate the request
    96→   * 2. Run risk checks
    97→   * 3. Execute on the exchange via CCXT
    98→   * 4. Record in database
    99→   * 5. Open/update position
   100→   */
   101→  async placeOrder(
   102→    userId: string,
   103→    request: PlaceOrderRequest,
   104→    ipAddress?: string,
   105→    userAgent?: string,
   106→  ) {
   107→    this.logger.log(
   108→      `📋 Order request: ${request.side} ${request.quantity} ${request.symbol} (${request.type})`,
   109→    );
   110→
   111→    // #1 FIX: Round quantity to appropriate precision BEFORE execution.
   112→    // This ensures margin = quantity * price matches exactly,
   113→    // preventing rounding errors that cause margin drift.
   114→    // Different asset classes need different precision:
   115→    // - Crypto: 6 decimal places (e.g., 0.001234 BTC)
   116→    // - Forex: 2 decimal places (e.g., 1000 units)
   117→    // - Stocks: 0 decimal places (whole shares)
   118→    const meta = getSymbolMetadata(request.symbol);
   119→    const precision = meta.assetClass === AssetClass.CRYPTO ? 6
   120→      : meta.assetClass === AssetClass.FOREX ? 2
   121→      : meta.assetClass === AssetClass.COMMODITY ? 2
   122→      : 0;
   123→    request.quantity = Math.round(request.quantity * Math.pow(10, precision)) / Math.pow(10, precision);
   124→
   125→    if (request.quantity <= 0) {
   126→      throw new BadRequestException(
   127→        `الكمية بعد التقريب أصبحت صفراً — قيمة الطلب صغيرة جداً (${request.quantity})`,
   128→      );
   129→    }
   130→
   131→    // Step 1: Validate credential ownership
   132→    // DATA ISOLATION: Use findFirst with userId to prevent accessing other users' credentials
   133→    const credential = await this.prisma.exchangeCredential.findFirst({
   134→      where: { id: request.credentialId, userId },
   135→    });
   136→
   137→    if (!credential) {
   138→      throw new NotFoundException('بيانات الاعتماد غير موجودة');
   139→    }
   140→
   141→    if (!credential.isValid) {
   142→      throw new BadRequestException(
   143→        'بيانات الاعتماد غير صالحة — يرجى التحقق من مفتاح API',
   144→      );
   145→    }
   146→
   147→    // Step 2: Check if credential has trade permission
   148→    // FIX: Skip permissions check for paper-trading and test exchange credentials.
   149→    // Paper-trading credentials have 'paper' as the API key (not real keys),
   150→    // and their permissions field is often the Prisma default "read" (a plain
   151→    // string, not a JSON array). JSON.parse("read") throws SyntaxError,
   152→    // which crashes the entire placeOrder — meaning paper trades NEVER create
   153→    // Position records in the DB, causing trades to disappear on page refresh.
   154→    // Paper trading is simulation — there are no real API permissions to check.
   155→    const isTestExchange = ['paper-trading', 'paper', 'demo', 'sandbox', 'simulation'].includes(credential.exchange)
   156→      || credential.exchange.endsWith('_test')
   157→      || credential.exchange.endsWith('_paper')
   158→      || credential.exchange.endsWith('-test')
   159→      || credential.exchange.endsWith('-paper');
   160→
   161→    if (!isTestExchange) {
   162→      const permissions = JSON.parse(credential.permissions || '["read"]');
   163→      if (!permissions.includes('trade')) {
   164→        throw new ForbiddenException(
   165→          'مفتاح API لا يملك صلاحية التداول — أضف مفتاحاً بصلاحية trade',
   166→        );
   167→      }
   168→    }
   169→
   170→    // Step 3: Get current market price for risk check
   171→    // FIX: For MARKET orders, use the provided price if available (e.g., from SmartExecutor
   172→    // which already fetched it). Only fetch from ExchangeService as fallback.
   173→    // Previously, MARKET orders always re-fetched the price, which could fail on Railway
   174→    // for some pairs, causing paper trades to be rejected.
   175→    let currentPrice = request.price;
   176→    if (!currentPrice) {
   177→      try {
   178→        const quote = await this.exchangeService.getQuote(request.symbol);
   179→        currentPrice = quote.price;
   180→      } catch (error: any) {
   181→        throw new BadRequestException(
   182→          `فشل في جلب سعر السوق لـ ${request.symbol}: ${error.message}`,
   183→        );
   184→      }
   185→    }
   186→
   187→    // Step 4: Risk checks — SKIPPED if already validated by RiskGatekeeper.
   188→    // V176 FIX: Removed duplicate RiskManager.checkOrderRisk() call.
   189→    // Previously, the V1 pipeline ran TWO risk checks:
   190→    //   1. RiskGatekeeper.validateOrder() in TradingController (5-point check)
   191→    //   2. RiskManager.checkOrderRisk() here (overlapping checks)
   192→    // This caused double latency and could produce contradictory results
   193→    // (e.g., RiskGatekeeper allows but RiskManager blocks, or vice versa).
   194→    // Now: RiskGatekeeper is the SOLE risk gateway for V1 (same as V2).
   195→    // RiskManager is still used for position sizing calculations.
   196→    // If RiskGatekeeper was NOT called (e.g., internal call from OrderDispatcher),
   197→    // the `skipRiskCheck` flag can be set to false to enforce the check here.
   198→    // V178 FIX: Inverted default — risk check runs by DEFAULT (skipRiskCheck === true means skip).
   199→    // Previously: skipRiskCheck !== false → undefined = skip (INSECURE).
   200→    // Now: skipRiskCheck === true → must explicitly opt-in to skip.
   201→    // Controllers set skipRiskCheck=true after RiskGatekeeper validates.
   202→    // Internal calls without a controller MUST go through risk check.
   203→    const skipRiskCheck = request.skipRiskCheck === true;
   204→    if (!skipRiskCheck) {
   205→      const riskCheck = await this.riskManager.checkOrderRisk(
   206→        userId,
   207→        request.symbol,
   208→        request.side,
   209→        request.quantity,
   210→        currentPrice,
   211→        credential.exchange,
   212→        credential.id,
   213→      );
   214→
   215→      if (!riskCheck.allowed) {
   216→        await this.auditService.log({
   217→          userId,
   218→          action: 'ORDER_REJECTED_RISK',
   219→          resource: 'order',
   220→          details: JSON.stringify({
   221→            symbol: request.symbol,
   222→            side: request.side,
   223→            reason: riskCheck.reason,
   224→          }),
   225→          ipAddress,
   226→          userAgent,
   227→        });
   228→
   229→        throw new ForbiddenException(
   230→          `🛡️ تم رفض الطلب: ${riskCheck.reason}`,
   231→        );
   232→      }
   233→    }
   234→
   235→    // Step 5: Execute order on the exchange
   236→    // FIX: Handle paper-trading separately — CCXT doesn't have a 'paper-trading' exchange,
   237→    // so calling _executeOnExchange with 'paper-trading' would always fail with
   238→    // "exchange not supported". This was the ROOT CAUSE of zero trade executions:
   239→    // RiskGatekeeper correctly bypasses paper-trading, but TradingService always
   240→    // tried to execute via CCXT, which fails for paper-trading credentials.
   241→    let execution: any;
   242→    if (credential.exchange === 'paper-trading') {
   243→      execution = await this._executePaperTrade(request, currentPrice, userId);
   244→    } else {
   245→      execution = await this._executeOnExchange(
   246→        credential.exchange,
   247→        credential.id,
   248→        request,
   249→        userId,
   250→      );
   251→    }
   252→
   253→    if (!execution.success) {
   254→      // Record the failed order
   255→      // idempotencyKey is required (String @unique)
   256→      const order = await this.prisma.order.create({
   257→        data: {
   258→          userId,
   259→          exchangeCredentialId: request.credentialId,
   260→          exchange: credential.exchange,
   261→          symbol: request.symbol,
   262→          side: request.side,
   263→          type: request.type,
   264→          status: 'REJECTED' as PrismaOrderStatus,
   265→          quantity: request.quantity,
   266→          price: request.price ?? null,
   267→          stopLoss: request.stopLoss ?? null,
   268→          idempotencyKey: request.idempotencyKey || `legacy-${Date.now()}-${crypto.randomUUID()}`,
   269→        },
   270→      });
   271→
   272→      await this.auditService.log({
   273→        userId,
   274→        action: 'ORDER_REJECTED_EXCHANGE',
   275→        resource: 'order',
   276→        details: JSON.stringify({
   277→          orderId: order.id,
   278→          symbol: request.symbol,
   279→          error: execution.error,
   280→        }),
   281→        ipAddress,
   282→        userAgent,
   283→      });
   284→
   285→      throw new BadRequestException(
   286→        `فشل في تنفيذ الطلب: ${execution.error}`,
   287→      );
   288→    }
   289→
   290→    // Step 6-9: Record order, update position, record trade, update signal — all in one transaction
   291→    // Note: averagePrice (not averageFillPrice) is the correct field name
   292→    // idempotencyKey is required (String @unique)
   293→    const order = await this.prisma.$transaction(async (tx) => {
   294→      const createdOrder = await tx.order.create({
   295→        data: {
   296→          userId,
   297→          exchangeCredentialId: request.credentialId,
   298→          exchange: credential.exchange,
   299→          symbol: request.symbol,
   300→          side: request.side,
   301→          type: request.type,
   302→          status:
   303→            (execution.filledQuantity || 0) >= request.quantity
   304→              ? ('FILLED' as PrismaOrderStatus)
   305→              : ('PARTIALLY_FILLED' as PrismaOrderStatus),
   306→          quantity: request.quantity,
   307→          price: request.price ?? null,
   308→          stopLoss: request.stopLoss ?? null,
   309→          filledQuantity: execution.filledQuantity || 0,
   310→          averagePrice: execution.averagePrice,
   311→          fee: execution.fee ?? null,
   312→          feeCurrency: execution.feeCurrency ?? null,
   313→          exchangeOrderId: execution.exchangeOrderId,
   314→          idempotencyKey: request.idempotencyKey || `legacy-${Date.now()}-${crypto.randomUUID()}`,
   315→        },
   316→      });
   317→
   318→      // Step 7: Update or open position
   319→      await this._updatePosition(userId, createdOrder, request, execution, tx);
   320→
   321→      // Step 8: Record trade
   322→      // ═══════════════════════════════════════════════════
   323→      // FIX: Validate trade data before recording. Previously,
   324→      // execution.filledQuantity could be null/undefined → 0,
   325→      // and execution.averagePrice could be null → currentPrice
   326→      // which might also be 0, producing phantom $0.00 trades.
   327→      // ═══════════════════════════════════════════════════
   328→      const tradeQuantity = execution.filledQuantity || 0;
   329→      const tradePrice = execution.averagePrice || currentPrice;
   330→
   331→      if (tradeQuantity <= 0 || tradePrice <= 0) {
   332→        this.logger.warn(
   333→          `Trade record skipped — invalid quantity (${tradeQuantity}) or price (${tradePrice}) for ${request.symbol}`,
   334→        );
   335→      } else {
   336→        await tx.trade.create({
   337→          data: {
   338→            userId,
   339→            orderId: createdOrder.id,
   340→            credentialId: request.credentialId,
   341→            exchange: credential.exchange,
   342→            symbol: request.symbol,
   343→            side: request.side,
   344→            type: 'ENTRY',
   345→            quantity: tradeQuantity,
   346→            price: tradePrice,
   347→            fee: execution.fee ?? 0,
   348→            feeCurrency: execution.feeCurrency,
   349→            source: request.source || (credential.exchange === 'paper-trading' ? 'auto_paper' : 'user_manual'),
   350→          },
   351→        });
   352→      }
   353→
   354→      // Step 9: If this was triggered by a signal, update signal status
   355→      // DATA ISOLATION: Added userId filter to prevent updating other users' signals
   356→      if (request.signalId) {
   357→        await tx.signal
   358→          .updateMany({
   359→            where: { id: request.signalId, userId },
   360→            data: { status: 'EXECUTED' },
   361→          })
   362→          .catch(() => {}); // Don't fail if signal not found
   363→      }
   364→
   365→      return createdOrder;
   366→    });
   367→
   368→    // Audit log
   369→    await this.auditService.log({
   370→      userId,
   371→      action: 'ORDER_PLACED',
   372→      resource: 'order',
   373→      details: JSON.stringify({
   374→        orderId: order.id,
   375→        symbol: request.symbol,
   376→        side: request.side,
   377→        type: request.type,
   378→        quantity: request.quantity,
   379→        filledQuantity: execution.filledQuantity,
   380→        averagePrice: execution.averagePrice,
   381→        // V176: riskCheck removed from scope — RiskGatekeeper already logged the risk assessment
   382→      }),
   383→      ipAddress,
   384→      userAgent,
   385→    });
   386→
   387→    // V172d FIX: Deduct margin from paperBalance atomically when opening a paper-trading position.
   388→    // Uses $executeRaw to prevent race condition: findUnique→update is NOT atomic.
   389→    // With two concurrent orders, both would read the same balance and each deduct their margin
   390→    // independently, ignoring the other's deduction (double-spending).
   391→    // $executeRaw UPDATE with arithmetic is atomic at the DB level.
   392→    if (credential.exchange === 'paper-trading' && execution.success) {
   393→      try {
   394→        const settings = await this.prisma.agentSettings.findUnique({
   395→          where: { userId },
   396→          select: { paperCryptoLeverage: true, paperForexLeverage: true, paperGoldLeverage: true },
   397→        });
   398→        const meta = getSymbolMetadata(request.symbol);
   399→        const cryptoLev = Number(settings?.paperCryptoLeverage) || 1;
   400→        const forexLev = Number(settings?.paperForexLeverage) || 50;
   401→        const goldLev = Number(settings?.paperGoldLeverage) || 20;
   402→        let leverage = 1;
   403→        if (meta.assetClass === AssetClass.FOREX) leverage = forexLev;
   404→        else if (meta.assetClass === AssetClass.COMMODITY) leverage = goldLev;
   405→        else leverage = cryptoLev;
   406→
   407→        const notional = request.quantity * currentPrice;
   408→        const marginToDeduct = leverage > 1 ? notional / leverage : notional;
   409→
   410→        // V176 FIX: Actually DEDUCT margin from paperBalance when opening a position.
   411→        // Previously (V175), margin was only "locked" (logged) but never deducted,
   412→        // allowing unlimited positions regardless of available balance.
   413→        // Now: paperBalance = paperBalance - marginToDeduct (atomic SQL update).
   414→        // On close: paperBalance = paperBalance + marginToDeduct + pnl.
   415→        // This ensures users cannot open more positions than their balance allows.
   416→        await this.prisma.$executeRaw`
   417→          UPDATE "AgentSettings"
   418→          SET "paperBalance" = "paperBalance" - ${marginToDeduct}
   419→          WHERE "userId" = ${userId}
   420→        `;
   421→        this.logger.log(
   422→          `📝 V176 Paper margin DEDUCTED: -$${marginToDeduct.toFixed(2)} (${request.symbol}, leverage: ${leverage}x)`,
   423→        );
   424→      } catch (err: any) {
   425→        this.logger.warn(`V172d Failed to deduct paper margin on open: ${err.message}`);
   426→      }
   427→    }
   428→
   429→    this.logger.log(
   430→      `✅ Order executed: ${order.id} — ${request.side} ${execution.filledQuantity}/${request.quantity} ${request.symbol} @ ${execution.averagePrice}`,
   431→    );
   432→
   433→    return order;
   434→  }
   435→  async cancelOrder(
   436→    userId: string,
   437→    orderId: string,
   438→    ipAddress?: string,
   439→    userAgent?: string,
   440→  ) {
   441→    // DATA ISOLATION: Use findFirst with userId to prevent accessing other users' orders
   442→    const order = await this.prisma.order.findFirst({
   443→      where: { id: orderId, userId },
   444→    });
   445→
   446→    if (!order) {
   447→      throw new NotFoundException('الطلب غير موجود');
   448→    }
   449→
   450→    if (!['PENDING', 'ACCEPTED', 'PARTIALLY_FILLED'].includes(order.status)) {
   451→      throw new BadRequestException(
   452→        `لا يمكن إلغاء طلب بحالة "${order.status}"`,
   453→      );
   454→    }
   455→
   456→    // Try to cancel on the exchange
   457→    if (order.exchangeOrderId) {
   458→      try {
   459→        const credential =
   460→          await this.prisma.exchangeCredential.findFirst({
   461→            where: { id: order.exchangeCredentialId!, userId },
   462→          });
   463→        if (credential) {
   464→          // SECURITY: Pass userId to verify credential ownership before decrypting
   465→          const { apiKey, apiSecret } =
   466→            await this.credentialsService.decryptCredential(credential.id, userId);
   467→          const exchange = this._getExchangeInstance(credential.exchange, apiKey, apiSecret, credential.id, (credential as any).testnet || false);
   468→          if (exchange) {
   469→            await exchange.cancelOrder(order.exchangeOrderId, order.symbol);
   470→          }
   471→        }
   472→      } catch (error: any) {
   473→        this.logger.warn(
   474→          `Failed to cancel order on exchange: ${error.message}`,
   475→        );
   476→        // Continue with local cancellation even if exchange cancel fails
   477→      }
   478→    }
   479→
   480→    const updated = await this.prisma.order.update({
   481→      where: { id: orderId },
   482→      data: { status: 'CANCELLED' },
   483→    });
   484→
   485→    await this.auditService.log({
   486→      userId,
   487→      action: 'ORDER_CANCELLED',
   488→      resource: 'order',
   489→      details: JSON.stringify({ orderId, symbol: order.symbol }),
   490→      ipAddress,
   491→      userAgent,
   492→    });
   493→
   494→    return updated;
   495→  }

   533→  /**
   534→   * Get all open positions for a user
   535→   * Uses parallel quote fetching and batch DB updates to avoid N+1 queries
   536→   */
   537→  async getOpenPositions(userId: string, credentialId?: string): Promise<any[]> {
   538→    // FIX: Include ALL positions including paper-trading.
   539→    // Paper positions are real simulated trades — they should appear in portfolio.
   540→    // Previously excluded, causing portfolioValue = 0 for paper traders.
   541→    // PositionManagerService also includes them (must be consistent).
   542→    // V209: Added credentialId parameter for server-side filtering by account.
   543→    const where: any = {
   544→      userId,
   545→      status: 'OPEN',
   546→    };
   547→    // V209: Filter by credentialId when provided (for account switching).
   548→    // V212 FIX: Position.credentialId is NOT NULL in schema, so we CANNOT use
   549→    // OR with { credentialId: null } — Prisma rejects it. Use direct filter instead.
   550→    // Trade.credentialId IS nullable, so OR with null is fine for Trade queries.
   551→    if (credentialId) {
   552→      where.credentialId = credentialId;
   553→    }
   554→    const positions = await this.prisma.position.findMany({
   555→      where,
   556→      orderBy: { openedAt: 'desc' },
   557→    });
   558→
   559→    if (positions.length === 0) return [];
   560→
   561→    // Fetch all quotes in parallel
   562→    const quotePromises = positions.map((pos) =>
   563→      this.exchangeService.getQuote(pos.symbol).catch(() => null),
   564→    );
   565→    const quotes = await Promise.allSettled(quotePromises);
   566→
   567→    // Build updates and results
   568→    const updates: any[] = [];
   569→    const results: any[] = [];
   570→
   571→    for (let i = 0; i < positions.length; i++) {
   572→      const position = positions[i];
   573→      const quoteResult = quotes[i];
   574→      const quote = quoteResult.status === 'fulfilled' ? quoteResult.value : null;
   575→
   576→      if (quote && quote.price) {
   577→        const currentPrice = quote.price;
   578→        const entryPrice = position.entryPrice.toNumber();
   579→        const quantity = position.quantity.toNumber();
   580→        const unrealizedPnl =
   581→          position.side === 'BUY'
   582→            ? (currentPrice - entryPrice) * quantity
   583→            : (entryPrice - currentPrice) * quantity;
   584→
   585→        updates.push(
   586→          this.prisma.position.update({
   587→            where: { id: position.id },
   588→            data: {
   589→              currentPrice,
   590→              unrealizedPnl,
   591→              highestPrice: Math.max(
   592→                position.highestPrice?.toNumber() ?? currentPrice,
   593→                currentPrice,
   594→              ),
   595→              lowestPrice: Math.min(
   596→                position.lowestPrice?.toNumber() ?? currentPrice,
   597→                currentPrice,
   598→              ),
   599→            },
   600→          }),
   601→        );
   602→
   603→        // Build enriched position for response
   604→        results.push({
   605→          ...position,
   606→          currentPrice,
   607→          unrealizedPnl,
   608→        });
   609→      } else {
   610→        // No quote available — return position as-is
   611→        this.logger.warn(
   612→          `Failed to update price for ${position.symbol}: quote unavailable`,
   613→        );
   614→        results.push(position);
   615→      }
   616→    }
   617→
   618→    // Batch update in transaction
   619→    if (updates.length > 0) {
   620→      await this.prisma.$transaction(updates).catch((err: any) => {
   621→        this.logger.warn(`Batch position update failed: ${err.message}`);
   622→      });
   623→    }
   624→
   625→    return results;
   626→  }

   628→  /**
   629→   * Get position summary
   630→   */
   631→  async getPositionSummary(userId: string, credentialId?: string) {
   632→    try {
   633→      const positions = await this.getOpenPositions(userId, credentialId);
   634→
   635→      const totalValue = positions.reduce(
   636→        (sum, p) => sum + (typeof p.quantity === 'number' ? p.quantity : Number(p.quantity)) * (Number(p.currentPrice) || Number(p.entryPrice)),
   637→        0,
   638→      );
   639→      const totalUnrealizedPnl = positions.reduce(
   640→        (sum, p) => sum + (Number(p.unrealizedPnl) || 0),
   641→        0,
   642→      );
   643→      const totalRealizedPnl = positions.reduce(
   644→        (sum, p) => sum + (Number(p.realizedPnl) || 0),
   645→        0,
   646→      );
   647→      // V148 FIX: Calculate leverage-aware used margin instead of returning
   648→      // totalValue (full notional) as margin. Previously, the frontend used
   649→      // totalValue as "initialMargin" which is WRONG for leveraged positions.
   650→      // For forex (50:1 leverage), a $108K notional position only needs $2,160 margin.
   651→      // Using totalValue as margin caused the "مستخدم" to show $108K instead of $2.16K.
   652→      const usedMargin = positions.reduce(
   653→        (sum, p) => sum + calculateMargin(
   654→          typeof p.quantity === 'number' ? p.quantity : Number(p.quantity),
   655→          Number(p.currentPrice) || Number(p.entryPrice),
   656→          p.symbol,
   657→        ),
   658→        0,
   659→      );
   660→
   661→      return {
   662→        totalPositions: positions.length,
   663→        totalValue,
   664→        totalUnrealizedPnl,
   665→        totalRealizedPnl,
   666→        usedMargin,
   667→        positions,
   668→      };
   669→    } catch (error: any) {
   670→      this.logger.error(
   671→        `Failed to get position summary: ${error.message}`,
   672→        error.stack,
   673→      );
   674→      // FIX: Previously returned fake "zero" summary that masked real failures
   675→      // (DB down, exchange unreachable). Users saw "0 positions, $0 P&L" and
   676→      // thought their positions were empty when the system was actually broken.
   677→      // Now we throw so the caller gets a proper error response.
   678→      throw new Error(`فشل في جلب ملخص المراكز: ${error.message}`);
   679→    }
   680→  }

   682→  /**
   683→   * Close a position (partially or fully)
   684→   */
   685→  async closePosition(
   686→    userId: string,
   687→    request: ClosePositionRequest,
   688→    ipAddress?: string,
   689→    userAgent?: string,
   690→    _retryCount = 0, // FIX: Internal retry counter for optimistic locking
   691→  ) {
   692→    // FIX: REMOVED the dynamic DDL that dropped unique constraints on Position table
   693→    // on EVERY closePosition() call. This was a dangerous hotfix that:
   694→    //   1. Ran destructive DDL (DROP INDEX CASCADE) inside business logic
   695→    //   2. Executed on every close call (not just once)
   696→    //   3. Could cause data corruption under concurrent close requests
   697→    //   4. Removed ALL unique constraints (except PK), allowing duplicate data
   698→    //
   699→    // The original problem (unique constraint violations during close) is now
   700→    // handled properly by the idempotent close logic below and the race condition
   701→    // fix in _updatePosition() (P2002 catch with retry).
   702→
   703→    // DATA ISOLATION: Use findFirst with userId to prevent accessing other users' positions
   704→    // FIX: Read version for optimistic locking — prevents concurrent close race condition
   705→    const position = await this.prisma.position.findFirst({
   706→      where: { id: request.positionId, userId },
   707→    });
   708→
   709→    if (!position) {
   710→      throw new NotFoundException('المركز غير موجود');
   711→    }
   712→
   713→    // ═══════════════════════════════════════════════════════════════════
   714→    // V214 LAST LINE OF DEFENSE: Block premature close of Agent positions
   715→    //
   716→    // PROBLEM: Old compiled JS on Railway (pre-V184/V213) still has the
   717→    // hardcoded 4h close for Agent positions. Despite V184 removing it
   718→    // and V213 adding a safety net in _monitorOpenPositions, the OLD code
   719→    // is still running and closing Agent positions at exactly 4h 0m.
   720→    //
   721→    // Evidence from June 12, 2026:
   722→    //   - 10+ Agent positions ALL closed at exactly 4h 0m
   723→    //   - closeReason = "Manual" (old default before V176)
   724→    //   - Smart Executor positions also closing at 4h with "Manual"
   725→    //   - This proves old pre-V141/V176 code is running on Railway
   726→    //
   727→    // FIX: Block ANY close of an Agent position that hasn't reached 48h,
   728→    // UNLESS the close is triggered by SL/TP hit (valid trading exit).
   729→    // This works regardless of which service calls closePosition — it's
   730→    // the single point of truth for all position closes.
   731→    // ═══════════════════════════════════════════════════════════════════
   732→    const isAgentPosition = position.source === 'agent';
   733→    const closeReasonStr = (request.closeReason || '').toUpperCase();
   734→    const isSLTPClose = closeReasonStr.includes('STOP_LOSS') || closeReasonStr.includes('TAKE_PROFIT');
   735→    const isManualClose = closeReasonStr === 'MANUAL' || closeReasonStr === '';
   736→
   737→    if (isAgentPosition && !isSLTPClose) {
   738→      const holdingMs = position.openedAt
   739→        ? Date.now() - new Date(position.openedAt).getTime()
   740→        : 0;
   741→      const holdingHours = holdingMs / (60 * 60 * 1000);
   742→      const AGENT_MIN_HOLDING_HOURS = 48;
   743→
   744→      if (holdingHours < AGENT_MIN_HOLDING_HOURS) {
   745→        // Agent position hasn't reached 48h — BLOCK the close
   746→        this.logger.error(
   747→          `🚨 V214 BLOCKED: Attempted to close Agent position ${position.id} (${position.symbol}) ` +
   748→          `at ${holdingHours.toFixed(1)}h — Agent positions must be held for ${AGENT_MIN_HOLDING_HOURS}h minimum. ` +
   749→          `closeReason="${request.closeReason || 'EMPTY'}" — ` +
   750→          `Only SL/TP closes are allowed before ${AGENT_MIN_HOLDING_HOURS}h. ` +
   751→          `This close was likely triggered by OLD code (pre-V184) still running on Railway.`
   752→        );
   753→
   754→        // Instead of throwing (which would break things), just skip and return
   755→        // the position as-is — don't close it
   756→        return {
   757→          order: null,
   758→          pnl: 0,
   759→          position,
   760→          blockedByV214: true,
   761→          reason: `Agent position held ${holdingHours.toFixed(1)}h — minimum is ${AGENT_MIN_HOLDING_HOURS}h`,
   762→        };
   763→      }
   764→
   765→      // Agent position has reached 48h — allow the close
   766→      this.logger.log(
   767→        `✅ V214 ALLOWED: Agent position ${position.id} (${position.symbol}) held ${holdingHours.toFixed(1)}h ` +
   768→        `≥ ${AGENT_MIN_HOLDING_HOURS}h — close allowed. closeReason="${request.closeReason}"`
   769→      );
   770→    }
   771→
   772→    // FIX: Optimistic locking — if another request already closed this position
   773→    // between our read and the upcoming update, the version won't match and we'll
   774→    // retry. This prevents double-close, duplicate EXIT trades, and PnL miscalculation.
   775→    const positionVersion = position.version ?? 0;
   776→
   777→    if (position.status !== 'OPEN') {
   778→      // ── FIX: Idempotent close — if position is already CLOSED/LIQUIDATED,
   779→      // it may still be open on the exchange (e.g. SmartExecutor closed it in
   780→      // DB without closing on exchange). Try to close on exchange anyway and
   781→      // return success instead of rejecting the user's request.
   782→      this.logger.warn(
   783→        `Position ${position.id} (${position.symbol}) status is ${position.status} — attempting exchange close for safety`,
   784→      );
   785→
   786→      // Try to close on exchange if we have valid credentials
   787→      try {
   788→        const staleCredential =
   789→          await this.prisma.exchangeCredential.findFirst({
   790→            where: { id: position.credentialId, userId },
   791→          });
   792→
   793→        if (staleCredential && staleCredential.exchange !== 'paper-trading') {
   794→          const staleSide = position.side === 'BUY' ? 'SELL' : 'BUY';
   795→          await this._executeOnExchange(
   796→            staleCredential.exchange,
   797→            staleCredential.id,
   798→            {
   799→              credentialId: staleCredential.id,
   800→              symbol: position.symbol,
   801→              side: staleSide as PrismaOrderSide,
   802→              type: PrismaOrderType.MARKET,
   803→              quantity: position.quantity.toNumber(),
   804→            },
   805→            userId,
   806→          );
   807→          this.logger.log(
   808→            `Successfully closed position ${position.id} on exchange despite DB status being ${position.status}`,
   809→          );
   810→        }
   811→      } catch (exchangeErr: any) {
   812→        this.logger.warn(
   813→          `Exchange close for already-closed position ${position.id} failed: ${exchangeErr.message}`,
   814→        );
   815→        // Not fatal — position may already be closed on exchange too
   816→      }
   817→
   818→      // Return the existing closed position (idempotent success)
   819→      return {
   820→        order: null,
   821→        pnl: position.realizedPnl?.toNumber() ?? 0,
   822→        position: await this.prisma.position.findUnique({
   823→          where: { id: position.id },
   824→        }),
   825→        alreadyClosed: true,
   826→      };
   827→    }
   828→
   829→    const posQuantity = position.quantity.toNumber();
   830→    const posEntryPrice = position.entryPrice.toNumber();
   831→    const posCurrentPrice = position.currentPrice?.toNumber() ?? null;
   832→    const posRealizedPnl = position.realizedPnl?.toNumber() ?? 0;
   833→    const posStopLoss = position.stopLoss?.toNumber() ?? null;
   834→
   835→    const closeQuantity = request.quantity ?? posQuantity;
   836→    if (closeQuantity > posQuantity) {
   837→      throw new BadRequestException(
   838→        `كمية الإغلاق (${closeQuantity}) أكبر من حجم المركز (${posQuantity})`,
   839→      );
   840→    }
   841→
   842→    // Execute closing order on exchange
   843→    // FIX: Handle paper-trading positions separately. Previously, closePosition
   844→    // always called _executeOnExchange() which requires decrypting credentials.
   845→    // But paper-trading credentials may have been deleted by _startupCleanup(),
   846→    // making paper positions un-closeable (NotFoundException on credential).
   847→    // Now: for paper-trading positions, use _executePaperTrade() directly.
   848→    const credential =
   849→      await this.prisma.exchangeCredential.findFirst({
   850→        where: { id: position.credentialId, userId },
   851→      });
   852→
   853→    const closeSide = position.side === 'BUY' ? 'SELL' : 'BUY';
   854→
   855→    let execution: any;
   856→
   857→    if (position.exchange === 'paper-trading' || !credential) {
   858→      // Paper-trading close: simulate the close using current market price.
   859→      // If credential was deleted by startup cleanup, we can still close the position
   860→      // because paper trading is simulated — no real exchange connection needed.
   861→      //
   862→      // FIX v114: Use entryPrice as the DEFAULT close price for paper trading.
   863→      // Previously, if posCurrentPrice was 0/null, it tried getQuote() which could
   864→      // hang for 30+ seconds when all price providers are exhausted (TwelveData
   865→      // rate-limited, Binance blocked, etc.). This caused close requests to timeout
   866→      // and positions to remain stuck OPEN.
   867→      //
   868→      // Now: Only try getQuote with a 3-second timeout. If it fails or times out,
   869→      // V172: Reduced getQuote timeout 3s → 1s. Paper positions always have
   870→      // currentPrice set by the position monitor — getQuote is rarely needed.
   871→      let closePrice = posCurrentPrice;
   872→      if (!closePrice || closePrice <= 0) {
   873→        try {
   874→          const quotePromise = this.exchangeService.getQuote(position.symbol);
   875→          const quote = await Promise.race([
   876→            quotePromise,
   877→            new Promise<null>((resolve) => setTimeout(() => resolve(null), 1000)),
   878→          ]);
   879→          closePrice = quote?.price || posEntryPrice;
   880→        } catch {
   881→          closePrice = posEntryPrice;
   882→        }
   883→      }
   884→      // Final safety: entryPrice is always available for an OPEN position
   885→      if (!closePrice || closePrice <= 0) {
   886→        closePrice = posEntryPrice;
   887→      }
   888→
   889→      execution = await this._executePaperTrade(
   890→        {
   891→          credentialId: position.credentialId,
   892→          symbol: position.symbol,
   893→          side: closeSide as PrismaOrderSide,
   894→          type: PrismaOrderType.MARKET,
   895→          quantity: closeQuantity,
   896→          price: closePrice,
   897→        },
   898→        closePrice,
   899→        userId,
   900→      );
   901→    } else {
   902→      // Real exchange close: use CCXT
   903→      execution = await this._executeOnExchange(
   904→        credential.exchange,
   905→        credential.id,
   906→        {
   907→          credentialId: credential.id,
   908→          symbol: position.symbol,
   909→          side: closeSide as PrismaOrderSide,
   910→          type: PrismaOrderType.MARKET,
   911→          quantity: closeQuantity,
   912→        },
   913→        userId,
   914→      );
   915→    }
   916→
   917→    if (!execution.success) {
   918→      const errorMsg = execution.error || '';
   919→
   920→      // FIX: Force-close on ANY exchange error when position is paper-trading
   921→      // or exchange is unreachable — not just for the specific Arabic text
   922→      // "رصيد غير متاح". Paper-trading positions have no real exchange state,
   923→      // so force-closing in DB is always safe. For real exchange positions,
   924→      // only force-close when the exchange is unreachable (the close might
   925→      // have partially executed, so we can't safely force-close).
   926→      // User-cancel errors (e.g. "order not found") should also be force-closed
   927→      // because the position is likely already closed on the exchange.
   928→      const isPaperTrading = position.exchange === 'paper-trading';
   929→      const isExchangeUnreachable = /timeout|ECONNREFUSED|ECONNRESET|ETIMEDOUT|network|unreachable/i.test(errorMsg);
   930→      const isUserCancel = /cancel|not found|already closed|unknown order/i.test(errorMsg);
   931→      const isInsufficientBalance = (errorMsg.includes('رصيد') && errorMsg.includes('غير متاح'))
   932→        || /insufficient.*balance|not enough/i.test(errorMsg);
   933→
   934→      const shouldForceClose = isPaperTrading || isExchangeUnreachable || isInsufficientBalance || isUserCancel;
   935→
   936→      if (shouldForceClose) {
   937→        this.logger.warn(
   938→          `⚡ Exchange close failed for ${position.symbol} — attempting force close (DB only). ` +
   939→          `Reason: ${isPaperTrading ? 'paper-trading' : isExchangeUnreachable ? 'exchange-unreachable' : isInsufficientBalance ? 'insufficient-balance' : 'user-cancel'}. ` +
   940→          `Error: ${errorMsg}`,
   941→        );
   942→        try {
   943→          return await this.forceClosePosition(
   944→            userId,
   945→            position.id,
   946→            `Auto force-close: ${isPaperTrading ? 'paper-trading position' : isExchangeUnreachable ? 'exchange unreachable' : isInsufficientBalance ? 'insufficient balance' : 'position likely already closed'} — ${errorMsg}`,
   947→            ipAddress,
   948→            userAgent,
   949→          );
   950→        } catch (forceErr: any) {
   951→          this.logger.error(
   952→            `❌ Force close also failed for ${position.id}: ${forceErr.message}`,
   953→          );
   954→        }
   955→      }
   956→
   957→      // FIX V114: For paper-trading positions, NEVER leave them stuck OPEN.
   958→      // Paper trading has no real exchange state to desync — force-close is always safe.
   959→      // If force-close already failed above, try ONE MORE TIME with a direct DB update
   960→      // as the absolute last resort. This prevents the "3 out of 4 close" pattern where
   961→      // the last position gets stuck because of a transient error during force-close.
   962→      if (isPaperTrading) {
   963→        this.logger.warn(
   964→          `🔴 V114 Paper-trading safety net: force-close failed for ${position.id}, attempting direct DB update as last resort`,
   965→        );
   966→        try {
   967→          await this.prisma.position.update({
   968→            where: { id: position.id },
   969→            data: {
   970→              status: 'CLOSED',
   971→              closedAt: new Date(),
   972→              realizedPnl: posRealizedPnl, // No PnL change since execution failed
   973→              exitPrice: posEntryPrice, // V140: Fallback to entry price when execution failed
   974→              closeReason: request.closeReason || 'FORCE_CLOSE', // V141
   975→            },
   976→          });
   977→          this._clearProcessedKeysForPosition(userId, position.symbol).catch(() => {});
   978→          this.logger.log(
   979→            `🔴 V114 Paper-trading position ${position.id} force-closed via direct DB update`,
   980→          );
   981→          return {
   982→            order: null,
   983→            pnl: 0,
   984→            position: await this.prisma.position.findUnique({ where: { id: position.id } }),
   985→            forceClosed: true,
   986→            safetyNetClose: true,
   987→          };
   988→        } catch (dbErr: any) {
   989→          this.logger.error(
   990→            `❌ V114 Even direct DB update failed for ${position.id}: ${dbErr.message}`,
   991→          );
   992→        }
   993→      }
   994→
   995→      // Only throw for real trading positions where the exchange might have
   996→      // partially executed — force-closing would lose sync with exchange state.
   997→      throw new BadRequestException(
   998→        `فشل في إغلاق المركز: ${execution.error}`,
   999→      );
  1000→    }
  1001→
  1002→    // Safe exit price: use explicit null checks instead of || operator
  1003→    // The || operator treats 0 as falsy, causing incorrect fallback to entry price
  1004→    const exitPrice =
  1005→      execution.averagePrice != null && execution.averagePrice > 0
  1006→        ? execution.averagePrice
  1007→        : (posCurrentPrice != null && posCurrentPrice > 0
  1008→          ? posCurrentPrice
  1009→          : posEntryPrice);
  1010→
  1011→    // FIX: Deduct actual trading fees from PnL.
  1012→    // For real exchanges, execution.fee contains the actual fee charged.
  1013→    // For paper trading, fee is simulated at 0.1% per leg (entry + exit).
  1014→    const grossPnl =
  1015→      position.side === 'BUY'
  1016→        ? (exitPrice - posEntryPrice) * closeQuantity
  1017→        : (posEntryPrice - exitPrice) * closeQuantity;
  1018→    // FIX V174: Only deduct EXIT fee from PnL.
  1019→    // Entry fee was already paid when the position opened (_executePaperTrade
  1020→    // charges 0.1% at entry and records it in the ENTRY trade).
  1021→    // Adding entryFeeEstimate here caused double-counting — charging entry fee twice.
  1022→    // BTC example: exitFee=$3.90 + entryFeeEstimate=$3.89 = $7.79 extra deduction.
  1023→    const exitFee = execution.fee ?? (exitPrice * closeQuantity * 0.001);
  1024→    const totalFees = exitFee; // exit fee only — entry fee already charged at open
  1025→    const pnl = grossPnl - totalFees;
  1026→
  1027→    // Record closing order, exit trade, and update position — all in one transaction
  1028→    // FIX: Optimistic locking — use version check in WHERE clause to prevent
  1029→    // concurrent close from double-executing. If version changed (another close
  1030→    // committed first), we retry the whole closePosition method.
  1031→    const { order: closedOrder } = await this.prisma.$transaction(async (tx) => {
  1032→      // FIX: Wrap order creation in try-catch — if it fails (e.g., idempotency collision),
  1033→      // still close the position. A failed audit trail is better than a stuck open position.
  1034→      let order: any = null;
  1035→      try { order = await tx.order.create({
  1036→        data: {
  1037→          userId,
  1038→          exchangeCredentialId: position.credentialId,
  1039→          exchange: position.exchange,
  1040→          symbol: position.symbol,
  1041→          side: closeSide,
  1042→          type: 'MARKET' as PrismaOrderType,
  1043→          status: 'FILLED' as PrismaOrderStatus,
  1044→          quantity: closeQuantity,
  1045→          stopLoss: posStopLoss,
  1046→          filledQuantity: execution.filledQuantity || closeQuantity,
  1047→          averagePrice: execution.averagePrice,
  1048→          fee: execution.fee ?? null,
  1049→          feeCurrency: execution.feeCurrency ?? null,
  1050→          exchangeOrderId: execution.exchangeOrderId,
  1051→          idempotencyKey: `close-${position.id}-${Date.now()}-${Math.random().toString(36).slice(2)}`,
  1052→        },
  1053→      }); } catch (orderErr: any) {
  1054→        this.logger.warn(`closePosition: Order creation failed (non-critical): ${orderErr.message}`);
  1055→        order = { id: 'manual-close-' + Date.now() };
  1056→      }
  1057→
  1058→      // Record exit trade
  1059→      await tx.trade.create({
  1060→        data: {
  1061→          userId,
  1062→          orderId: order.id,

  1225→   * The key format is: smart-executor:processed:{briefId}:{userId}
  1226→   * Since briefId is opaque, we scan all keys for this userId, get each key's value,
  1227→   * and check if the associated position's symbol matches the closed position.
  1228→   *
  1229→   * We also clear the corresponding DB-persisted keys (smart-executor:processed:*:userId:db).
  1230→   */
  1231→  private async _clearProcessedKeysForPosition(userId: string, symbol: string): Promise<void> {
  1232→    try {
  1233→      const pattern = `smart-executor:processed:*:${userId}`;
  1234→      const keys = await this.redis.scanKeys(pattern);
  1235→      let cleared = 0;
  1236→
  1237→      for (const key of keys) {
  1238→        try {
  1239→          // The key format is: smart-executor:processed:{briefId}:{userId}
  1240→          // Extract the briefId from the key
  1241→          const parts = key.split(':');
  1242→          // Expected: ['smart-executor', 'processed', briefId, userId]
  1243→          const briefId = parts.length >= 4 ? parts[2] : null;
  1244→
  1245→          if (!briefId) continue;
  1246→

  1248→          const brief = await this.prisma.tradingBrief.findUnique({
  1249→            where: { id: briefId },
  1250→            select: { pair: true },
  1251→          });

  1255→            await this.redis.del(key);

  1259→                where: { key: `${key}:db` },

  1261→            } catch { /* non-critical */ }
  1262→            cleared++;
  1263→            this.logger.debug(
  1264→              `🗑️ Cleared processedKey ${key} for closed position ${symbol} (user: ${userId})`,
  1265→            );
  1266→          }
  1267→        } catch (keyErr: any) {
  1268→          this.logger.warn(
  1269→            `Failed to check/clear processed key ${key}: ${keyErr.message}`,
  1270→          );
  1271→        }
  1272→      }
  1273→
  1274→      if (cleared > 0) {

  1283→    }
  1284→  }

  1287→   * FIX V114: Close position with automatic retry on optimistic lock failure
  1288→   * AND transient errors (timeout, network, rate limit).
  1289→   *
  1290→   * Previously, this only retried on OPTIMISTIC_LOCK_FAILURE. But many
  1291→   * transient errors (CCXT timeout, network reset, rate limit) also benefit
  1292→   * from a retry — especially for paper-trading positions where a retry
  1293→   * is always safe (no real exchange state to desync).
  1294→   *
  1295→   * Retry schedule:
  1296→   * - OPTIMISTIC_LOCK_FAILURE: 100ms delay (fast — the other tx just committed)
  1297→   * - Transient errors: 1s, 2s, 3s exponential backoff
  1298→   */
  1299→  async closePositionWithRetry(
  1300→    userId: string,
  1301→    request: ClosePositionRequest,
  1302→    ipAddress?: string,
  1303→    userAgent?: string,
  1304→    maxRetries = 3,
  1305→  ): Promise<any> {
  1306→    try {
  1307→      return await this.closePosition(userId, request, ipAddress, userAgent, 0);
  1308→    } catch (error: any) {
  1309→      const errMsg = error?.message || '';
  1310→
  1311→      // Check if this is an optimistic lock failure — fast retry
  1312→      if (errMsg.includes('OPTIMISTIC_LOCK_FAILURE') && maxRetries > 0) {
  1313→        this.logger.warn(`Optimistic lock failure on closePosition for ${request.positionId} — retrying (${maxRetries} attempts left)`);
  1314→        await new Promise(resolve => setTimeout(resolve, 100));
  1315→        return this.closePositionWithRetry(userId, request, ipAddress, userAgent, maxRetries - 1);
  1316→      }
  1317→
  1318→      // FIX V114: Also retry on transient errors (timeout, network, rate limit)
  1319→      // These are temporary failures that may succeed on retry.
  1320→      const isTransientError = /timeout|ETIMEDOUT|ECONNREFUSED|ECONNRESET|rate.?limit|too many|429|network|unreachable|fetch failed|Service Unavailable|502|504/i.test(errMsg);
  1321→      if (isTransientError && maxRetries > 0) {
  1322→        const delayMs = (4 - maxRetries) * 1000; // 1s, 2s, 3s backoff
  1323→        this.logger.warn(`Transient error on closePosition for ${request.positionId} — retrying in ${delayMs}ms (${maxRetries} attempts left). Error: ${errMsg.substring(0, 100)}`);
  1324→        await new Promise(resolve => setTimeout(resolve, delayMs));
  1325→        return this.closePositionWithRetry(userId, request, ipAddress, userAgent, maxRetries - 1);
  1326→      }
  1327→
  1328→      throw error;
  1329→    }
