   101→    }
   102→
   103→    if (this.isMonitoring) {
   104→      return; // Skip if previous cycle still running
   105→    }
   106→
   107→    this.isMonitoring = true;
   108→
   109→    try {
   110→      // ═══════════════════════════════════════════════════════════
   111→      // RLS BYPASS: Background service queries across ALL users.
   112→      // We must enable RLS bypass to access positions from all users.
   113→      // After the monitor cycle, we disable bypass to restore isolation.
   114→      // ═══════════════════════════════════════════════════════════
   115→      await this.prisma.enableRlsBypass();

   117→      // Step 1: Get all open positions
   118→      // ROOT FIX: Include ALL positions regardless of source or exchange.
   119→      // Previously, positions from 'smart_executor', 'agent', 'paper_trading'
   120→      // were EXCLUDED from monitoring. This meant:
   121→      //   1. Auto-traded positions NEVER had their SL/TP checked → stayed open forever
   122→      //   2. Paper-trading positions were never monitored → no automated exits
   123→      //   3. Smart Executor "only 1 trade" bug — first position stays open forever,
   124→      //      blocking all new trades when user hits maxOpenPositions
   125→      //
   126→      // Now: ALL positions are monitored for SL/TP using real prices.
   127→      //
   128→      // V141 FIX: Skip positions with source='agent' — the Agent monitors its
   129→      // own positions via _monitorOpenPositions (every 60s). Having both systems
   130→      // monitor the same positions causes:
   131→      //   1. Race conditions (double-close attempts, OPTIMISTIC_LOCK_FAILURE errors)
   132→      //   2. Trailing stop overrides (Position Monitor modifies Agent position SL)
   133→      //   3. Attribution confusion (which system triggered the close?)
   134→      // Each system manages its own positions independently.
   135→      // V141 FIX: Skip SL/TP monitoring and trailing stop for Agent positions
   136→      // (the Agent monitors its own positions via _monitorOpenPositions every 60s).
   137→      // V143 FIX: But DO allow price/PnL updates for Agent positions — otherwise
   138→      // the dashboard shows stale prices for Agent positions for up to 60 seconds.
   139→      //
   140→      // Previously (V141), Agent positions were completely excluded from monitoring.
   141→      // This meant their currentPrice and unrealizedPnl were only updated when the
   142→      // Agent cycle ran (every 60s). Now we split:
   143→      //   - Agent positions: price/PnL updates ONLY (no SL/TP check, no trailing stop)
   144→      //   - Other positions: full monitoring (SL/TP, trailing stop, price updates)
   145→      let positions: any[];
   146→      let agentPositions: any[];
   147→      let nonAgentPositions: any[];

   150→        nonAgentPositions = await this.prisma.position.findMany({
   151→          where: {
   152→            status: 'OPEN',
   153→            entryPrice: { gt: 0 },

   156→        });
   157→
   158→        // V175 FIX: Agent positions NOW get full SL/TP monitoring
   159→        // Previously agent monitored its own positions every 60s only
   160→        // This caused TP/SL misses when agent was busy with council sessions
   161→        // Now position-monitor handles SL/TP for ALL positions every 10s
   162→        agentPositions = await this.prisma.position.findMany({
   163→          where: {
   164→            status: 'OPEN',
   165→            entryPrice: { gt: 0 },
   166→            source: 'agent',
   167→          },
   168→        });
   169→
   170→        positions = [...nonAgentPositions, ...agentPositions];
   171→      } catch (dbError: any) {
   172→        // Table may not exist yet (e.g., Prisma db:push hasn't run or Position model is new)
   173→        if (dbError.message?.includes('does not exist')) {
   174→          this.logger.warn('🛡️ Position table not found — skipping monitor cycle. Run `prisma db push` to create it.');
   175→          return;
   176→        }
   177→        throw dbError;
   178→      }
   179→
   180→      if (positions.length === 0) {
   181→        return;
   182→      }
   183→
   184→      this.logger.debug(`🛡️ Monitoring ${positions.length} open positions`);
   185→
   186→      let slTriggered = 0;
   187→      let tpTriggered = 0;
   188→      let trailingUpdated = 0;
   189→      let alertsSent = 0;

   203→        const quoteResult = quotes[i];

   205→          quoteResult.status === 'fulfilled' && quoteResult.value?.price
   206→            ? quoteResult.value.price
   207→            : null;
   208→
   209→        try {
   210→          const result = await this._monitorPosition(position, currentPrice, priceUpdates);
   211→          if (result.slTriggered) slTriggered++;
   212→          if (result.tpTriggered) tpTriggered++;
   213→          if (result.trailingUpdated) trailingUpdated++;
   214→          if (result.alertSent) alertsSent++;
   215→        } catch (error: any) {
   216→          this.logger.error(
   217→            `🛡️ Monitor error for position ${position.id}: ${error.message}`,
   218→          );
   219→        }
   220→      }
   221→
   222→      // Step 4: Batch update positions that only need price/PnL updates (no SL/TP hit)
   223→      if (priceUpdates.length > 0) {
   224→        try {
   225→          await this.prisma.$transaction(priceUpdates);
   226→          this.logger.debug(`🛡️ Batch updated ${priceUpdates.length} position price/PnL records`);
   227→        } catch (error: any) {
   228→          this.logger.error(`🛡️ Batch price update failed: ${error.message}`);
   229→        }
   230→      }
   231→
   232→      if (slTriggered > 0 || tpTriggered > 0 || trailingUpdated > 0) {
   233→        this.logger.log(
   234→          `🛡️ Monitor cycle: ${slTriggered} SL, ${tpTriggered} TP, ${trailingUpdated} trailing, ${alertsSent} alerts`,
   235→        );
   236→      }
   237→
   238→      // Store monitor stats
   239→      await this.redis.set(
   240→        'monitor:last_cycle',
   241→        JSON.stringify({
   242→          timestamp: new Date().toISOString(),
   243→          positionsMonitored: positions.length,
   244→          slTriggered,
   245→          tpTriggered,
   246→          trailingUpdated,
   247→          alertsSent,
   248→        }),
   249→        300000, // 5 min TTL
   250→      );
   251→    } catch (error: any) {
   252→      this.logger.error(`🛡️ Position monitor cycle failed: ${error.message}`);
   253→      // V185: الشفاء الذاتي — تسجيل الفشل
   254→      this.selfHealing?.reportFailure('position-monitor', error.message);
   255→    } finally {
   256→      this.isMonitoring = false;
   257→      // RLS: Disable bypass after background service completes
   258→      await this.prisma.disableRlsBypass().catch(() => {});
   259→      // V185: الشفاء الذاتي — تسجيل النجاح
   260→      this.selfHealing?.reportSuccess('position-monitor');
   261→    }
   262→  }

   264→  /**
   265→   * Get monitor status
   266→   */
   267→  async getMonitorStatus(): Promise<{
   268→    lastCycle: any;
   269→    openPositions: number;
   270→    nearSL: number;
   271→    nearTP: number;
   272→  }> {
   273→    const lastCycleRaw = await this.redis.get('monitor:last_cycle');
   274→    const lastCycle = lastCycleRaw ? JSON.parse(lastCycleRaw) : null;
   275→
   276→    let openPositions = 0;
   277→    let nearSL = 0;
   278→    let nearTP = 0;
   279→
   280→    try {
   281→      // FIX: Previously ran count() AND findMany() as two separate queries
   282→      // for the same filter — N+1 pattern. Now combined into a single query.
   283→      const allPositions = await this.prisma.position.findMany({
   284→        where: { status: 'OPEN' },
   285→      });
   286→      openPositions = allPositions.length;
   287→
   288→      // Fetch all quotes in parallel — deduplicate by symbol first
   289→      // FIX: Previously fetched a quote for EACH position, even if multiple
   290→      // positions shared the same symbol. Now deduplicates by symbol.
   291→      const uniqueSymbols = [...new Set(allPositions.map(p => p.symbol))];
   292→      const quoteMap = new Map<string, any>();
   293→      const quotePromises = uniqueSymbols.map(async (symbol: string) => {
   294→        try {
   295→          const quote = await this.exchangeService.getQuote(symbol);
   296→          if (quote?.price) quoteMap.set(symbol, quote);
   297→        } catch { /* ignore individual quote failures */ }
   298→      });
   299→      await Promise.allSettled(quotePromises);
   300→
   301→      for (const pos of allPositions) {
   302→        const quote = quoteMap.get(pos.symbol);
   303→        if (!quote?.price) continue;
   304→
   305→        const currentPrice = quote.price;
   306→
   307→        if (pos.stopLoss) {
   308→          const slDistance = Math.abs(currentPrice - pos.stopLoss.toNumber()) / pos.entryPrice.toNumber();
   309→          if (slDistance < 0.01) nearSL++;
   310→        }
   311→
   312→        if (pos.takeProfit) {
   313→          const tpDistance = Math.abs(currentPrice - pos.takeProfit.toNumber()) / pos.entryPrice.toNumber();
   314→          if (tpDistance < 0.01) nearTP++;
   315→        }
   316→      }
   317→    } catch (dbError: any) {
   318→      if (dbError.message?.includes('does not exist')) {
   319→        this.logger.warn('🛡️ Position table not found — returning empty monitor status.');
   320→      } else {
   321→        throw dbError;
   322→      }
   323→    }
   324→
   325→    return { lastCycle, openPositions, nearSL, nearTP };
   326→  }

   328→  // ── Private: Position Monitoring ──
   329→
   330→  private async _monitorPosition(position: any, currentPrice: number | null, priceUpdates: any[]): Promise<{
   331→    slTriggered: boolean;
   332→    tpTriggered: boolean;
   333→    trailingUpdated: boolean;
   334→    alertSent: boolean;
   335→  }> {
   336→    const result = {
   337→      slTriggered: false,
   338→      tpTriggered: false,
   339→      trailingUpdated: false,
   340→      alertSent: false,
   341→    };
   342→
   343→    // Use pre-fetched price or skip
   344→    if (currentPrice === null) {
   345→      return result; // Skip if can't get price
   346→    }
   347→
   348→    // V143: For Agent positions, ONLY update price/PnL — no SL/TP checks,
   349→    // no trailing stop modifications. The Agent manages its own SL/TP exits.
   350→    const isAgentPosition = position.source === 'agent';
   351→
   352→    // FIX: Convert Prisma Decimal fields to numbers for safe comparison.
   353→    // Prisma Decimal objects don't compare correctly with JS `<=` / `>=` operators.
   354→    const entryPrice = position.entryPrice?.toNumber?.() ?? Number(position.entryPrice);
   355→    const quantity = position.quantity?.toNumber?.() ?? Number(position.quantity);
   356→    const stopLossNum = position.stopLoss?.toNumber?.() ?? (position.stopLoss ? Number(position.stopLoss) : null);
   357→    const takeProfitNum = position.takeProfit?.toNumber?.() ?? (position.takeProfit ? Number(position.takeProfit) : null);
   358→
   359→    // Calculate unrealized P&L
   360→    const unrealizedPnl =
   361→      position.side === 'BUY'
   362→        ? (currentPrice - entryPrice) * quantity
   363→        : (entryPrice - currentPrice) * quantity;
   364→
   365→    const pnlPercent = (unrealizedPnl / (entryPrice * quantity)) * 100;
   366→
   367→    // ── V187 FIX: Agent SL/TP + MAX_HOLDING check ──
   368→    // Previously, Agent positions did early return after SL/TP check,
   369→    // which meant MAX_HOLDING_TIME was NEVER checked for Agent positions.
   370→    // This caused Agent positions to stay open indefinitely (V184 removed
   371→    // the Agent's own 4h close, and the Position Monitor never checked
   372→    // holding time for Agents due to the early return).
   373→    //
   374→    // Now: Agent positions check SL/TP first (like before), but then
   375→    // fall through to the MAX_HOLDING_TIME check (48h for Agents).
   376→    if (isAgentPosition) {
   377→      // SL check for agent
   378→      if (stopLossNum !== null) {
   379→        const agentSlHit = position.side === 'BUY' ? currentPrice <= stopLossNum : currentPrice >= stopLossNum;
   380→        if (agentSlHit) {
   381→          this.logger.warn(`🚨 AGENT SL HIT: ${position.symbol} @ ${currentPrice} (SL: ${stopLossNum})`);
   382→          await this._closePosition(position, currentPrice, 'STOP_LOSS');
   383→          // V180 FIX: Set cooldown after Agent SL to prevent immediate re-open
   384→          try {
   385→            const cooldownKey = `cooldown:${position.userId}:${position.symbol}`;
   386→            await this.redis.set(cooldownKey, 'STOP_LOSS', this.COOLDOWN_TTL_MS);
   387→          } catch { /* non-critical */ }
   388→          this._checkSanctuary(position.userId).catch(() => {});
   389→          result.slTriggered = true;
   390→          return result;
   391→        }
   392→      }
   393→      // TP check for agent
   394→      if (takeProfitNum !== null) {
   395→        const agentTpHit = position.side === 'BUY' ? currentPrice >= takeProfitNum : currentPrice <= takeProfitNum;
   396→        if (agentTpHit) {
   397→          this.logger.warn(`🎯 AGENT TP HIT: ${position.symbol} @ ${currentPrice} (TP: ${takeProfitNum})`);
   398→          await this._closePosition(position, currentPrice, 'TAKE_PROFIT');
   399→          // V180 FIX: Set cooldown after Agent TP too
   400→          try {
   401→            const cooldownKey = `cooldown:${position.userId}:${position.symbol}`;
   402→            await this.redis.set(cooldownKey, 'TAKE_PROFIT', this.COOLDOWN_TTL_MS);
   403→          } catch { /* non-critical */ }
   404→          this._checkSanctuary(position.userId).catch(() => {});
   405→          result.tpTriggered = true;
   406→          return result;
   407→        }
   408→      }
   409→      // No SL/TP hit — update price/PnL and highest/lowest, then fall through to MAX_HOLDING check
   410→      priceUpdates.push(
   411→        this.prisma.position.update({
   412→          where: { id: position.id },
   413→          data: {
   414→            currentPrice,
   415→            unrealizedPnl,
   416→            highestPrice:
   417→              position.side === 'BUY'
   418→                ? Math.max(position.highestPrice || currentPrice, currentPrice)
   419→                : position.highestPrice || currentPrice,
   420→            lowestPrice:
   421→              position.side === 'SELL'
   422→                ? Math.min(position.lowestPrice || currentPrice, currentPrice)
   423→                : position.lowestPrice || currentPrice,
   424→          },
   425→        }),
   426→      );
   427→      // V187: DO NOT return here — fall through to MAX_HOLDING_TIME check below
   428→      // Agent positions should be checked for 48h max holding time.
   429→    }
   430→
   431→    // ── Below: Full monitoring for ALL positions (including Agent) ──
   432→
   433→    // ── V176 FIX: Stale paper-trading position detector ──
   434→    // Issue #10: Paper positions without SL/TP were never auto-closed.
   435→    // They stayed open for 131+ hours, blocking new trades and showing
   436→    // stale P&L. Now: any paper position older than 48h without SL/TP
   437→    // is automatically closed with reason='STALE_POSITION'.
   438→    if (position.exchange === 'paper-trading' && position.openedAt) {
   439→      const holdingMs = Date.now() - new Date(position.openedAt).getTime();
   440→      const holdingHours = holdingMs / (60 * 60 * 1000);
   441→      const hasSLTP = stopLossNum !== null || takeProfitNum !== null;
   442→
   443→      if (holdingHours > this.STALE_PAPER_POSITION_MAX_HOURS && !hasSLTP) {
   444→        this.logger.warn(
   445→          `🔴 V176 STALE POSITION: ${position.symbol} held ${holdingHours.toFixed(1)}h without SL/TP — auto-closing`,
   446→        );
   447→        await this._closePosition(position, currentPrice, 'STALE_POSITION');
   448→        // V180 FIX: Set cooldown after STALE_POSITION close to prevent immediate re-open
   449→        try {
   450→          const cooldownKey = `cooldown:${position.userId}:${position.symbol}`;
   451→          await this.redis.set(cooldownKey, 'STALE_POSITION', this.COOLDOWN_TTL_MS);
   452→        } catch { /* non-critical */ }
   453→        result.slTriggered = true;
   454→        return result;
   455→      }
   456→    }
   457→
   458→    // ── V176 FIX: Cooldown check after auto-close ──
   459→    // Issue #11: After TIME_EXPIRED auto-close, the SmartExecutor immediately
   460→    // re-opened the same position, creating a loop of trades every 8-10 seconds.
   461→    // Now: after auto-close, the same userId+symbol is blocked for 5 minutes.
   462→    try {
   463→      const cooldownKey = `cooldown:${position.userId}:${position.symbol}`;
   464→      const cooldownReason = await this.redis.get(cooldownKey);
   465→      if (cooldownReason) {
   466→        this.logger.debug(
   467→          `⏳ V176 COOLDOWN: ${position.symbol} blocked for user ${position.userId} (reason: ${cooldownReason})`,
   468→        );
   469→        // Just update price — don't trigger any SL/TP/trailing while in cooldown
   470→        priceUpdates.push(
   471→          this.prisma.position.update({
   472→            where: { id: position.id },
   473→            data: { currentPrice, unrealizedPnl },
   474→          }),
   475→        );
   476→        return result;
   477→      }
   478→    } catch { /* non-critical */ }
   479→
   480→    // ── MAX_HOLDING_TIME: Smart Executor + Agent positions ──
   481→    // Counter-trend (reversal): 45 دقيقة
   482→    // Smart Executor مع الاتجاه: 4 ساعات
   483→    // Agent: 48 ساعة (swing trading) — V214: HARD ENFORCED, no fallback to 4h
   484→    if ((position.source === 'smart_executor' || position.source === 'agent') && position.openedAt) {
   485→      const holdingMs = Date.now() - new Date(position.openedAt).getTime();
   486→
   487→      const isAgent = position.source === 'agent';
   488→
   489→      // V214 HARD ENFORCEMENT: Agent positions MUST get 48h — no exceptions.
   490→      // Even if timeframe=null or Redis is empty, Agent always gets 48h.

   832→            position.side === 'SELL'
   833→              ? Math.min(position.lowestPrice || currentPrice, currentPrice)
   834→              : position.lowestPrice || currentPrice,
   835→        },
   836→      }),
   837→    );
   838→
   839→    return result;
   840→  }

   842→  private async _closePosition(
   843→    position: any,
   844→    currentPrice: number,
   845→    reason: 'STOP_LOSS' | 'TAKE_PROFIT' | 'TIME_EXPIRED' | 'STALE_POSITION',
   846→  ): Promise<void> {
   847→    try {
   848→      // FIX: Use closePositionWithRetry + convert Decimal to number
   849→      // V141: Pass closeReason so it's stored on the Position record
   850→      await this.tradingService.closePositionWithRetry(
   851→        position.userId,
   852→        {
   853→          positionId: position.id,
   854→          quantity: typeof position.quantity?.toNumber === 'function' 
   855→            ? position.quantity.toNumber() 
   856→            : Number(position.quantity),
   857→          closeReason: reason, // V141: STOP_LOSS or TAKE_PROFIT
   858→        },
   859→        undefined,
   860→        undefined,
   861→        3, // max retries for OPTIMISTIC_LOCK_FAILURE
   862→      );
   863→
   864→      await this.audit.log({
   865→        userId: position.userId,
   866→        action: `POSITION_CLOSED_${reason}`,
   867→        resource: 'position-monitor',
   868→        details: JSON.stringify({
   869→          positionId: position.id,
   870→          symbol: position.symbol,
   871→          closePrice: currentPrice,
   872→          entryPrice: position.entryPrice,
   873→          side: position.side,
   874→          quantity: position.quantity,
   875→        }),
   876→      });
   877→
   878→      // V176: Record the trade closed event for real-time performance monitoring
   879→      // This is fail-safe — if it errors, we log and continue (never block trading)
   880→      try {
   881→        const entryPrice = position.entryPrice?.toNumber?.() ?? Number(position.entryPrice);
   882→        const quantity = position.quantity?.toNumber?.() ?? Number(position.quantity);
   883→        await this.performanceEvents.recordTradeClosed({
   884→          userId: position.userId,
   885→          symbol: position.symbol,
   886→          side: position.side,
   887→          source: position.source || 'unknown',
   888→          pnl: position.side === 'BUY'
   889→            ? (currentPrice - entryPrice) * quantity
   890→            : (entryPrice - currentPrice) * quantity,
   891→          entryPrice,
   892→          exitPrice: currentPrice,
   893→          quantity,
   894→          openedAt: position.openedAt ? new Date(position.openedAt) : new Date(),
   895→          closedAt: new Date(),
   896→          closeReason: reason,
   897→        });
   898→      } catch (err: any) {
   899→        this.logger.debug(`Failed to record trade closed event: ${err.message}`);
   900→      }
   901→
   902→      // V185: مجلة التداول — تسجيل إغلاق الصفقة لتغذية حلقة التعلم
   903→      // هذا هو المفتاح الرئيسي: بدون هذا السطر، حلقة journal → accuracy → memory لا تحصل على بيانات أبداً
   904→      try {
   905→        if (this.journal) {
   906→          const entryPrice = position.entryPrice?.toNumber?.() ?? Number(position.entryPrice);
   907→          const quantity = position.quantity?.toNumber?.() ?? Number(position.quantity);
   908→          const pnl = position.side === 'BUY'
   909→            ? (currentPrice - entryPrice) * quantity
   910→            : (entryPrice - currentPrice) * quantity;
   911→          const pnlPct = (pnl / (entryPrice * quantity)) * 100;
   912→
   913→          await this.journal.recordTradeClose(
   914→            position.id,
   915→            currentPrice,
   916→            pnl,
   917→            pnlPct,
   918→            { tags: [reason] },
   919→          );
   920→        }
   921→      } catch (journalErr: any) {
   922→        this.logger.debug(`V185 Journal: Failed to record trade close: ${journalErr.message}`);
   923→      }
   924→    } catch (error: any) {
   925→      this.logger.error(
   926→        `🛡️ Failed to close position ${position.id}: ${error.message}`,
   927→      );
   928→
   929→      // FIX V114: If closePositionWithRetry failed, try force-close as fallback.
   930→      // This is critical for SL/TP-triggered closes — if the close fails, the
   931→      // position stays OPEN with its SL/TP already triggered, which could lead
   932→      // to unlimited losses. Force-close is always safe for paper-trading positions.
   933→      // For real exchange positions, force-close is safe when the error suggests
   934→      // the exchange might already have the position closed.
   935→      try {
   936→        this.logger.warn(
   937→          `🛡️ V114 Attempting force-close for position ${position.id} after closePositionWithRetry failed`,
   938→        );
   939→        await this.tradingService.forceClosePosition(
   940→          position.userId,
   941→          position.id,
   942→          `V114 Position Monitor fallback: ${reason} triggered but closePositionWithRetry failed — ${error.message?.substring(0, 100)}`,
   943→        );
   944→        this.logger.log(
   945→          `🛡️ V114 Force-close succeeded for position ${position.id} (${reason})`,
   946→        );
   947→      } catch (forceErr: any) {
   948→        this.logger.error(
   949→          `🛡️ V114 Force-close also failed for position ${position.id}: ${forceErr.message}`,
   950→        );
   951→      }
   952→    }
   953→  }

   955→  private _calculateTrailingStop(position: any, currentPrice: number): number | null {
   956→    if (position.side === 'BUY') {
   957→      // For long positions, trail below the highest price
   958→      const highestPrice = position.highestPrice || currentPrice;
   959→      const newHigh = Math.max(highestPrice, currentPrice);
   960→      return newHigh * (1 - this.TRAILING_DISTANCE_PCT);
   961→    } else if (position.side === 'SELL') {
   962→      // For short positions, trail above the lowest price
   963→      const lowestPrice = position.lowestPrice || currentPrice;
   964→      const newLow = Math.min(lowestPrice, currentPrice);
   965→      return newLow * (1 + this.TRAILING_DISTANCE_PCT);
   966→    }
   967→    return null;
   968→  }

   971→    userId: string,
   972→    type: string,

   976→    await this.redis.set(

   978→      JSON.stringify({ type, data, timestamp: new Date().toISOString() }),
   979→      86400000, // 24 hours
   980→    );

   983→   * يحسب MAX_HOLDING بحسب الإطار الزمني للصفقة

   986→   * H1/H4   → 48 ساعة  (swing)
   987→   * D1/W1   → 7 أيام   (position)
   988→   * Agent   → 48 ساعة  (swing default)
   989→   */
   990→  private _getMaxHoldingMs(timeframe: string | null, isAgent: boolean, positionId?: string, symbol?: string): number {
   991→    const H = 60 * 60 * 1000;
   992→    let maxHoldingMs: number;
   993→    let reason: string;
   994→
   995→    if (isAgent) {
   996→      maxHoldingMs = 48 * H;
   997→      reason = 'Agent → 48h';
   998→    } else if (!timeframe) {
   999→      maxHoldingMs = 8 * H;
  1000→      reason = 'No timeframe → 8h default';
  1001→    } else {
  1002→      const tf = timeframe.toUpperCase();
  1003→      if (tf === 'M1' || tf === 'M5') { maxHoldingMs = 4 * H; reason = `${tf} → 4h`; }
  1004→      else if (tf === 'M15' || tf === 'M30') { maxHoldingMs = 12 * H; reason = `${tf} → 12h`; }
  1005→      else if (tf === 'H1' || tf === 'H2' || tf === 'H4') { maxHoldingMs = 48 * H; reason = `${tf} → 48h`; }
  1006→      else if (tf === 'D1' || tf === 'D3') { maxHoldingMs = 7 * 24 * H; reason = `${tf} → 7d`; }
  1007→      else if (tf === 'W1' || tf === 'W2') { maxHoldingMs = 14 * 24 * H; reason = `${tf} → 14d`; }
  1008→      else { maxHoldingMs = 8 * H; reason = `Unknown TF ${tf} → 8h fallback`; }
  1009→    }
  1010→
  1011→    // V213 DIAGNOSTIC: Log every MAX_HOLDING calculation for Agent positions
  1012→    // This helps diagnose why Agent positions close at 4h instead of 48h.
  1013→    if (isAgent || (positionId && symbol)) {
  1014→      this.logger.log(
  1015→        `🛡️ V213 _getMaxHoldingMs: ${symbol || '?'} id=${positionId?.slice(0,12) || '?'}... ` +
  1016→        `isAgent=${isAgent} timeframe=${timeframe || 'null'} → maxHolding=${(maxHoldingMs / H).toFixed(0)}h (${reason})`
  1017→      );
  1018→    }
  1019→
  1020→    return maxHoldingMs;
  1021→  }
