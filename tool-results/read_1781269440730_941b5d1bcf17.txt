   153→
   154→  /**
   155→   * Startup cleanup: Purge ONLY phantom/stale data, preserving legitimate user states.
   156→   *
   157→   * FIX: Previous version deleted ALL data on every restart, including:
   158→   *   - User executor states that were explicitly enabled
   159→   *   - Valid TradingBriefs from the Strategic Council
   160→   *   - Legitimate trade history
   161→   *   - AgentSettings that users intentionally configured
   162→   *
   163→   * New behavior:
   164→   *   - Only purges PHANTOM positions (zero-value / stale > 24h)
   165→   *   - Only purges EXPIRED TradingBriefs (past their validUntil)
   166→   *   - Preserves DB-persisted user states (users explicitly enabled them)
   167→   *   - Preserves AgentSettings (user configuration)
   168→   *   - Still clears Redis states (volatile, will be restored from DB)
   169→   */
   170→  private async _startupCleanup(): Promise<void> {
   171→    try {
   172→      // SUSTAINABLE FIX: Skip cleanup if DB is not available.
   173→      // Each query on an unavailable DB creates a new connection pool,
   174→      // leaking PostgreSQL connection slots and causing cascading failures.
   175→      if (!this.prisma?.isAvailable?.()) {
   176→        this.logger.warn('⚔️ Skipping startup cleanup — DB not yet available');
   177→        return;
   178→      }
   179→
   180→      this.logger.log('⚔️ Running startup phantom cleanup (preserving user data)...');
   181→
   182→      // ── STEP 1: Clean phantom/stale positions (zero-value only) ──
   183→      try {
   184→        const purgeResult = await this.purgePhantomPositions();
   185→        if (purgeResult.deleted > 0) {
   186→          this.logger.log(`⚔️ STARTUP: Purged ${purgeResult.deleted} phantom (zero-value) position(s)`);
   187→        }
   188→      } catch (purgeErr: any) {
   189→        this.logger.warn(`⚔️ Startup phantom purge failed (non-critical): ${purgeErr.message}`);
   190→      }
   191→
   192→      // ── STEP 2: Auto-close stale paper-trading positions (>24h) ──
   193→      try {
   194→        const staleClosed = await this._autoCloseStalePaperPositions();
   195→        if (staleClosed > 0) {
   196→          this.logger.log(`⚔️ STARTUP: Auto-closed ${staleClosed} stale paper position(s) (>24h)`);
   197→        }
   198→      } catch (staleErr: any) {
   199→        this.logger.warn(`⚔️ Startup stale cleanup failed (non-critical): ${staleErr.message}`);
   200→      }
   201→
   202→      // ── STEP 3: Delete only EXPIRED TradingBriefs (not all) ──
   203→      // Previous code deleted ALL briefs, but valid briefs should be kept
   204→      // so the executor can continue working after a restart.
   205→      try {
   206→        const deletedBriefs = await this.prisma.tradingBrief.deleteMany({
   207→          where: {
   208→            OR: [
   209→              { expiresAt: { lt: new Date() } },  // Expired briefs
   210→              { isActive: false },                  // Deactivated briefs
   211→            ],
   212→          },
   213→        });
   214→        if (deletedBriefs.count > 0) {
   215→          this.logger.log(`⚔️ STARTUP: Purged ${deletedBriefs.count} expired TradingBrief(s) (preserving active ones)`);
   216→        }
   217→      } catch (briefErr: any) {
   218→        this.logger.warn(`⚔️ Failed to purge expired TradingBrief records: ${briefErr.message}`);
   219→      }
   220→
   221→      // ── STEP 4: Clear Redis user states (volatile cache only) ──
   222→      // V140B FIX: Only clear Redis cache, NOT DB states!
   223→      // Previously this deleted BOTH Redis and DB user states on every restart,
   224→      // which meant users had to manually re-enable the executor after EVERY
   225→      // NestJS restart (crash, deployment, Railway cycling). This caused the
   226→      // executor to "stop after a few minutes" because NestJS restarts frequently.
   227→      //
   228→      // Now: Redis cache is cleared (volatile, will be re-populated from DB),
   229→      // but DB states are PRESERVED so auto-restore can re-enable the user.
   230→      try {
   231→        const userKeys = await this.redis.scanKeys(`${this.REDIS_USER_STATE_PREFIX}*`);
   232→        for (const key of userKeys) {
   233→          await this.redis.del(key);
   234→        }
   235→        if (userKeys.length > 0) {
   236→          this.logger.log(`⚔️ STARTUP: Cleared ${userKeys.length} volatile Redis user state(s) (DB states preserved)`);
   237→        }
   238→      } catch (redisErr: any) {
   239→        this.logger.warn(`⚔️ Failed to clear executor Redis states: ${redisErr.message}`);
   240→      }
   241→
   242→      // ── STEP 4.1: V140B — REMOVED destructive DB cleanup ──
   243→      // Previously deleted ALL SMART_EXECUTOR_USER_STATE:* from DB on startup.
   244→      // This was needed in V136 to clean up auto-created states from AuthService,
   245→      // but now that AuthService no longer auto-creates states (V136 removed it),
   246→      // ALL remaining DB states are from users who EXPLICITLY enabled the executor.
   247→      // Deleting them caused the "executor stops after a few minutes" bug.
   248→      //
   249→      // Instead: Only delete states that have enabled=false (already disabled).
   250→      try {
   251→        const disabledStates = await this.prisma.setting.deleteMany({
   252→          where: {
   253→            key: { startsWith: this.DB_USER_STATE_KEY },
   254→            value: { contains: '"enabled":false' },
   255→          },
   256→        });
   257→        if (disabledStates.count > 0) {
   258→          this.logger.log(`⚔️ STARTUP: Cleaned up ${disabledStates.count} already-disabled DB executor state(s)`);
   259→        }
   260→      } catch (dbCleanErr: any) {
   261→        this.logger.warn(`⚔️ Failed to clean up disabled DB user states: ${dbCleanErr.message}`);
   262→      }
   263→
   264→      // ── STEP 5: Clear global executor state from Redis ──
   265→      try {
   266→        await this.redis.del(this.REDIS_GLOBAL_STATE);
   267→      } catch {}
   268→
   269→      // ── STEP 5.1: Clear stale position-lock keys from Redis (V130) ──
   270→      // Before V130, ExposureManager.canOpenPosition() created Redis locks
   271→      // (position-lock:userId:symbol) that were never released. These stale
   272→      // locks can persist after deployment and block ALL trade execution.
   273→      // Even though we no longer use canOpenPosition() in the execution path,
   274→      // clearing these on startup ensures a clean state.
   275→      try {
   276→        const lockKeys = await this.redis.scanKeys('position-lock:*');
   277→        for (const key of lockKeys) {
   278→          await this.redis.del(key);
   279→        }
   280→        if (lockKeys.length > 0) {
   281→          this.logger.log(`⚔️ STARTUP: Cleared ${lockKeys.length} stale position-lock key(s) from Redis (V130 fix)`);
   282→        }
   283→      } catch (lockErr: any) {
   284→        this.logger.warn(`⚔️ Failed to clear position-lock keys: ${lockErr.message}`);
   285→      }
   286→
   287→      // ── STEP 5.2: Clear stale idempotency keys from Redis (V132) ──
   288→      // V132 changed the idempotency key structure. Old cross-source keys
   289→      // (sha256(userId:symbol:side)) can block valid re-execution. Clear them.
   290→      try {
   291→        const idempotencyKeys = await this.redis.scanKeys('idempotency:*');
   292→        for (const key of idempotencyKeys) {
   293→          await this.redis.del(key);
   294→        }
   295→        if (idempotencyKeys.length > 0) {
   296→          this.logger.log(`⚔️ STARTUP: Cleared ${idempotencyKeys.length} stale idempotency key(s) from Redis (V132 fix)`);
   297→        }
   298→      } catch (idempErr: any) {
   299→        this.logger.warn(`⚔️ Failed to clear idempotency keys: ${idempErr.message}`);
   300→      }
   301→
   302→      // ── STEP 5.3: Clear OLD format circuit breaker keys from Redis (V137) ──
   303→      // #18: Consolidated into TradeCoordinationService.cleanupV1CircuitBreakerKeys()
   304→      // This ensures a single cleanup implementation shared by SmartExecutor and Agent.
   305→      try {
   306→        const oldCbCleaned = await this.tradeCoordination.cleanupV1CircuitBreakerKeys();
   307→        if (oldCbCleaned > 0) {
   308→          this.logger.log(`⚔️ STARTUP: Cleared ${oldCbCleaned} old-format circuit breaker key(s) via TradeCoordinationService (V137/#18)`);
   309→        }
   310→      } catch (cbCleanErr: any) {
   311→        this.logger.warn(`⚔️ Failed to clear old circuit breaker keys: ${cbCleanErr.message}`);
   312→      }
   313→
   314→      // ── STEP 5.5: Clear stale price cache from Redis ──
   315→      // FIX: If the old Promise.any() bug cached wrong prices (e.g., $34.98 for BTC),
   316→      // those stale entries can persist for up to 5 minutes in the fallback cache.
   317→      // On startup, purge ALL price-related Redis keys so fresh data is fetched.
   318→      try {
   319→        const priceCachePatterns = [
   320→          'fallback:quote:*',   // FreeFallbackAdapter cache (5min TTL)
   321→          'fallback:lastprice:*', // FreeFallbackAdapter poisoned cache (24h TTL)
   322→          'binance:quote:*',    // BinanceAdapter cache (3s TTL)
   323→          'twelvedata:quote:*', // TwelveDataAdapter cache (10min TTL)
   324→          'exchange:quote:*',   // ExchangeService cache (30s TTL)
   325→          'aggregator:quote:*', // AggregatorService cache
   326→        ];
   327→        let purgedPriceKeys = 0;
   328→        for (const pattern of priceCachePatterns) {
   329→          try {
   330→            const keys = await this.redis.scanKeys(pattern);
   331→            for (const key of keys) {
   332→              await this.redis.del(key);
   333→              purgedPriceKeys++;
   334→            }
   335→          } catch { /* pattern scan failed — skip */ }
   336→        }
   337→        if (purgedPriceKeys > 0) {
   338→          this.logger.log(`⚔️ STARTUP: Purged ${purgedPriceKeys} stale price cache key(s) from Redis`);
   339→        }
   340→      } catch (priceErr: any) {
   341→        this.logger.warn(`⚔️ Failed to purge price cache: ${priceErr.message}`);
   342→      }
   343→
   344→      // ── STEP 6: Delete stale AutonomousTrade records (>7 days old) ──
   345→      try {
   346→        const sevenDaysAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
   347→        const deletedAutoTrades = await this.prisma.autonomousTrade.deleteMany({
   348→          where: { createdAt: { lt: sevenDaysAgo } },
   349→        });
   350→        if (deletedAutoTrades.count > 0) {
   351→          this.logger.log(`⚔️ STARTUP: Purged ${deletedAutoTrades.count} stale AutonomousTrade(s) (>7 days)`);
   352→        }
   353→      } catch (autoTradeErr: any) {
   354→        this.logger.warn(`⚔️ Failed to purge stale AutonomousTrade records: ${autoTradeErr.message}`);
   355→      }
   356→
   357→      // ── STEP 7: Delete stale PaperOrder records (>7 days old) ──
   358→      try {
   359→        const sevenDaysAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
   360→        const deletedPaperOrders = await this.prisma.paperOrder.deleteMany({
   361→          where: { createdAt: { lt: sevenDaysAgo } },
   362→        });
   363→        if (deletedPaperOrders.count > 0) {
   364→          this.logger.log(`⚔️ STARTUP: Purged ${deletedPaperOrders.count} stale PaperOrder(s) (>7 days)`);
   365→        }
   366→      } catch (paperErr: any) {
   367→        this.logger.warn(`⚔️ Failed to purge stale PaperOrder records: ${paperErr.message}`);
   368→      }
   369→
   370→      // ── STEP 8: V143 — Clean up stale processedKey DB entries ──
   371→      // The Setting table accumulates `smart-executor:processed:*:db` entries
   372→      // for every brief execution. These never get cleaned up and grow indefinitely.
   373→      // Briefs expire within 24h max, so any processedKey older than 48h is stale.
   374→      try {
   375→        const twoDaysAgo = new Date(Date.now() - 2 * 24 * 60 * 60 * 1000);
   376→        const staleProcessedKeys = await this.prisma.setting.deleteMany({
   377→          where: {
   378→            key: { startsWith: this.REDIS_PROCESSED_PREFIX },
   379→            updatedAt: { lt: twoDaysAgo },
   380→          },
   381→        });
   382→        if (staleProcessedKeys.count > 0) {
   383→          this.logger.log(`⚔️ V143 STARTUP: Purged ${staleProcessedKeys.count} stale processedKey DB entries (>48h old)`);
   384→        }
   385→      } catch (processedKeyErr: any) {
   386→        this.logger.warn(`⚔️ V143 Failed to purge stale processedKey entries: ${processedKeyErr.message}`);
   387→      }
   388→
   389→      // ── REMOVED: Auto-enable for paper-trading users ──
   390→      // ROOT FIX: Auto-enabling the Smart Executor for ALL users with paper-trading
   391→      // credentials violates explicit user consent. It causes:
   392→      //   1. Trades executing without the user's knowledge
   393→      //   2. Both Agent AND Executor running simultaneously (duplicate trades)
   394→      //   3. Users confused about why trades appear they didn't authorize
   395→      // The executor MUST ONLY be enabled by explicit user action ("تشغيل" button).
   396→      //
   397→      // Cross-system coordination is handled by ExposureManagerService:
   398→      //   - Tracks total open positions across BOTH systems (executor + agent)
   399→      //   - Prevents exceeding global limits regardless of source
   400→      //   - One position per symbol enforced at the exposure level
   401→      //   - No mutual exclusion lock needed — exposure-based coordination
   402→
   403→      this.logger.log('⚔️ Startup cleanup complete (user data preserved)');
   404→
   405→      // ── V140B: Auto-restore explicitly-enabled users from DB ──
   406→      // After cleanup, re-populate Redis from DB for users who had
   407→      // explicitly enabled the executor (enabled: true). This prevents
   408→      // the "executor stops after a few minutes" bug caused by NestJS
   409→      // restarts (Railway cycling, crashes, deployments).
   410→      //
   411→      // This is SAFE because V136 already removed AuthService auto-creation.

   491→   * where one user stopping the executor kills it for ALL users.
   492→   * The tick loop is lightweight (just reads Redis keys) when no users are enabled.

   494→  async start(userId?: string): Promise<ExecutorStatus> {

   526→   * Stop the Smart Executor globally
   527→   * FIX: Made PRIVATE — This method should ONLY be called by the system
   528→   * (e.g., module destroy, nuclear cleanup). It must NEVER be triggered by
   529→   * individual user actions, because it kills the tick loop for ALL users.
   530→   *
   531→   * Individual users should use enableUser/disableUser instead.

   537→  private async stop(userId?: string): Promise<ExecutorStatus> {
   538→    // FIX: Check if there are still enabled users before stopping.

   540→    const enabledUsers = await this._getEnabledUsers();
   541→    const otherUsersEnabled = enabledUsers.some(id => id !== userId);
   542→
   543→    if (otherUsersEnabled) {
   544→      this.logger.warn(`⚔️ Cannot stop executor — ${enabledUsers.length} user(s) still enabled. Only individual disable is allowed.`);
   545→      return this.getStatus();
   546→    }
   547→
   548→    if (!this.isRunning) {
   549→      return this.getStatus();

   558→
   559→    this.logger.log('⚔️ Smart Executor STOPPED — no enabled users remain');

   561→    await this.redis.set(

   567→    await this.audit.log({
   568→      userId: userId || 'system',

   570→      resource: 'smart-executor',
   571→      details: JSON.stringify({ stoppedAt: new Date() }),

   577→  /**
   578→   * Enable executor for a specific user

   581→   *
   582→   * The user selects which account to trade on from their settings page.
   583→   * The executor reads activeCredentialId from user settings and executes

   587→  async enableUser(userId: string, config?: {
   588→    maxOpenPositions?: number;
   589→    riskPerTradePercent?: number;
   590→  }): Promise<UserExecutorState> {
   591→    // ── Auto-start the executor if it's not running ──
   592→    if (!this.isRunning) {
   593→      this.logger.log(`⚔️ Executor not running — auto-starting on behalf of user ${userId}`);
   594→      try {
   595→        await this.start(userId);
   596→      } catch (error: any) {
   597→        this.logger.warn(`⚔️ Failed to auto-start executor for user ${userId}: ${error.message}`);
   598→      }
   599→    }
   600→
   601→    // ── Read activeCredentialId from user settings (the user chose it) ──
   602→    let activeCredentialId: string | undefined;
   603→    try {
   604→      const activeSetting = await this.prisma.setting.findFirst({
   605→        where: { key: `user:${userId}:activeCredentialId` },
   606→      });
   607→      if (activeSetting?.value) {
   608→        activeCredentialId = activeSetting.value;
   609→        this.logger.log(`⚔️ V126 User ${userId} has active account: ${activeCredentialId}`);
   610→      } else {
   611→        this.logger.log(`⚔️ V126 User ${userId} has no active account selected — will skip execution until one is set`);
   612→      }
   613→    } catch (err: any) {
   614→      this.logger.warn(`⚔️ V126 Could not read activeCredentialId for user ${userId}: ${err.message}`);
   615→    }
   616→
   617→    // ── Read risk settings from user's saved preferences. ──
   618→    const userRiskSettings = await this._loadUserRiskSettings(userId);
   619→
   620→    // V135: Read credential metadata for display (isTestnet, isPaperTrading, exchangeName)
   621→    let isPaperTrading = false;
   622→    let isTestnet = false;
   623→    let exchangeName: string | undefined;
   624→    if (activeCredentialId) {
   625→      try {
   626→        const cred = await this.prisma.exchangeCredential.findFirst({
   627→          where: { id: activeCredentialId, userId },
   628→          select: { testnet: true, exchange: true },
   629→        });
   630→        if (cred) {
   631→          isPaperTrading = cred.exchange === 'paper-trading';
   632→          isTestnet = cred.testnet === true && cred.exchange !== 'paper-trading';
   633→          exchangeName = cred.exchange;
   634→        } else {
   635→          // CRITICAL FIX: activeCredentialId points to a deleted credential.
   636→          // Auto-select a new valid credential for this user instead of
   637→          // silently falling through to paper trading mode.
   638→          this.logger.warn(
   639→            `⚔️ CRITICAL: activeCredentialId=${activeCredentialId} not found for user ${userId} ` +
   640→            `— credential was deleted. Auto-selecting new credential.`
   641→          );
   642→          // Clear stale setting
   643→          try {
   644→            await this.prisma.setting.deleteMany({
   645→              where: { key: `user:${userId}:activeCredentialId` },
   646→            });
   647→          } catch { /* ignore */ }
   648→          // Find a new valid credential
   649→          const newCred = await this.prisma.exchangeCredential.findFirst({
   650→            where: { userId, isValid: true },
   651→            orderBy: { createdAt: 'desc' },
   652→            select: { id: true, testnet: true, exchange: true },
   653→          });
   654→          if (newCred) {
   655→            activeCredentialId = newCred.id;
   656→            isPaperTrading = newCred.exchange === 'paper-trading';
   657→            isTestnet = newCred.testnet === true && newCred.exchange !== 'paper-trading';
   658→            exchangeName = newCred.exchange;
   659→            // Persist the new selection
   660→            await this.prisma.setting.upsert({
   661→              where: { key: `user:${userId}:activeCredentialId` },
   662→              update: { value: newCred.id },
   663→              create: { key: `user:${userId}:activeCredentialId`, value: newCred.id },
   664→            });
   665→            this.logger.log(
   666→              `⚔️ Auto-selected new credential ${newCred.id} (${newCred.exchange}) for user ${userId}`
   667→            );
   668→          } else {
   669→            activeCredentialId = undefined;
   670→            isPaperTrading = true;
   671→          }
   672→        }
   673→      } catch (err: any) {
   674→        this.logger.warn(`⚔️ V135 Could not read credential metadata for user ${userId}: ${err.message}`);
   675→      }
   676→    } else {
   677→      isPaperTrading = true;  // No credential = paper mode
   678→    }
   679→
   680→    const state: UserExecutorState = {
   681→      enabled: true,
   682→      dailyPnL: 0,
   683→      dailyTrades: 0,
   684→      dailyResetAt: new Date().toISOString(),
   685→      lastTradeAt: null,
   686→      consecutiveLosses: 0,
   687→      maxOpenPositions: config?.maxOpenPositions || userRiskSettings.maxOpenPositions,
   688→      riskPerTradePercent: config?.riskPerTradePercent || userRiskSettings.riskPerTradePercent,
   689→      activeCredentialId,
   690→      isPaperTrading,  // V135
   691→      isTestnet,       // V135
   692→      exchangeName,    // V135
   693→    };
   694→
   695→    // ── Persist to BOTH Redis AND Database ──
   696→    await this.redis.set(
   697→      `${this.REDIS_USER_STATE_PREFIX}${userId}`,
   698→      JSON.stringify(state),
   699→      86400000 * 7,
   700→    );
   701→
   702→    await this._persistUserStateToDB(userId, state);
   703→
   704→    this.logger.log(
   705→      `⚔️ V126 Executor enabled for user ${userId} ` +
   706→      `(activeCredential: ${activeCredentialId || 'none'}) — saved to Redis + DB`
   707→    );
   708→
   709→    await this.audit.log({
   710→      userId,
   711→      action: 'SMART_EXECUTOR_USER_ENABLED',
   712→      resource: 'smart-executor',
   713→      details: JSON.stringify(state),
   714→    });
   715→
   716→    return state;
   717→  }

   792→  /**
   793→   * Get current executor status
   794→   */
   795→  async getStatus(userId?: string): Promise<ExecutorStatus> {
   796→    let todayExecutions = 0;
   797→    let todayPnL = 0;
   798→    let openPositions = 0;
   799→    let activeBriefs = 0;
   800→
   801→    // FIX: Separate try-catch blocks so one failing query doesn't prevent others
   802→    // The previous single try-catch meant that if Position table didn't exist,
   803→    // the activeBriefs count would never be reached, always showing 0.
   804→
   805→    try {
   806→      const startOfDay = new Date();
   807→      startOfDay.setHours(0, 0, 0, 0);
   808→
   809→      // FIX: Count executions from BOTH AuditLog AND Trade table.
   810→      // Previously, only AuditLog entries with action='SMART_EXECUTOR_TRADE' were counted.
   811→      // But trades can be executed by multiple sources (smart_executor, agent, auto_paper),
   812→      // and the AuditLog approach misses trades executed by other paths.
   813→      // Now we also count Trade records where source is 'smart_executor' or 'auto_paper'.
   814→
   815→      // 1. Count from AuditLog (primary — created by _executeBriefForUser)
   816→      const auditWhere: any = {
   817→        action: 'SMART_EXECUTOR_TRADE',
   818→        createdAt: { gte: startOfDay },
   819→      };
   820→      if (userId) auditWhere.userId = userId;
   821→      const todayLogs = await this.prisma.auditLog.findMany({ where: auditWhere });
   822→      todayExecutions = todayLogs.length;
   823→
   824→      // 2. If AuditLog count is 0, try counting from Trade table as fallback
   825→      // This catches trades that were executed but didn't create an AuditLog entry
   826→      if (todayExecutions === 0) {
   827→        try {
   828→          const tradeWhere: any = {
   829→            source: { in: ['smart_executor', 'auto_paper'] },
   830→            type: 'ENTRY',
   831→            executedAt: { gte: startOfDay },
   832→          };
   833→          if (userId) tradeWhere.userId = userId;
   834→          todayExecutions = await this.prisma.trade.count({ where: tradeWhere });
   835→        } catch (tradeErr: any) {
   836→          this.logger.debug(`getStatus: trade count fallback failed: ${tradeErr.message}`);
   837→        }
   838→      }
   839→    } catch (e: any) {
   840→      this.logger.debug(`getStatus: auditLog query failed: ${e.message}`);
   841→    }
   842→
   843→    try {
   844→      const posWhere: any = { status: 'OPEN' };
   845→      if (userId) {
   846→        posWhere.userId = userId;
   847→      } else {
   848→        // If no userId provided (global status), only count positions with valid trade value
   849→        // to avoid phantom positions inflating the count
   850→        posWhere.AND = [
   851→          { entryPrice: { gt: 0 } },
   852→        ];
   853→      }
   854→      openPositions = await this.prisma.position.count({ where: posWhere });
   855→    } catch (e: any) {
   856→      this.logger.debug(`getStatus: position count failed: ${e.message}`);
   857→    }
   858→
   859→    try {
   860→      activeBriefs = await this.councilService.getActiveBriefsCount();
   861→    } catch (e: any) {
   862→      this.logger.debug(`getStatus: activeBriefs count failed: ${e.message}`);
   863→    }
   864→
   865→    // FIX: Calculate todayPnL from Trade table instead of hardcoding 0
   866→    // Previously, todayPnL was always 0 because there was no calculation.
   867→    // Now we sum up PnL from closed trades (EXIT type) by the executor today.
   868→    try {
   869→      const startOfDay = new Date();
   870→      startOfDay.setHours(0, 0, 0, 0);
   871→      const pnlWhere: any = {
   872→        type: { in: ['EXIT', 'PARTIAL_EXIT'] },
   873→        executedAt: { gte: startOfDay },
   874→        pnl: { not: null },
   875→      };
   876→      if (userId) pnlWhere.userId = userId;
   877→      const pnlTrades = await this.prisma.trade.findMany({
   878→        where: pnlWhere,
   879→        select: { pnl: true },
   880→      });
   881→      todayPnL = pnlTrades.reduce((sum, t) => sum + (Number(t.pnl) || 0), 0);
   882→    } catch (e: any) {
   883→      this.logger.debug(`getStatus: todayPnL calculation failed: ${e.message}`);
   884→    }
   885→
   886→    // Check if daily loss limit reached
   887→    // V135 FIX: Previously, this checked ALL users including simulated accounts
   888→    // (paper/testnet). Simulated accounts should bypass the daily loss limit
   889→    // (matching _processUserBriefs behavior). Without this bypass:
   890→    //   - Paper/testnet accounts would show "daily limit exceeded" even with 0 trades
   891→    //   - The widget would show "حد يومي" status incorrectly
   892→    //   - This was the root cause of the user's reported bug: "لماذا يعطيني تم
   893→    //     تجاوز الحد اليومي وهو لم ينفذ اصلا"
   894→    let dailyLossLimitReached = false;
   895→    try {
   896→      const threshold = this.config.maxDailyLossPercent; // default 5%
   897→
   898→      if (userId) {
   899→        // User-specific check
   900→        const userState = await this.getUserState(userId);
   901→        if (userState && userState.enabled) {
   902→          // V135: Check if user's active credential is simulated (paper/testnet)
   903→          let isSimulated = false;
   904→          try {
   905→            const activeCredId = userState.activeCredentialId;
   906→            if (activeCredId) {
   907→              const cred = await this.prisma.exchangeCredential.findFirst({
   908→                where: { id: activeCredId, userId },
   909→                select: { testnet: true, exchange: true },
   910→              });
   911→              if (cred && (cred.testnet === true || this._isSimulatedExchange(cred.exchange))) {
   912→                isSimulated = true;
   913→              }
   914→            } else {
   915→              // No credential = paper mode by default
   916→              isSimulated = true;
   917→            }
   918→          } catch (credErr: any) {
   919→            this.logger.debug(`getStatus: could not check credential type for ${userId}: ${credErr.message}`);
   920→          }
   921→
   922→          // V135: SKIP daily loss limit check for simulated accounts
   923→          if (!isSimulated) {
   924→            const portfolio = await this._getPortfolioValue(userId);
   925→            if (portfolio > 0) {
   926→              const lossLimit = portfolio * (threshold / 100);
   927→              dailyLossLimitReached = todayPnL < -lossLimit;
   928→            }
   929→          } else {
   930→            this.logger.debug(`getStatus: daily limit check BYPASSED for user ${userId} — simulated account`);
   931→          }
   932→        }
   933→      } else {
   934→        // Global check: check if ANY enabled user has hit the daily loss limit
   935→        const enabledUsers = await this._getEnabledUsers();
   936→        for (const uid of enabledUsers) {
   937→          try {
   938→            const userState = await this.getUserState(uid);
   939→            if (userState && userState.enabled) {

  1470→  /**
  1471→   * Start the monitoring tick loop
  1472→   * Each tick: read active briefs → check prices → execute if conditions met for enabled users
  1473→   */
  1474→  private _startTickLoop(): void {
  1475→    this.tickInterval = setInterval(async () => {
  1476→      if (!this.isRunning || this.isTicking) return;  // FIX: Prevent concurrent ticks
  1477→
  1478→      this.isTicking = true;
  1479→      try {
  1480→        await this._tick();
  1481→      } catch (error: any) {
  1482→        this.logger.error(`⚔️ Tick error: ${error.message}`);
  1483→      } finally {
  1484→        this.isTicking = false;
  1485→      }
  1486→    }, this.config.tickIntervalMs);
  1487→
  1488→    // ═══════════════════════════════════════════════════════════════════
  1489→    // V132: Subscribe to council completion events for immediate execution.
  1490→    // Previously, the executor waited for the next tick (up to 10s) after
  1491→    // a council session completed. Now it's notified immediately via Redis
  1492→    // pub/sub, reducing latency from "up to 10s" to "near-instant".
  1493→    // ═══════════════════════════════════════════════════════════════════
  1494→    try {
  1495→      const subscriber = this.redis.duplicateSubscriber();
  1496→      if (subscriber) {
  1497→        subscriber.subscribe('council:session_complete');
  1498→        subscriber.on('message', (channel: string, message: string) => {
  1499→          if (channel === 'council:session_complete' && this.isRunning && !this.isTicking) {
  1500→            this.logger.log('⚔️ Council session complete event received — triggering immediate tick');
  1501→            this.isTicking = true;
  1502→            this._tick()
  1503→              .catch((err: any) => this.logger.error(`⚔️ Event-triggered tick failed: ${err.message}`))
  1504→              .finally(() => { this.isTicking = false; });
  1505→          }
  1506→        });
  1507→        this.logger.log('⚔️ Subscribed to council:session_complete events');
  1508→      }
  1509→    } catch (subErr: any) {
  1510→      this.logger.warn(`⚔️ Could not subscribe to council events: ${subErr.message} — falling back to polling`);
  1511→    }
  1512→  }

  1514→  /**
  1515→   * Single tick: Get active briefs, find enabled users, check conditions per user
  1516→   */
  1517→  private async _tick(): Promise<void> {
  1518→    // ═══════════════════════════════════════════════════════════════════
  1519→    // V134: Use CONSOLIDATED briefs — ONE direction per pair.
  1520→    //
  1521→    // ROOT CAUSE FIX for "opened and closed after 1 second":
  1522→    //   getActiveBriefs() returns ALL briefs including conflicting
  1523→    //   directions for the same pair (M1=BUY, M5=SELL). Processing
  1524→    //   these sequentially caused: open BUY → close BUY → open SELL → loop.
  1525→    //
  1526→    //   getConsolidatedBriefs() merges conflicting briefs into ONE
  1527→    //   direction per pair (weighted vote by confidence * timeframe).
  1528→    //   This guarantees the executor sees at most ONE brief per pair.
  1529→    // ═══════════════════════════════════════════════════════════════════
  1530→    let consolidatedBriefs: TradingBriefDTO[] = [];
  1531→    try {
  1532→      consolidatedBriefs = await this.councilService.getConsolidatedBriefs();
  1533→    } catch (e: any) {
  1534→      this.logger.error(`⚔️ Failed to get consolidated briefs: ${e.message}`);
  1535→      return;
  1536→    }
  1537→
  1538→    if (consolidatedBriefs.length === 0) {
  1539→      this.logger.debug('⚔️ No consolidated briefs to execute — waiting for Strategic Council');
  1540→      return;
  1541→    }
  1542→
  1543→    // ── TIMEFRAME FILTER: Only process briefs for executor timeframes (M1/M5/M15) ──
  1544→    // The Smart Executor handles quick/scalping trades only.
  1545→    // Briefs for M30+ are handled by the Autonomous Agent.
  1546→    const executorBriefs = consolidatedBriefs.filter(
  1547→      (brief: TradingBriefDTO) => isExecutorTimeframe(brief.timeframe)
  1548→    );
  1549→
  1550→    if (executorBriefs.length === 0) {
  1551→      this.logger.debug(
  1552→        `⚔️ ${consolidatedBriefs.length} consolidated briefs but none match executor timeframes [${EXECUTOR_TIMEFRAMES.join(',')}] — waiting`,
  1553→      );
  1554→      return;
  1555→    }
  1556→
  1557→    this.logger.debug(`⚔️ V134 Tick: ${executorBriefs.length} consolidated executor briefs (one per pair, no conflicts)`);
  1558→
  1559→    // Get users with executor enabled
  1560→    const enabledUsers = await this._getEnabledUsers();
  1561→
  1562→    if (enabledUsers.length === 0) {
  1563→      this.logger.debug(`⚔️ ${executorBriefs.length} consolidated briefs available but no enabled users — skipping`);
  1564→      return;
  1565→    }
  1566→
  1567→    this.logger.debug(`⚔️ V134 Tick: ${executorBriefs.length} consolidated executor briefs, ${enabledUsers.length} users`);
  1568→
  1569→    // Process each enabled user
  1570→    for (const userId of enabledUsers) {
  1571→      try {

  1573→      } catch (error: any) {

  1575→      }
  1576→    }
  1577→  }
